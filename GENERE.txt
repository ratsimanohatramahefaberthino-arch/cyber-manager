using System.Text.Json;
using System.Text.Json.Serialization;

public class AgentConfig
{
    [JsonPropertyName("server_url")]
    public string ServerUrl { get; set; } = "http://127.0.0.1:8000";

    [JsonPropertyName("nom_poste")]
    public string NomPoste { get; set; } = Environment.MachineName;

    [JsonPropertyName("identifiant_unique")]
    public string IdentifiantUnique { get; set; } = "";

    [JsonPropertyName("agent_token")]
    public string AgentToken { get; set; } = "";

    [JsonIgnore]
    public static string ConfigPath
    {
        get
        {
            var dir = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                "CyberManagerAgent");
            Directory.CreateDirectory(dir);
            return Path.Combine(dir, "agent-config.json");
        }
    }

    public static AgentConfig LoadOrCreate()
    {
        if (File.Exists(ConfigPath))
        {
            try
            {
                var json = File.ReadAllText(ConfigPath);
                var config = JsonSerializer.Deserialize<AgentConfig>(json);
                if (config != null)
                {
                    if (string.IsNullOrEmpty(config.IdentifiantUnique))
                    {
                        config.IdentifiantUnique = Guid.NewGuid().ToString();
                        config.Save();
                    }
                    return config;
                }
            }
            catch { }
        }

        var nouveau = new AgentConfig
        {
            IdentifiantUnique = Guid.NewGuid().ToString()
        };
        nouveau.Save();
        return nouveau;
    }

    public void Save()
    {
        var options = new JsonSerializerOptions { WriteIndented = true };
        File.WriteAllText(ConfigPath, JsonSerializer.Serialize(this, options));
    }
}

using System.Net;
using System.Net.NetworkInformation;
using System.Net.Sockets;
using System.Text;
using System.Text.Json;
using System.Text.Json.Serialization;

public enum ApiStatut { Ok, NonAutorise, Injoignable, Erreur }

public record ApiReponse<T>(ApiStatut Statut, T? Donnees = default);

public record ActivationResult(bool Succes, string Message, string? Avertissement = null)
{
    public static ActivationResult Echec(string message) => new(false, message);
}

/// <summary>Résultat d'une action de session demandée depuis le widget (suspendre / quitter).</summary>
public record ResultatAction(bool Succes, string Message, int? Montant = null);

public record InfosReseau(string Ip, string Mac);

public class ApiClient
{
    public const string Version = "1.2.0";

    private readonly HttpClient _http;
    private readonly AgentConfig _config;
    private InfosReseau _reseau = new("", "");
    private DateTime _reseauLuLe = DateTime.MinValue;

    public ApiClient(AgentConfig config)
    {
        _config = config;
        _http = new HttpClient { Timeout = TimeSpan.FromSeconds(60) };
    }

    /// <summary>IP et MAC de la carte réseau réellement utilisée pour joindre le serveur.</summary>
    public InfosReseau Reseau
    {
        get
        {
            if (DateTime.UtcNow - _reseauLuLe > TimeSpan.FromSeconds(60))
            {
                _reseau = LireReseau(_config.ServerUrl);
                _reseauLuLe = DateTime.UtcNow;
            }
            return _reseau;
        }
    }

    // ------------------------------------------------------------------
    // Appels API
    // ------------------------------------------------------------------

    public async Task<ApiStatut> HandshakeAsync()
    {
        var net = Reseau;
        var payload = new
        {
            identifiant_unique = _config.IdentifiantUnique,
            nom_poste = _config.NomPoste,
            adresse_ip = net.Ip,
            adresse_mac = net.Mac,
            agent_version = Version
        };

        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, "/api/agent/handshake", payload, avecToken: false);
        if (statut != ApiStatut.Ok) return statut;

        try
        {
            using var doc = JsonDocument.Parse(corps!);
            var root = doc.RootElement;
            var token = root.TryGetProperty("agent_token", out var t) ? t.GetString() : null;
            var nom = root.TryGetProperty("nom_poste", out var n) ? n.GetString() : null;

            if (string.IsNullOrEmpty(token)) return ApiStatut.Erreur;

            var modifie = token != _config.AgentToken;
            _config.AgentToken = token;
            if (!string.IsNullOrWhiteSpace(nom) && nom != _config.NomPoste)
            {
                _config.NomPoste = nom;   // le serveur fait foi pour le nom
                modifie = true;
            }
            if (modifie) _config.Save();
            return ApiStatut.Ok;
        }
        catch { return ApiStatut.Erreur; }
    }

    public async Task<ApiReponse<HeartbeatResponse>> SendHeartbeatAsync()
    {
        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, "/api/agent/heartbeat",
            new { adresse_ip = Reseau.Ip });
        if (statut != ApiStatut.Ok) return new(statut);

        try
        {
            var hb = JsonSerializer.Deserialize<HeartbeatResponse>(corps!);
            return hb is { Ok: true } ? new(ApiStatut.Ok, hb) : new(ApiStatut.Erreur);
        }
        catch { return new(ApiStatut.Erreur); }
    }

    public async Task<ApiReponse<List<AgentCommand>>> GetCommandsAsync()
    {
        var (statut, corps) = await EnvoyerAsync(HttpMethod.Get, "/api/agent/commandes");
        if (statut != ApiStatut.Ok) return new(statut);

        try
        {
            using var doc = JsonDocument.Parse(corps!);
            var commandes = new List<AgentCommand>();
            if (doc.RootElement.TryGetProperty("commandes", out var arr))
            {
                foreach (var item in arr.EnumerateArray())
                {
                    commandes.Add(new AgentCommand
                    {
                        Id = item.GetProperty("id").GetInt32(),
                        Type = item.GetProperty("type").GetString() ?? "",
                        Payload = item.TryGetProperty("payload", out var p) && p.ValueKind != JsonValueKind.Null
                            ? p.GetRawText() : null
                    });
                }
            }
            return new(ApiStatut.Ok, commandes);
        }
        catch { return new(ApiStatut.Erreur); }
    }

    public async Task ConfirmerCommandeAsync(int id, bool succes, string message = "")
    {
        await EnvoyerAsync(HttpMethod.Post, $"/api/agent/commandes/{id}/confirmer", new { succes, message });
    }

    /// <summary>Demande d'activation (bouton ACTIVER). Le serveur crée la session et connecte le poste.</summary>
    public async Task<ActivationResult> ActiverAsync()
    {
        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, "/api/agent/activer", new { });

        switch (statut)
        {
            case ApiStatut.Injoignable:
                return ActivationResult.Echec("Serveur injoignable. Réessayez dans quelques instants.");

            case ApiStatut.NonAutorise:
                _config.AgentToken = "";   // forcera un nouveau handshake
                return ActivationResult.Echec("Poste en cours de reconnexion au serveur. Réessayez dans 5 secondes.");

            case ApiStatut.Erreur:
                return ActivationResult.Echec(LireChamp(corps, "message") ?? "Activation refusée par le serveur.");

            default:
                return new ActivationResult(true,
                    LireChamp(corps, "message") ?? "Poste activé.",
                    LireChamp(corps, "avertissement"));
        }
    }

        /// <summary>Le client suspend sa session depuis le widget : chrono figé côté serveur.</summary>
    public Task<ResultatAction> SuspendreSessionAsync() =>
        ActionSessionAsync("/api/agent/session/suspendre", "Session suspendue.");

    /// <summary>Le client quitte depuis le widget : session clôturée, montant calculé, réseau coupé.</summary>
    public Task<ResultatAction> FermerSessionAsync() =>
        ActionSessionAsync("/api/agent/session/fermer", "Session terminée.");


    /// <summary>Le client écrit à l'accueil (chat client → admin).</summary>
    public Task<ResultatAction> EnvoyerMessageAsync(string texte) =>
        ActionSessionAsync("/api/agent/message", "Message envoyé.", new { texte });

    private async Task<ResultatAction> ActionSessionAsync(string chemin, string messageParDefaut, object? payload = null)
    {
        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, chemin, payload ?? new { });

        switch (statut)
        {
            case ApiStatut.Injoignable:
                return new ResultatAction(false, "Serveur injoignable. Réessayez dans quelques instants.");

            case ApiStatut.NonAutorise:
                _config.AgentToken = "";   // forcera un nouveau handshake
                return new ResultatAction(false, "Poste en cours de reconnexion au serveur. Réessayez dans 5 secondes.");

            case ApiStatut.Erreur:
                return new ResultatAction(false, LireChamp(corps, "message") ?? "Action refusée par le serveur.");

            default:
                return new ResultatAction(true, LireChamp(corps, "message") ?? messageParDefaut, LireEntier(corps, "montant"));
        }
    }

    // ------------------------------------------------------------------
    // Helpers
    // ------------------------------------------------------------------

    private async Task<(ApiStatut statut, string? corps)> EnvoyerAsync(
        HttpMethod methode, string chemin, object? payload = null, bool avecToken = true)
    {
        try
        {
            using var req = new HttpRequestMessage(methode, _config.ServerUrl.TrimEnd('/') + chemin);
            req.Headers.Accept.ParseAdd("application/json");
            if (avecToken && !string.IsNullOrEmpty(_config.AgentToken))
                req.Headers.Add("X-Agent-Token", _config.AgentToken);

            if (methode != HttpMethod.Get)
                req.Content = new StringContent(
                    JsonSerializer.Serialize(payload ?? new { }), Encoding.UTF8, "application/json");

            using var res = await _http.SendAsync(req);
            var corps = await res.Content.ReadAsStringAsync();

            if (res.StatusCode == HttpStatusCode.Unauthorized) return (ApiStatut.NonAutorise, corps);
            return (res.IsSuccessStatusCode ? ApiStatut.Ok : ApiStatut.Erreur, corps);
        }
        catch
        {
            return (ApiStatut.Injoignable, null);
        }
    }

    private static string? LireChamp(string? json, string champ)
    {
        if (string.IsNullOrWhiteSpace(json)) return null;
        try
        {
            using var doc = JsonDocument.Parse(json);
            return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == JsonValueKind.String
                ? v.GetString() : null;
        }
        catch { return null; }
    }

        private static int? LireEntier(string? json, string champ)
    {
        if (string.IsNullOrWhiteSpace(json)) return null;
        try
        {
            using var doc = JsonDocument.Parse(json);
            return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == JsonValueKind.Number
                ? v.GetInt32() : null;
        }
        catch { return null; }
    }

    /// <summary>
    /// Détermine l'IP locale utilisée pour joindre le serveur, puis la MAC de cette même carte
    /// (et non d'une carte virtuelle/VPN au hasard).
    /// </summary>
    private static InfosReseau LireReseau(string serverUrl)
    {
        string ip = "", mac = "";
        try
        {
            if (Uri.TryCreate(serverUrl, UriKind.Absolute, out var uri))
            {
                using var s = new Socket(AddressFamily.InterNetwork, SocketType.Dgram, 0);
                s.Connect(uri.Host, uri.Port);   // UDP : aucun paquet envoyé
                var local = (s.LocalEndPoint as IPEndPoint)?.Address;
                if (local != null)
                {
                    ip = local.ToString();
                    foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
                    {
                        if (nic.OperationalStatus != OperationalStatus.Up) continue;
                        if (!nic.GetIPProperties().UnicastAddresses.Any(a => a.Address.Equals(local))) continue;
                        mac = FormaterMac(nic.GetPhysicalAddress().ToString());
                        if (mac != "") break;
                    }
                }
            }
        }
        catch { }

        // Repli : première carte active non loopback
        if (mac == "")
        {
            try
            {
                foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
                {
                    if (nic.OperationalStatus != OperationalStatus.Up) continue;
                    if (nic.NetworkInterfaceType == NetworkInterfaceType.Loopback) continue;
                    mac = FormaterMac(nic.GetPhysicalAddress().ToString());
                    if (mac != "") break;
                }
            }
            catch { }
        }

        return new InfosReseau(ip, mac);
    }

    private static string FormaterMac(string brut) =>
        brut.Length == 12
            ? string.Join(":", Enumerable.Range(0, 6).Select(i => brut.Substring(i * 2, 2))).ToUpperInvariant()
            : "";
}

using Microsoft.Web.WebView2.Core;
using Microsoft.Web.WebView2.WinForms;
using System.ComponentModel;
using System.Runtime.InteropServices;
using System.Text.Json;


/// <summary>
/// Fenêtre plein écran de verrouillage.
///
/// Principe : on mémorise l'ÉTAT SOUHAITÉ (verrouillé / déverrouillé) et on l'applique dès que
/// la fenêtre est prête. Une demande de verrouillage ne peut donc jamais être perdue, même si
/// WebView2 n'a pas fini de charger : la fenêtre sombre couvre déjà l'écran (échec « fermé »).
///
/// Étape 5 : chat (toasts + panneau du widget + écran de verrouillage), alerte 5 min, fin de forfait
/// avec verrouillage local de sécurité, extinction avec décompte annulable.
/// </summary>
public class ClientWindow : Form
{
    public static ClientWindow? Instance { get; private set; }
    public static Action<string> Journal { get; set; } = Console.WriteLine;

    /// <summary>Appelé quand le client clique sur ACTIVER.</summary>
    public Func<Task<ActivationResult>>? ActivationHandler { get; set; }

    /// <summary>Appelés par le widget flottant (Suspendre / Quitter).</summary>
    public Func<Task<ResultatAction>>? SuspendreHandler { get; set; }
    public Func<Task<ResultatAction>>? QuitterHandler { get; set; }

    /// <summary>Envoie un message du client à l'accueil.</summary>
    public Func<string, Task<ResultatAction>>? MessageHandler { get; set; }

    /// <summary>Exécute l'extinction réelle de Windows ; retourne false en cas d'échec.</summary>
    public Func<bool>? ExtinctionHandler { get; set; }

    /// <summary>Demande un heartbeat immédiat (après recharge, désactivation…).</summary>
    public Action? ReveillerHeartbeat { get; set; }

    public bool IsLocked => _locked;

    private const int MaxHistorique = 60;

    private readonly bool _devMode;
    private readonly WebView2 _web;
    private readonly Label _attente;
    private readonly System.Windows.Forms.Timer _garde;
    private readonly System.Windows.Forms.Timer _finTimer;     // verrouillage local de sécurité après fin de forfait
    private readonly System.Windows.Forms.Timer _arretTimer;   // décompte avant extinction

    private volatile bool _desiredLocked = true;   // au démarrage : verrouillé
    private bool _locked;
    private bool _webReady;
    private bool _activationEnCours;
    private long _graceTicks;                       // ignore les ordres de verrouillage « heartbeat » un court instant

    private long _graceVerrouTicks;                 // ignore les « déverrouille » d'un heartbeat parti juste avant un verrouillage
    private SessionWidget? _widget;
    private ChatPanel? _chat;
    private SessionInfo? _session;                  // dernière session connue (heartbeat)

    private readonly List<MessageChat> _historique = new();
    private int _nonLus;
    private string? _messageVerrou;                 // bandeau affiché sur l'écran de verrouillage (ex : forfait terminé)

    private volatile bool _arretEnCours;
    private int _arretRestant;

    private string _poste = "", _ip = "", _etat = "serveur_injoignable";

    private readonly LowLevelKeyboardProc? _proc;
    private IntPtr _hook = IntPtr.Zero;

    public ClientWindow(bool devMode)
    {
        Instance = this;
        _devMode = devMode;

        Text = "Cyber Manager Agent";
        BackColor = Color.FromArgb(11, 18, 38);

        // Ces propriétés sont fixées UNE fois : les modifier plus tard recréerait le handle de la fenêtre
        if (devMode)
        {
            FormBorderStyle = FormBorderStyle.Sizable;
            StartPosition = FormStartPosition.CenterScreen;
            ClientSize = new Size(1100, 700);
            ShowInTaskbar = true;
        }
        else
        {
            FormBorderStyle = FormBorderStyle.None;
            StartPosition = FormStartPosition.Manual;
            Bounds = SystemInformation.VirtualScreen;
            ShowInTaskbar = false;
            TopMost = true;
        }

        _web = new WebView2
        {
            Dock = DockStyle.Fill,
            DefaultBackgroundColor = Color.FromArgb(11, 18, 38)
        };
        _attente = new Label
        {
            Dock = DockStyle.Fill,
            BackColor = Color.FromArgb(11, 18, 38),
            ForeColor = Color.FromArgb(148, 163, 184),
            Font = new Font("Segoe UI", 16),
            TextAlign = ContentAlignment.MiddleCenter,
            Text = "Cyber Manager — démarrage…"
        };
        Controls.Add(_web);
        Controls.Add(_attente);
        _attente.BringToFront();

        // Remet la fenêtre au premier plan si quelque chose passe devant
        _garde = new System.Windows.Forms.Timer { Interval = 1500 };
        _garde.Tick += (_, _) =>
        {
            if (_locked && !_devMode)
            {
                if (!TopMost) TopMost = true;
                if (!ContainsFocus) { BringToFront(); Activate(); }
            }
        };
        _garde.Start();

        // Fin de forfait : si le serveur n'a pas pris la main dans les 6 s, on verrouille nous-mêmes
        _finTimer = new System.Windows.Forms.Timer { Interval = 6000 };
        _finTimer.Tick += (_, _) =>
        {
            _finTimer.Stop();
            if (!_desiredLocked && _session?.Mode == "forfait")
            {
                Journal("[Session] Forfait terminé : verrouillage local (le serveur n'a pas répondu à temps).");
                LockScreen();
            }
        };

        _arretTimer = new System.Windows.Forms.Timer { Interval = 1000 };
        _arretTimer.Tick += (_, _) => TickArret();

        if (!devMode)
        {
            _proc = HookCallback;
            _hook = PoserHook(_proc);
        }
    }

    // ------------------------------------------------------------------
    // API publique (appelable depuis n'importe quel thread)
    // ------------------------------------------------------------------

    public void LockScreen()
    {
        _desiredLocked = true;
        Interlocked.Exchange(ref _graceVerrouTicks, DateTime.UtcNow.AddSeconds(6).Ticks);
        DemanderApplication();
    }

    public void UnlockScreen()
    {
        _desiredLocked = false;
        Interlocked.Exchange(ref _graceTicks, DateTime.UtcNow.AddSeconds(6).Ticks);
        DemanderApplication();
    }

    /// <summary>Applique l'ordre du serveur reçu via heartbeat.</summary>
    public void SyncFromServer(bool doitEtreVerrouille)
    {
        if (doitEtreVerrouille)
        {
            // Un heartbeat parti juste avant l'activation peut revenir avec l'ancien état
            if (DateTime.UtcNow.Ticks < Interlocked.Read(ref _graceTicks)) return;
            _desiredLocked = true;
            DemanderApplication();   // idempotent : sert aussi d'auto-réparation
        }
        else if (_desiredLocked)
        {
            // Idem dans l'autre sens : un heartbeat parti avant Suspendre/Quitter ne doit pas rouvrir l'écran
            if (DateTime.UtcNow.Ticks < Interlocked.Read(ref _graceVerrouTicks)) return;
            UnlockScreen();
        }
    }

    public void MettreAJourInfos(string poste, string ip, string etat)
    {
        _poste = poste; _ip = ip; _etat = etat;
        SurUi(PousserInfos);
    }

    /// <summary>Reçoit la session en cours (ou null) depuis le heartbeat ; pilote le widget flottant.</summary>
    public void MettreAJourSession(SessionInfo? session)
    {
        var precedente = _session;
        _session = session;
        SurUi(() =>
        {
            // Forfait rechargé : on annule la fin locale
            if (session?.RestantS is > 0) AnnulerFinForfait();
            // Session terminée : la discussion du client suivant repart de zéro
            if (precedente != null && session == null) ViderHistorique();
            MettreAJourWidget();
        });
    }

    /// <summary>Message de l'accueil (commande « message »).</summary>
    public void RecevoirMessage(string texte) => SurUi(() => SurMessageRecu(texte));

    /// <summary>Notification de recharge (commande « recharger »). Le widget se recale via le heartbeat.</summary>
    public void NotifierRecharge(int? minutes) => SurUi(() =>
    {
        if (_desiredLocked) return;
        var texte = minutes is > 0
            ? $"{minutes} minute(s) ont été ajoutées à votre forfait."
            : "Votre forfait a été rechargé.";
        ToastForm.Afficher("Temps ajouté", texte, NiveauToast.Info, 7000);
    });

    /// <summary>Extinction demandée par le serveur : écran verrouillé + décompte annulable.</summary>
    public void DemanderArret(int secondes)
    {
        _arretRestant = Math.Max(1, secondes);
        _arretEnCours = true;
        Journal($"[Arrêt] Extinction demandée par le serveur dans {_arretRestant} s.");
        SurUi(DemarrerArret);
    }

    // ------------------------------------------------------------------
    // Application de l'état (thread UI uniquement)
    // ------------------------------------------------------------------

    private void SurUi(Action action)
    {
        if (IsDisposed || !IsHandleCreated) return;
        if (InvokeRequired) BeginInvoke(action);
        else action();
    }

    private void DemanderApplication()
    {
        if (IsDisposed || !IsHandleCreated) return;   // OnShown appliquera l'état souhaité
        if (InvokeRequired) BeginInvoke(new Action(AppliquerEtatSouhaite));
        else AppliquerEtatSouhaite();
    }

    private void AppliquerEtatSouhaite()
    {
        if (IsDisposed) return;

        if (_desiredLocked)
        {
            var etaitVerrouille = _locked;
            var etaitCache = !Visible;
            if (!_devMode)
            {
                var cible = SystemInformation.VirtualScreen;
                if (Bounds != cible) Bounds = cible;
                TopMost = true;
            }
            if (etaitCache) Show();
            _locked = true;

            if (etaitCache || !_devMode)
            {
                BringToFront();
                Activate();
                if (_webReady) _web.Focus();
            }

            if (!etaitVerrouille)
            {
                // Les messages passent désormais par l'écran de verrouillage
                if (_chat is { IsDisposed: false }) _chat.Hide();
                ToastForm.FermerTous();
                PousserInfos();
            }
        }
        else
        {
            _locked = false;
            _messageVerrou = null;
            TopMost = false;
            if (Visible) Hide();
        }

        MettreAJourWidget();   // écran verrouillé → widget fermé ; déverrouillé + session → widget affiché
    }

    protected override async void OnShown(EventArgs e)
    {
        base.OnShown(e);
        AppliquerEtatSouhaite();
        if (_arretEnCours) DemarrerArret();
        await InitialiserWebViewAsync();
    }

    protected override void OnFormClosing(FormClosingEventArgs e)
    {
        // Alt+F4 & co. impossibles tant que le poste est verrouillé
        if (_locked && !_devMode && e.CloseReason == CloseReason.UserClosing)
            e.Cancel = true;
        base.OnFormClosing(e);
    }

    protected override void Dispose(bool disposing)
    {
        if (disposing)
        {
            _garde.Dispose();
            _finTimer.Dispose();
            _arretTimer.Dispose();
            if (_hook != IntPtr.Zero) { UnhookWindowsHookEx(_hook); _hook = IntPtr.Zero; }
            if (_widget is { IsDisposed: false }) _widget.Dispose();
            _widget = null;
            if (_chat is { IsDisposed: false }) _chat.Fermer();
            _chat = null;
        }
        base.Dispose(disposing);
    }

    // ------------------------------------------------------------------
    // Widget flottant de session (thread UI uniquement)
    // ------------------------------------------------------------------

    private void MettreAJourWidget()
    {
        if (IsDisposed) return;

        var session = _session;
        if (_desiredLocked || session == null)
        {
            FermerWidget();
            return;
        }

        if (_widget == null || _widget.IsDisposed)
        {
            var w = new SessionWidget
            {
                SuspendreHandler = SuspendreDepuisWidgetAsync,
                QuitterHandler = QuitterDepuisWidgetAsync
            };
            w.CinqMinutes += SurCinqMinutes;
            w.ForfaitTermine += SurForfaitTermine;
            w.MessagesDemandes += BasculerChat;
            w.NonLus = _nonLus;
            w.FormClosed += (_, _) => { if (ReferenceEquals(_widget, w)) _widget = null; };
            _widget = w;
        }
        _widget.MettreAJour(session);
    }

    private void FermerWidget()
    {
        if (_chat is { IsDisposed: false }) _chat.Hide();
        var w = _widget;
        _widget = null;
        if (w != null && !w.IsDisposed) w.Fermer();
    }

    private async Task<ResultatAction> SuspendreDepuisWidgetAsync()
    {
        var handler = SuspendreHandler;
        if (handler == null) return new ResultatAction(false, "L'agent n'est pas encore prêt. Réessayez.");

        var r = await handler();
        if (r.Succes)
        {
            _session = null;
            LockScreen();   // verrouille l'écran et ferme le widget
        }
        return r;
    }

    private async Task<ResultatAction> QuitterDepuisWidgetAsync()
    {
        var handler = QuitterHandler;
        if (handler == null) return new ResultatAction(false, "L'agent n'est pas encore prêt. Réessayez.");

        var r = await handler();
        if (r.Succes)
        {
            // Le client doit connaître son montant avant que l'écran ne se verrouille
            if (r.Montant is > 0)
            {
                var texte = $"Session terminée.\n\nMontant à payer : {SessionWidget.FormaterAr(r.Montant.Value)}\n\nMerci de régler à l'accueil.";
                if (_widget is { IsDisposed: false } w)
                    MessageBox.Show(w, texte, "Cyber Manager", MessageBoxButtons.OK, MessageBoxIcon.Information);
                else
                    MessageBox.Show(texte, "Cyber Manager", MessageBoxButtons.OK, MessageBoxIcon.Information);
            }
            _session = null;
            ViderHistorique();
            LockScreen();
        }
        return r;
    }

    /// <summary>Envoi d'un message du client (panneau du widget ou écran de verrouillage).</summary>
    private async Task<ResultatAction> EnvoyerMessageClientAsync(string texte)
    {
        texte = (texte ?? "").Trim();
        if (texte.Length == 0) return new ResultatAction(false, "Message vide.");
        if (texte.Length > 500) texte = texte[..500];

        var handler = MessageHandler;
        if (handler == null) return new ResultatAction(false, "L'agent n'est pas encore prêt. Réessayez.");

        var r = await handler(texte);
        if (r.Succes) AjouterHistorique(new MessageChat(true, texte, DateTime.Now.ToString("HH:mm")));
        return r;
    }

                s.IsZoomControlEnabled = false;

                _web.CoreWebView2.NewWindowRequested += (_, a) => a.Handled = true;
                _web.CoreWebView2.NavigationStarting += (_, a) => { if (a.IsUserInitiated) a.Cancel = true; };
                _web.CoreWebView2.WebMessageReceived += OnWebMessage;
                _web.NavigationCompleted += (_, a) =>

                    PousserInfos();
                    foreach (var m in _historique)

                    if (_desiredLocked) _web.Focus();
                    Journal("[WebView2] Écran de verrouillage chargé.");
                };

                _web.NavigateToString(LockScreenHtml.Page);
                return;
            }
            catch (Exception ex)
            {
                Journal($"[WebView2] Initialisation impossible : {ex.Message}. Nouvel essai dans 5 s.");
                _attente.Text = "Cyber Manager\nInitialisation de l'écran de verrouillage…\n(WebView2 Runtime requis)";

    private async void OnWebMessage(object? sender, CoreWebView2WebMessageReceivedEventArgs e)
    {
        string message;
        try { message = e.TryGetWebMessageAsString(); } catch { return; }

        if (message == "activer") { await ActiverAsync(); return; }
        if (message == "annuler_arret") { AnnulerArret(); return; }
        if (message.StartsWith("msg:", StringComparison.Ordinal)) await EnvoyerDepuisVerrouAsync(message[4..]);
    }

            Poster(new { type = "message_statut", ok = false, erreur = "Erreur inattendue. Réessayez." });
        }
    }

    private async Task ActiverAsync()
    {
        if (_activationEnCours) return;

        _activationEnCours = true;
        try
        {
            Poster(new { type = "activation", statut = "encours" });

            var handler = ActivationHandler;
            var res = handler == null
                ? ActivationResult.Echec("L'agent n'est pas encore prêt. Réessayez.")
                : await handler();

            if (res.Succes)
            {
                Poster(new { type = "activation", statut = "ok", message = res.Message });
                await Task.Delay(700);   // laisse voir « ACTIVÉ » un instant
                UnlockScreen();
            }
            else
            {
                Poster(new { type = "activation", statut = "erreur", message = res.Message });
            }
        }
        catch (Exception ex)
        {
            Journal($"[Activation] Erreur : {ex.Message}");
            Poster(new { type = "activation", statut = "erreur", message = "Erreur inattendue. Réessayez." });
        }
        finally { _activationEnCours = false; }
    }

    // Blocage clavier (Win, Alt+Tab, Alt+F4, Alt+Échap, Ctrl+Échap) — inactif en mode DEV
    // Ctrl+Alt+Suppr ne peut pas être bloqué par une application (limite Windows).
    // ------------------------------------------------------------------

    private delegate IntPtr LowLevelKeyboardProc(int nCode, IntPtr wParam, IntPtr lParam);

    private const int WH_KEYBOARD_LL = 13;
    private const int WM_KEYDOWN = 0x0100, WM_KEYUP = 0x0101, WM_SYSKEYDOWN = 0x0104, WM_SYSKEYUP = 0x0105;

    private static IntPtr PoserHook(LowLevelKeyboardProc proc)
    {
        using var p = System.Diagnostics.Process.GetCurrentProcess();
        using var m = p.MainModule;
        return SetWindowsHookEx(WH_KEYBOARD_LL, proc, GetModuleHandle(m!.ModuleName), 0);
    }

    private IntPtr HookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        if (nCode >= 0 && _locked)
        {
            var msg = (int)wParam;
            if (msg is WM_KEYDOWN or WM_KEYUP or WM_SYSKEYDOWN or WM_SYSKEYUP)
            {
                var vk = Marshal.ReadInt32(lParam);          // vkCode
                var flags = Marshal.ReadInt32(lParam, 8);    // flags
                var alt = (flags & 0x20) != 0;               // LLKHF_ALTDOWN
                var ctrl = (GetAsyncKeyState(0x11) & 0x8000) != 0;

                var bloquer =
                    vk is 0x5B or 0x5C or 0x5D                    // Win gauche/droite, touche Menu
                    || (alt && vk is (0x09 or 0x1B or 0x73 or 0x20)) // Alt+Tab / Échap / F4 / Espace
                    || (ctrl && vk == 0x1B);                       // Ctrl+Échap
                if (bloquer) return (IntPtr)1;
            }
        }
        return CallNextHookEx(IntPtr.Zero, nCode, wParam, lParam);
    }

    [DllImport("user32.dll")] private static extern IntPtr SetWindowsHookEx(int idHook, LowLevelKeyboardProc lpfn, IntPtr hMod, uint dwThreadId);
    [DllImport("user32.dll")] private static extern bool UnhookWindowsHookEx(IntPtr hhk);
    [DllImport("user32.dll")] private static extern IntPtr CallNextHookEx(IntPtr hhk, int nCode, IntPtr wParam, IntPtr lParam);
    [DllImport("user32.dll")] private static extern short GetAsyncKeyState(int vKey);
    [DllImport("kernel32.dll")] private static extern IntPtr GetModuleHandle(string? lpModuleName);
}

    private void Rafraichir()
    {
        var delta = (int)Math.Max(0, (Environment.TickCount64 - _recuTick) / 1000);
        var forfait = _mode == "forfait" && _restantBase.HasValue;

        _lblCaption.Text = forfait ? "TEMPS RESTANT" : "TEMPS ÉCOULÉ";
        var secondes = forfait ? Math.Max(0, _restantBase!.Value - delta) : _ecouleBase + delta;
        _lblTemps.Text = Duree(secondes);
        _lblTemps.ForeColor = forfait && secondes <= 300 ? Color.FromArgb(248, 113, 113) : Ambre;

        _lblMontant.Text = _mode == "libre" ? "Sans facturation" : "À payer : " + FormaterAr(_montant);
        _lblDebut.Text = "Début : " + _debut;

        if (!forfait) return;

        // Alerte « 5 minutes restantes » (une fois ; réarmée si le forfait est rechargé)
        if (secondes > 300) _alerte5 = false;
        else if (secondes > 0 && !_alerte5)
        {
            _alerte5 = true;
            CinqMinutes?.Invoke();
        }

        // Fin de forfait détectée localement (même si le serveur est injoignable)
        if (secondes > 0) _finSignalee = false;
        else if (!_finSignalee)
        {
            _finSignalee = true;
            ForfaitTermine?.Invoke();
        }
    }

    private static string Duree(int secondes)
    {
        var t = TimeSpan.FromSeconds(Math.Max(0, secondes));
        return $"{(int)t.TotalHours:00}:{t.Minutes:00}:{t.Seconds:00}";
    }

    private static string FormaterHeure(string? iso) =>
        DateTimeOffset.TryParse(iso, CultureInfo.InvariantCulture, DateTimeStyles.None, out var d)
            ? d.ToLocalTime().ToString("HH:mm", CultureInfo.InvariantCulture)
            : "—";

    // ------------------------------------------------------------------
    // Actions Suspendre / Quitter
    // ------------------------------------------------------------------

    private async Task ExecuterAsync(Func<Task<ResultatAction>>? handler, string question, string titre)
    {
        if (_occupe || handler == null) return;

        var choix = MessageBox.Show(this, question, titre,
            MessageBoxButtons.YesNo, MessageBoxIcon.Question, MessageBoxDefaultButton.Button2);
        if (choix != DialogResult.Yes || IsDisposed) return;

        DefinirOccupe(true);
        try
        {
            var r = await handler();
            if (IsDisposed) return;   // le widget a été fermé entre-temps (succès ou ordre du serveur)
            if (!r.Succes)
                MessageBox.Show(this, r.Message, titre, MessageBoxButtons.OK, MessageBoxIcon.Warning);
        }
        catch (Exception ex)
        {
            ClientWindow.Journal($"[Widget] Erreur : {ex.Message}");
            if (!IsDisposed)
                MessageBox.Show(this, "Erreur inattendue. Réessayez.", titre, MessageBoxButtons.OK, MessageBoxIcon.Warning);
        }
        finally
        {
            if (!IsDisposed) DefinirOccupe(false);
        }
    }

    private void DefinirOccupe(bool occupe)
    {
        _occupe = occupe;
        _btnCacher.Enabled = !occupe;

using System.Runtime.InteropServices;

public enum NiveauToast { Info, Message, Alerte, Danger }

/// <summary>
/// Notification flottante non bloquante (bas-droite, au-dessus du widget), avec animation
/// d'entrée / sortie. N'a jamais le focus. Thread UI uniquement.
/// </summary>
public class ToastForm : Form
{
    private static readonly List<ToastForm> Actifs = new();

    private static readonly Color Fond = Color.FromArgb(19, 28, 58);
    private static readonly Color Texte = Color.FromArgb(226, 232, 240);

    private readonly System.Windows.Forms.Timer _anim = new() { Interval = 15 };
    private readonly System.Windows.Forms.Timer _vie;
    private readonly string _titre, _texte;
    private readonly Color _accent;
    private readonly float _k;
    private int _etape, _cibleX;
    private bool _sortie;

    /// <summary>Appelé si l'utilisateur clique sur la notification.</summary>
    public Action? Clic { get; set; }

    private ToastForm(string titre, string texte, NiveauToast niveau, int dureeMs)
    {
        _k = Math.Max(1f, DeviceDpi / 96f);
        _titre = titre;
        _texte = texte;
        _accent = niveau switch
        {
            NiveauToast.Danger => Color.FromArgb(220, 38, 38),
            NiveauToast.Alerte => Color.FromArgb(245, 158, 11),
            NiveauToast.Message => Color.FromArgb(139, 92, 246),
            _ => Color.FromArgb(56, 189, 248)
        };

        FormBorderStyle = FormBorderStyle.None;
        StartPosition = FormStartPosition.Manual;
        ShowInTaskbar = false;
        TopMost = true;
        BackColor = Fond;
        DoubleBuffered = true;
        Cursor = Cursors.Hand;
        ClientSize = new Size(S(310), S(88));
        Opacity = 0;

        _anim.Tick += (_, _) => Animer();
        _vie = new System.Windows.Forms.Timer { Interval = Math.Max(1500, dureeMs) };
        _vie.Tick += (_, _) => { _vie.Stop(); Fermer(false); };
    }

    /// <summary>Affiche une notification (3 maximum à l'écran).</summary>
    public static ToastForm Afficher(string titre, string texte, NiveauToast niveau = NiveauToast.Info,
        int dureeMs = 7000, Action? clic = null)
    {
        if (Actifs.Count >= 3) Actifs[0].Fermer(true);

        var t = new ToastForm(titre, texte, niveau, dureeMs) { Clic = clic };
        Actifs.Add(t);
        Repositionner();
        t.Show();
        t._anim.Start();
        t._vie.Start();

        try
        {
            if (niveau is NiveauToast.Alerte or NiveauToast.Danger) System.Media.SystemSounds.Exclamation.Play();
            else if (niveau == NiveauToast.Message) System.Media.SystemSounds.Asterisk.Play();
        }
        catch { }
        return t;
    }

    public static void FermerTous()
    {
        foreach (var t in Actifs.ToArray()) t.Fermer(true);
    }

    public void Fermer(bool immediat)
    {
        if (IsDisposed) return;
        if (immediat) { Close(); return; }
        _sortie = true;
        _anim.Start();
    }

    // ------------------------------------------------------------------

    private static void Repositionner()
    {
        var zone = Screen.PrimaryScreen?.WorkingArea ?? SystemInformation.WorkingArea;
        var y = 0;
        // Les plus récentes en bas, au-dessus de la zone du widget (180 px + marges)
        for (var i = Actifs.Count - 1; i >= 0; i--)
        {
            var t = Actifs[i];
            var bas = zone.Bottom - t.S(12 + 180 + 8) - y;
            t._cibleX = zone.Right - t.Width - t.S(12);
            t.Location = new Point(t._cibleX + (t._etape < 14 ? t.S(40) : 0), bas - t.Height);
            y += t.Height + t.S(8);
        }
    }

    private void Animer()
    {
        if (IsDisposed) return;
        _etape += _sortie ? -1 : 1;
        var t = Math.Clamp(_etape / 14.0, 0, 1);
        var e = 1 - Math.Pow(1 - t, 3);
        Opacity = e;
        Left = _cibleX + (int)Math.Round(S(40) * (1 - e));

        if (_sortie && _etape <= 0) { Close(); return; }
        if (!_sortie && _etape >= 14) _anim.Stop();
    }

    protected override bool ShowWithoutActivation => true;

    protected override CreateParams CreateParams
    {
        get
        {
            var cp = base.CreateParams;
            cp.ExStyle |= 0x80 | 0x08000000;   // WS_EX_TOOLWINDOW | WS_EX_NOACTIVATE
            return cp;
        }
    }

    protected override void OnMouseClick(MouseEventArgs e)
    {
        base.OnMouseClick(e);
        try { Clic?.Invoke(); } catch { }
        Fermer(false);
    }

    protected override void OnPaint(PaintEventArgs e)
    {
        base.OnPaint(e);
        using var accent = new SolidBrush(_accent);
        e.Graphics.FillRectangle(accent, 0, 0, S(5), Height);
        using var pen = new Pen(Color.FromArgb(90, 139, 92, 246));
        e.Graphics.DrawRectangle(pen, 0, 0, Width - 1, Height - 1);

        using var fTitre = new Font("Segoe UI", 9.5f, FontStyle.Bold);
        using var fTexte = new Font("Segoe UI", 9f);
        TextRenderer.DrawText(e.Graphics, _titre, fTitre,
            new Rectangle(S(16), S(8), Width - S(26), S(20)), Color.White,
            TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding);
        TextRenderer.DrawText(e.Graphics, _texte, fTexte,
            new Rectangle(S(16), S(30), Width - S(26), Height - S(38)), Texte,
            TextFormatFlags.WordBreak | TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding);
    }

    protected override void OnFormClosed(FormClosedEventArgs e)
    {
        Actifs.Remove(this);
        _anim.Dispose();
        _vie.Dispose();
        Repositionner();
        base.OnFormClosed(e);
    }

    private int S(int v) => (int)Math.Round(v * _k);
}

void Log(string msg)
{

static int Backoff(int echecs) => Math.Min(30_000, 2_000 << Math.Min(echecs, 4));   // 4s, 8s, 16s, 30s…

// Lecture d'un champ du payload JSON d'une commande (jamais journalisé)
static string? LireTexte(string? json, string champ)
{
    if (string.IsNullOrWhiteSpace(json)) return null;
    try
    {
        using var doc = System.Text.Json.JsonDocument.Parse(json);
        return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == System.Text.Json.JsonValueKind.String
            ? v.GetString() : null;
    }
    catch { return null; }
}

static int? LireEntier(string? json, string champ)
{
    if (string.IsNullOrWhiteSpace(json)) return null;
    try
    {
        using var doc = System.Text.Json.JsonDocument.Parse(json);
        return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == System.Text.Json.JsonValueKind.Number
            ? v.GetInt32() : null;
    }
    catch { return null; }
}

    // --- Boucle Commandes (3 s, backoff si panne) ---------------------------
    _ = Task.Run(async () =>
    {
        await Task.Delay(3_000);
        var echecs = 0;
        while (true)
        {
            var delai = 3_000;
            try
            {
                if (!string.IsNullOrEmpty(config.AgentToken))
                {
                    var r = await api.GetCommandsAsync();
                    if (r.Statut == ApiStatut.Ok)
                    {
                        echecs = 0;
                        foreach (var cmd in r.Donnees!)
                        {
                            Log($"Commande reçue : {cmd.Type} (id={cmd.Id})");
                            var (ok, detail) = Executer(cmd, fenetreUi);
                            await api.ConfirmerCommandeAsync(cmd.Id, ok, detail);
                        }
                    }
                    else if (r.Statut != ApiStatut.NonAutorise)
                    {
                        echecs++;
                        delai = Backoff(echecs);
                    }
                }
            }
            catch (Exception ex) { Log($"Commandes : erreur inattendue : {ex.Message}"); }

            await Task.Delay(delai);
        }
    });

    // Exécution d'une commande reçue du serveur
    (bool ok, string detail) Executer(AgentCommand cmd, ClientWindow fenetre)
    {
        try
        {
            switch (cmd.Type)
            {
                case "demarrer":
                case "debloquer":
                case "reprendre":
                    fenetre.UnlockScreen();
                    return (true, "Poste déverrouillé");

                case "verrouiller":
                case "suspendre":
                case "fermer":
                    fenetre.LockScreen();
                    return (true, "Poste verrouillé");

                case "desactiver":
                    // L'écran affiche tout de suite « Poste désactivé » (ACTIVER devient INDISPONIBLE)
                    etatServeur = "desactive";
                    PousserInfos();
                    fenetre.LockScreen();   // ferme aussi le widget s'il est ouvert
                    Reveiller();
                    return (true, "Poste désactivé et verrouillé");

                case "arreter":
                    fenetre.DemanderArret(DelaiArretSecondes);
                    return (true, $"Extinction dans {DelaiArretSecondes} s");

                case "recharger":
                {
                    var minutes = LireEntier(cmd.Payload, "minutes_ajoutees");
                    fenetre.NotifierRecharge(minutes);
                    Reveiller();   // le widget se recale tout de suite sur le temps du serveur
                    return (true, "Temps ajouté");
                }

                case "message":
                {
                    var texte = LireTexte(cmd.Payload, "message");
                    if (string.IsNullOrWhiteSpace(texte)) return (false, "Message vide");
                    fenetre.RecevoirMessage(texte.Trim());
                    return (true, "Message affiché");
                }

                default:
                    return (false, $"Commande inconnue : {cmd.Type}");
            }
        }
        catch (Exception ex)
        {
            Log($"Exécution de '{cmd.Type}' échouée : {ex.Message}");
            return (false, "Erreur d'exécution");
        }
    }

    uiThread.Join();
    Log(">>> Application terminée.");
}
catch (Exception ex)
{
    Log($">>> ERREUR FATALE : {ex}");
}

    private const REGEX_MAC = '/^([0-9A-Fa-f]{2}[:-]){5}[0-9A-Fa-f]{2}$/';

    /**
     * Handshake : l'agent s'annonce. Crée le poste s'il est inconnu.
     * Le nom d'un poste existant n'est PAS écrasé (l'admin garde la main).
     * Si le poste revient après une panne / un redémarrage avec une session ouverte,
     * le temps de panne est déduit et Internet est rétabli ; la réponse contient la session en cours.
     */
    public function handshake(Request $request, PosteSessionService $sessions): JsonResponse
    {
        $data = $request->validate([
            'identifiant_unique' => ['required', 'string', 'max:64'],
            'nom_poste'          => ['required', 'string', 'max:100'],
            'adresse_ip'         => ['nullable', 'ip'],
            'adresse_mac'        => ['nullable', 'string', 'regex:' . self::REGEX_MAC],
            'agent_version'      => ['nullable', 'string', 'max:32'],
        ]);

        $mac = isset($data['adresse_mac'])
            ? strtoupper(str_replace('-', ':', $data['adresse_mac']))
            : null;

        $poste = Poste::where('identifiant_unique', $data['identifiant_unique'])->first();

        $retourApresPanne = false;
        $ancienSigne      = null;

        if ($poste === null) {
            $poste = Poste::create([
                'nom_poste'              => $data['nom_poste'],
                'identifiant_unique'     => $data['identifiant_unique'],
                'adresse_ip'             => $data['adresse_ip'] ?? null,
                'adresse_mac'            => $mac,
                'agent_version'          => $data['agent_version'] ?? null,
                'agent_token'            => Str::random(64),
                'etat'                   => Poste::ETAT_DISPONIBLE,
                'actif'                  => true,
                'dernier_heartbeat'      => now(),
                'derniere_communication' => now(),
            ]);
            $nouveau = true;
        } else {
            $retourApresPanne = !$poste->estEnLigne();
            $ancienSigne      = $poste->dernier_heartbeat;

            $poste->update([
                'adresse_ip'             => $data['adresse_ip'] ?? $poste->adresse_ip,
                'adresse_mac'            => $mac ?? $poste->adresse_mac,
                'agent_version'          => $data['agent_version'] ?? $poste->agent_version,
                'agent_token'            => $poste->agent_token ?: Str::random(64),
                'dernier_heartbeat'      => now(),
                'derniere_communication' => now(),
                'etat'                   => $poste->etat === Poste::ETAT_HORS_LIGNE
                    ? ($poste->sessionActive() ? Poste::ETAT_OCCUPE : Poste::ETAT_DISPONIBLE)
                    : $poste->etat,
            ]);
            $nouveau = false;
        }

        // Redémarrage du PC en pleine session : déduction de la panne + Internet rétabli
        if ($retourApresPanne) {
            $sessions->traiterRetourEnLigne($poste, $ancienSigne);
            $poste->refresh();
        }

        $etat    = $poste->etatAffiche();
        $session = $poste->sessionActive();

        return response()->json([
            'ok'                   => true,
            'nouveau'              => $nouveau,
            'poste_id'             => $poste->id,
            'agent_token'          => $poste->agent_token,
            'nom_poste'            => $poste->nom_poste,
            'etat'                 => $etat,
            'doit_etre_verrouille' => $this->doitEtreVerrouille($etat),
            'session'              => $session ? $this->resumeSession($session) : null,
            'message'              => $nouveau ? 'Poste enregistré.' : 'Poste reconnu.',
        ]);
    }

    /**
     * Heartbeat : « je suis vivant » + l'agent récupère l'état à appliquer.
     * C'est aussi ici que l'on constate l'expiration d'un forfait (réactivité de ~5 s).
     */
    public function heartbeat(Request $request, PosteSessionService $sessions): JsonResponse
    {
        $poste = $this->authentifier($request);
        if ($poste === null) {
            return response()->json(['ok' => false, 'message' => 'Token invalide.'], 401);
        }

        $data = $request->validate(['adresse_ip' => ['nullable', 'ip']]);

        // À lire AVANT la mise à jour du heartbeat
        $etaitEnLigne = $poste->estEnLigne();
        $ancienSigne  = $poste->dernier_heartbeat;

        $maj = ['dernier_heartbeat' => now(), 'derniere_communication' => now()];
        if (!empty($data['adresse_ip'])) {
            $maj['adresse_ip'] = $data['adresse_ip'];
        }
        if ($poste->etat === Poste::ETAT_HORS_LIGNE) {
            $maj['etat'] = $poste->sessionActive() ? Poste::ETAT_OCCUPE : Poste::ETAT_DISPONIBLE;
        }
        $poste->update($maj);

        if ($etaitEnLigne) {
            $sessions->expirerSiNecessaire($poste);
        } else {
            $sessions->traiterRetourEnLigne($poste, $ancienSigne);
        }
        $poste->refresh();

        $etat    = $poste->etatAffiche();
        $session = $poste->sessionActive();

        return response()->json([
            'ok'                   => true,
            'nom_poste'            => $poste->nom_poste,
            'etat'                 => $etat,
            'doit_etre_verrouille' => $this->doitEtreVerrouille($etat),
            'session'              => $session ? $this->resumeSession($session) : null,
        ]);
    }

    /**
     * Activation demandée par le client (bouton ACTIVER de l'écran de verrouillage).
     * Crée une session postpayée (libre-service), connecte le poste au réseau.
     */
    public function activer(Request $request, PosteSessionService $sessions): JsonResponse
    {
        $poste = $this->authentifier($request);
        if ($poste === null) {
            return response()->json(['ok' => false, 'message' => 'Token invalide.'], 401);
        }

        // 1) Transaction courte : verrou de ligne pour éviter un double clic / une double session
        [$code, $corps, $posteFinal] = DB::transaction(function () use ($poste, $sessions) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $p->update(['dernier_heartbeat' => now(), 'derniere_communication' => now()]);

            switch ($p->etatAffiche()) {
                case Poste::ETAT_DESACTIVE:
                    return [403, ['ok' => false, 'message' => 'Poste désactivé, contactez le gérant.'], null];

                case Poste::ETAT_SUSPENDU:
                    return [409, ['ok' => false, 'message' => "Votre session est en pause. Adressez-vous à l'accueil."], null];

                case Poste::ETAT_VERROUILLE:
                    return [409, ['ok' => false, 'message' => "Ce poste est verrouillé par l'administrateur."], null];

                case Poste::ETAT_OCCUPE:
                    $s = $p->sessionActive();
                    // On renvoie aussi le poste : connecterReseau() est idempotent et ré-assure Internet
                    return [200, [
                        'ok'      => true,
                        'message' => 'Session déjà active.',
                        'session' => $s ? $this->resumeSession($s) : null,
                    ], $s ? $p : null];
            }

            // Disponible → on démarre une session libre-service
            $session = $sessions->creerSession($p, 'postpaye', null);

            // Trace d'audit : commande « demarrer » déjà exécutée (l'agent se déverrouille via la réponse)
            $this->tracer($p, 'demarrer', 'Activation depuis le poste client',
                ['session_id' => $session->id, 'type' => 'postpaye']);

            $p->update(['etat' => Poste::ETAT_OCCUPE]);

            return [200, [
                'ok'      => true,
                'message' => 'Poste activé.',
                'session' => $this->resumeSession($session),
            ], $p];
        });

        // 2) Hors transaction : l'appel MikroTik peut être lent, il ne doit pas bloquer la base
        if ($posteFinal !== null) {
            $reseau = $sessions->connecterReseau($posteFinal);
            if ($reseau === false) {
                $corps['avertissement'] = "Connexion Internet momentanément indisponible. Prévenez l'accueil.";
            }
        }

        return response()->json($corps, $code);
    }

    /** Widget client : bouton « Suspendre ». Chrono figé, poste « suspendu », Internet coupé. */
    public function suspendreSession(Request $request, PosteSessionService $sessions): JsonResponse
    {
        $poste = $this->authentifier($request);
        if ($poste === null) {
            return response()->json(['ok' => false, 'message' => 'Token invalide.'], 401);
        }

        $ok = DB::transaction(function () use ($poste, $sessions) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $p->update(['dernier_heartbeat' => now(), 'derniere_communication' => now()]);

            $session = $p->sessionActive();
            if ($session === null || !$sessions->suspendre($p)) {
                return false;
            }

            $this->tracer($p, 'suspendre', 'Suspension depuis le widget client', ['session_id' => $session->id]);
            return true;
        });

        if (!$ok) {
            return response()->json(['ok' => false, 'message' => 'Aucune session active à suspendre.'], 409);
        }

        // Hors transaction : coupure Internet (mode dégradé si MikroTik injoignable)
        $corps = ['ok' => true, 'message' => 'Session suspendue.'];
        if ($sessions->deconnecterReseau($poste) === false) {
            $corps['avertissement'] = "Coupure Internet non confirmée. Prévenez l'accueil.";
        }

        return response()->json($corps);
    }

    /** L'écran doit-il être verrouillé pour cet état ? (source unique handshake / heartbeat) */
    private function doitEtreVerrouille(string $etat): bool

        return in_array($etat, [
            Poste::ETAT_DISPONIBLE,
            Poste::ETAT_VERROUILLE,

    /** Widget client : bouton « Quitter ». Clôture la session, calcule le montant, coupe Internet. */
    public function fermerSession(Request $request, PosteSessionService $sessions): JsonResponse
    {
        $poste = $this->authentifier($request);
        if ($poste === null) {
            return response()->json(['ok' => false, 'message' => 'Token invalide.'], 401);
        }

        // 1) Transaction courte
        $resultat = DB::transaction(function () use ($poste, $sessions) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $p->update(['dernier_heartbeat' => now(), 'derniere_communication' => now()]);

            $session = $sessions->cloturer($p, 'client');
            if ($session === null) {
                return null;
            }

            $this->tracer($p, 'fermer', 'Fermeture depuis le widget client', ['session_id' => $session->id]);
            return [$p, (int) $session->montant_total];
        });

        if ($resultat === null) {
            return response()->json(['ok' => false, 'message' => 'Aucune session active à fermer.'], 409);
        }

        [$posteFinal, $montant] = $resultat;

        // 2) Hors transaction : coupure Internet (mode dégradé si MikroTik injoignable)
        $corps = ['ok' => true, 'message' => 'Session terminée.', 'montant' => $montant];
        if ($sessions->deconnecterReseau($posteFinal) === false) {
            $corps['avertissement'] = "Coupure Internet non confirmée. Prévenez l'accueil.";
        }

        return response()->json($corps);
    }

    public function confirmerCommande(Request $request, PosteCommande $commande): JsonResponse
    {
        $poste = $this->authentifier($request);
        if ($poste === null || $commande->poste_id !== $poste->id) {

            'resultat'    => $data['message'] ?? null,

        ]);

    private function authentifier(Request $request): ?Poste
    {
        $token = $request->header('X-Agent-Token');
        if (!$token) {
            return null;
        }

        return Poste::where('agent_token', $token)->first();
    }

    /** Trace d'audit : action initiée par le client, déjà exécutée (jamais renvoyée à l'agent par le polling). */
    private function tracer(Poste $poste, string $type, string $resultat, array $payload = []): void
    {
        PosteCommande::create([
            'poste_id'    => $poste->id,
            'type'        => $type,
            'payload'     => $payload + ['origine' => 'client'],
            'statut'      => 'executee',
            'resultat'    => $resultat,
            'recue_at'    => now(),
            'executee_at' => now(),
        ]);
    }

    /** Instantané de session envoyé à l'agent : le serveur reste la référence du temps et du montant. */
    private function resumeSession(Session $session): array
    {
        $mode    = $session->mode_facturation ?: 'forfait';
        $ecoule  = (int) $session->secondesConsommees();
        $prevue  = (int) $session->duree_prevue;
        $restant = ($mode === 'forfait' && $prevue > 0) ? max(0, $prevue * 60 - $ecoule) : null;

        return [
            'id'               => $session->id,
            'mode'             => $mode,
            'date_heure_debut' => $session->date_heure_debut?->toIso8601String(),
            'date_heure_fin'   => $session->date_heure_fin_prevue?->toIso8601String(),
            'duree_prevue'     => $prevue,
            'montant_actuel'   => app(PosteSessionService::class)->montant($session),
            'etat'             => $session->etat,
            'ecoule_s'         => $ecoule,
            'restant_s'        => $restant,
        ];
    }

        ]);

        if ($poste->etatAffiche() !== Poste::ETAT_DISPONIBLE) {
            return $this->echec($request, "Le poste {$poste->nom_poste} n'est pas disponible.");
        }

        $duree = $data['type'] === 'forfait' ? (int) $data['duree_minutes'] : null;

        $session = $this->sessions->creerSession($poste, $data['type'], $duree);
        $reseau  = $this->sessions->connecterReseau($poste);   // mode dégradé : n'échoue jamais

        PosteCommande::create([
            'poste_id' => $poste->id,
            'type'     => 'demarrer',
            'payload'  => [
                'session_id'    => $session->id,
                'duree_minutes' => $duree,
                'type'          => $data['type'],
                'montant'       => 0,
            ],
        ]);

        $poste->update(['etat' => Poste::ETAT_OCCUPE]);

        return $this->ok($request, "Poste {$poste->nom_poste} démarré." . $this->avertissementReseau($reseau));
    }

    public function debloquer(Request $request, Poste $poste)
    {
        if (!$poste->sessionActive()) {
            // Déblocage sans facturation : session de mode "libre" (montant 0)
            $this->sessions->creerSession($poste, 'libre', null);
        }

        $this->sessions->reprendre($poste);   // reprend si en pause, poste « utilisé »
        $reseau = $this->sessions->connecterReseau($poste);

        PosteCommande::create([
            'poste_id' => $poste->id,
            'type'     => 'debloquer',
            'payload'  => ['session_id' => $poste->sessionActive()?->id],
        ]);

        return $this->ok($request, "Poste {$poste->nom_poste} débloqué." . $this->avertissementReseau($reseau));
    }

    public function fermer(Request $request, Poste $poste)
    {
        $session = $this->terminer($poste, 'admin');
        if ($session === null) {
            return $this->echec($request, 'Aucune session active à fermer.');
        }

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);

        $montant = ($session->mode_facturation ?? 'forfait') === 'libre'
            ? ''
            : ' Montant : ' . number_format((int) $session->montant_total, 0, ',', ' ') . ' Ar.';

        return $this->ok($request, "Session de {$poste->nom_poste} fermée.{$montant}");
    }

    public function suspendre(Request $request, Poste $poste)
    {
        $ok = DB::transaction(function () use ($poste) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            return $p !== null && $this->sessions->suspendre($p);
        });

        if (!$ok) {
            return $this->echec($request, 'Aucune session active.');
        }

        $reseau = $this->sessions->deconnecterReseau($poste);   // la pause coupe Internet

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'suspendre']);

        return $this->ok($request, "Session de {$poste->nom_poste} suspendue." . $this->avertissementReseau($reseau, 'coupé'));
    }

    public function reprendre(Request $request, Poste $poste)
    {
        if (!$this->sessions->reprendre($poste)) {
            return $this->echec($request, 'Aucune session à reprendre.');
        }

        $reseau = $this->sessions->connecterReseau($poste);   // Internet avait été coupé à la suspension

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'reprendre']);

        return $this->ok($request, "Session de {$poste->nom_poste} reprise." . $this->avertissementReseau($reseau));
    }

    public function verrouiller(Request $request, Poste $poste)
    {
        if (!$poste->sessionActive()) {
            return $this->echec($request, 'Aucune session active à verrouiller.');
        }

        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);
        $poste->update(['etat' => Poste::ETAT_VERROUILLE]); // le chrono continue

        return $this->ok($request, "Poste {$poste->nom_poste} verrouillé (le chrono continue).");
    }

    public function desactiver(Request $request, Poste $poste)
    {
        $this->terminer($poste, 'desactivation');
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'desactiver']);
        $poste->update(['etat' => Poste::ETAT_DESACTIVE]);

        return $this->ok($request, "Poste {$poste->nom_poste} désactivé.");
    }

    public function activer(Request $request, Poste $poste)
    {
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'verrouiller']);
        $poste->update(['etat' => Poste::ETAT_DISPONIBLE]);

        return $this->ok($request, "Poste {$poste->nom_poste} réactivé.");
    }

    public function arreter(Request $request, Poste $poste)
    {
        $this->terminer($poste, 'arret');
        PosteCommande::create(['poste_id' => $poste->id, 'type' => 'arreter']);
        $poste->update(['etat' => Poste::ETAT_DISPONIBLE]);

        return $this->ok($request, "Commande d'arrêt envoyée à {$poste->nom_poste}.");
    }

    public function recharger(Request $request, Poste $poste)
    {
        $data = $request->validate(['minutes' => ['required', 'integer', 'min:1', 'max:1440']]);

        // Sous verrou : évite une course avec l'expiration automatique du forfait
        $resultat = DB::transaction(function () use ($poste, $data) {
            $p       = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $session = $p?->sessionActive();

            if ($session === null) {
                return 'aucune';
            }
            if (($session->mode_facturation ?? 'forfait') !== 'forfait') {
                return 'non_forfait';
            }

            $nouvelleDuree = (int) $session->duree_prevue + $data['minutes'];
            $session->update([
                'date_heure_fin_prevue' => $session->date_heure_fin_prevue
                    ? $session->date_heure_fin_prevue->copy()->addMinutes($data['minutes'])
                    : now()->addMinutes($data['minutes']),
                'duree_prevue' => $nouvelleDuree,
            ]);

            return $nouvelleDuree;
        });

        if ($resultat === 'aucune') {
            return $this->echec($request, 'Aucune session active sur ce poste.');
        }
        if ($resultat === 'non_forfait') {
            return $this->echec($request, "Cette session n'est pas un forfait : rien à recharger.");
        }

        PosteCommande::create([
            'poste_id' => $poste->id,
            'type'     => 'recharger',
            'payload'  => ['minutes_ajoutees' => (int) $data['minutes'], 'duree_totale_min' => $resultat],
        ]);

        return $this->ok($request, "{$data['minutes']} min ajoutées à {$poste->nom_poste} (forfait : {$resultat} min).");
    }

    /** Chat admin → client. */
    public function envoyerMessage(Request $request, Poste $poste)
    {
        $data = $request->validate(['message' => ['required', 'string', 'max:500']]);

        PosteMessage::create([
            'poste_id' => $poste->id,
            'sens'     => PosteMessage::SENS_ADMIN,
            'contenu'  => $data['message'],
        ]);

        PosteCommande::create([
            'poste_id' => $poste->id,
            'type'     => 'message',
            'payload'  => ['message' => $data['message']],
        ]);

        return $this->ok($request, 'Message envoyé.');
    }

        /**
     * Identifiants HotSpot du poste (compte MikroTik utilisé pour ouvrir Internet).
     * Identifiant vide = dissociation. Mot de passe vide = on conserve l'ancien.
     * Le mot de passe n'est jamais renvoyé ni journalisé.
     */
    public function enregistrerHotspot(Request $request, Poste $poste)
    {
        $data = $request->validate([
            'hotspot_username' => ['nullable', 'string', 'min:3', 'max:64', 'regex:/^\S+$/',
                Rule::unique('postes', 'hotspot_username')->ignore($poste->id)],
            'hotspot_password' => ['nullable', 'string', 'min:4', 'max:64', 'regex:/^\S+$/'],
        ], [
            'hotspot_username.min'    => "L'identifiant doit contenir au moins 3 caractères.",
            'hotspot_username.max'    => "L'identifiant ne doit pas dépasser 64 caractères.",
            'hotspot_username.regex'  => "L'identifiant ne doit pas contenir d'espaces.",
            'hotspot_username.unique' => 'Ce compte HotSpot est déjà utilisé par un autre poste.',
            'hotspot_password.min'    => 'Le mot de passe doit contenir au moins 4 caractères.',
            'hotspot_password.max'    => 'Le mot de passe ne doit pas dépasser 64 caractères.',
            'hotspot_password.regex'  => 'Le mot de passe ne doit pas contenir d\'espaces.',
        ]);

        $username = trim((string) ($data['hotspot_username'] ?? ''));
        $password = trim((string) ($data['hotspot_password'] ?? ''));

        if ($username === '') {
            $poste->update(['hotspot_username' => null, 'hotspot_password' => null]);
            Log::info("Identifiants HotSpot du poste {$poste->nom_poste} supprimés.");

            return $this->ok($request, "Identifiants HotSpot de {$poste->nom_poste} supprimés : Internet ne sera plus activé pour ce poste.");
        }

        if ($password === '' && blank($poste->hotspot_password)) {
            return $this->echec($request, 'Le mot de passe est obligatoire pour un nouveau compte.');
        }

        $compteChange = $username !== $poste->hotspot_username;

        $poste->update([
            'hotspot_username' => $username,
            'hotspot_password' => $password !== '' ? $password : $poste->hotspot_password,
        ]);
        Log::info("Identifiants HotSpot du poste {$poste->nom_poste} mis à jour (compte « {$username} »).");

        $message = "Identifiants HotSpot de {$poste->nom_poste} enregistrés.";

        if ($this->sessions->compteHotspotExiste($username) === false) {
            $message .= " Attention : le compte « {$username} » est introuvable sur MikroTik — créez-le avant d'utiliser ce poste.";
        }

        // Session en cours avec un autre compte : on rebascule Internet sur le nouveau
        $session = $poste->sessionActive();
        if ($compteChange && $session !== null && $session->etat === 'en_cours') {
            $message .= $this->avertissementReseau($this->sessions->connecterReseau($poste));
        }

        return $this->ok($request, $message);
    }

    private function tousLesPostes(): array
    {
        return Poste::orderBy('nom_poste')
            ->withCount(['messages as non_lus' => fn ($q) => $q
                ->where('sens', PosteMessage::SENS_CLIENT)
                ->whereNull('lu_at')])
            ->get()
            ->map(fn (Poste $p) => $this->donnees($p))
            ->values()
            ->all();
    }

    private function donnees(Poste $p): array
    {
        $etat = $p->etatAffiche();
        $s    = $p->sessionActive();

        $session = null;
        if ($s) {
            $mode    = $s->mode_facturation ?: 'forfait';
            $ecoule  = (int) $s->secondesConsommees();
            $prevue  = (int) $s->duree_prevue;
            $restant = ($mode === 'forfait' && $prevue > 0) ? max(0, $prevue * 60 - $ecoule) : null;

            $session = [
                'id'         => $s->id,
                'mode'       => $mode,
                'etat'       => $s->etat,
                'debut'      => $s->date_heure_debut?->format('H:i:s'),
                'ecoule_s'   => $ecoule,
                'restant_s'  => $restant,
                'duree_min'  => $prevue ?: null,
                'montant'    => $mode === 'libre' ? 0 : $this->sessions->montant($s, $ecoule),
            ];
        }

        return [
            'id'       => $p->id,
            'nom'      => $p->nom_poste,
            'etat'     => $etat,
            'ip'       => $p->adresse_ip,
            'mac'      => $p->adresse_mac,
            'version'  => $p->agent_version,
            'vu_il_y_a' => $p->dernier_heartbeat?->diffForHumans(),
            'session'  => $session,
            'non_lus'  => (int) ($p->non_lus ?? 0),
            'hotspot_username'  => $p->hotspot_username,
            'hotspot_configure' => filled($p->hotspot_username) && filled($p->hotspot_password),
            'actions'  => $this->actionsPour($etat, $s !== null, $session['mode'] ?? null),
        ];
    }

    /** Actions autorisées selon l'état — source unique pour le menu et les boutons. */
    private function actionsPour(string $etat, bool $aSession, ?string $modeSession = null): array
    {
        $cles = match ($etat) {
            Poste::ETAT_DISPONIBLE => ['demarrer', 'debloquer', 'desactiver', 'arreter'],
            Poste::ETAT_OCCUPE     => ['fermer', 'suspendre', 'verrouiller', 'recharger', 'desactiver', 'arreter'],
            Poste::ETAT_SUSPENDU   => ['reprendre', 'fermer', 'desactiver', 'arreter'],
            Poste::ETAT_VERROUILLE => $aSession
                ? ['debloquer', 'fermer', 'recharger', 'desactiver', 'arreter']
                : ['debloquer', 'desactiver', 'arreter'],
            // Poste injoignable avec une session ouverte : le gérant peut la clôturer à la main
            Poste::ETAT_HORS_LIGNE => $aSession ? ['fermer', 'desactiver'] : ['desactiver'],
            Poste::ETAT_DESACTIVE  => ['activer'],
            default                => [],
        };

        // Recharger n'a de sens que pour un forfait (durée limitée)
        if ($modeSession !== 'forfait') {
            $cles = array_values(array_diff($cles, ['recharger']));
        }

        return array_map(function (string $cle) use ($etat) {
            $def = self::ACTIONS[$cle];
            if ($cle === 'debloquer' && $etat === Poste::ETAT_VERROUILLE) {
                $def['label'] = 'Déverrouiller';
            }
            return ['cle' => $cle] + $def;
        }, $cles);
    }

    // Helpers
    // ------------------------------------------------------------------

    /**
     * Clôture la session active (sous verrou) puis coupe Internet.
     * Si le poste est injoignable, la session est facturée jusqu'à son dernier signe de vie.
     */
    private function terminer(Poste $poste, string $motif): ?Session
    {
        $arret = $poste->estEnLigne() ? null : $poste->dernier_heartbeat;

        $session = DB::transaction(function () use ($poste, $motif, $arret) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            return $p !== null ? $this->sessions->cloturer($p, $motif, $arret) : null;
        });

        $this->sessions->deconnecterReseau($poste);   // mode dégradé : n'échoue jamais

        return $session;
    }

    /** Complément de message quand MikroTik n'a pas répondu (false = échec, null/true = rien à signaler). */
    private function avertissementReseau(?bool $reseau, string $verbe = 'activé'): string
    {
        return $reseau === false
            ? " Attention : Internet n'a pas pu être {$verbe} (MikroTik injoignable) — vérifiez le routeur."
            : '';
    }

    private function ok(Request $request, string $message)
    {
        return $request->expectsJson()
            ? response()->json(['ok' => true, 'message' => $message])
            : back()->with('success', $message);
    }

    private function echec(Request $request, string $message)
    {
        return $request->expectsJson()
            ? response()->json(['ok' => false, 'message' => $message], 422)
            : back()->with('error', $message);
    }
}

<?php

namespace App\Services;

use App\Models\Poste;
use App\Models\Session;
use Carbon\CarbonInterface;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

/**
 * Logique partagée entre l'interface admin, l'API agent et la maintenance planifiée :
 * création / suspension / reprise / clôture de session et connexion du poste au réseau (MikroTik).
 *
 * POLITIQUE « poste injoignable pendant une session » :
 *  - le temps d'injoignabilité n'est jamais facturé (facturation jusqu'au dernier heartbeat) ;
 *  - la session reste ouverte pendant la tolérance (config postes.hors_ligne_tolerance_minutes) ;
 *    si le poste revient, la panne est déduite (duree_hors_ligne_secondes) et la session reprend ;
 *  - au-delà de la tolérance, clôture automatique (motif « hors_ligne ») + coupure MikroTik.
 *
 * MODE DÉGRADÉ : aucune méthode réseau ne lève d'exception ; une panne MikroTik est loggée
 * (jamais de mot de passe ni de token dans les logs) et la session s'ouvre / se ferme quand même.
 */
class PosteSessionService
{
    public function __construct(
        private MikroTikService $mikroTik,
        private TarifService $tarifs
    ) {}

    // ------------------------------------------------------------------
    // Cycle de vie de la session (base de données)
    // ------------------------------------------------------------------

    public function creerSession(Poste $poste, string $mode, ?int $duree): Session
    {
        try {
            $tarifId = $this->tarifs->tarifStandard()->id;
        } catch (\Throwable) {
            $tarifId = null;
        }

        $session = Session::create([
            'poste_id'              => $poste->id,
            'type_session'          => 'ethernet',
            'date_heure_debut'      => now(),
            'date_heure_fin_prevue' => $duree ? now()->addMinutes($duree) : null,
            'duree_prevue'          => $duree ?? 0,
            'etat'                  => 'en_cours',
            'tarif_id'              => $tarifId,
            'montant_initial'       => 0,
            'montant_total'         => 0,
            'description'           => "Session {$mode}" . ($duree ? " de {$duree} minutes." : '.'),
        ]);

        $session->mode_facturation = $mode;
        $session->save();

        return $session;
    }

    /**
     * Montant (Ar) de la session. Le mode « libre » n'est jamais facturé.
     *
     * @param int|null $secondes Durée facturable à utiliser ; null = durée consommée à l'instant T
     */
    public function montant(Session $session, ?int $secondes = null): int
    {
        if (($session->mode_facturation ?? 'forfait') === 'libre') {
            return 0;
        }

        try {
            $tarif   = $session->tarif ?: $this->tarifs->tarifStandard();
            $minutes = intdiv($secondes ?? $session->secondesConsommees(), 60);

            return $this->tarifs->montantPourMinutes($minutes, $tarif);
        } catch (\Throwable) {
            return 0;
        }
    }

    /**
     * Suspend la session active : chrono figé, poste « suspendu ».
     * Base de données uniquement (à appeler dans une transaction) ; la coupure réseau
     * se fait ensuite via deconnecterReseau().
     *
     * @return bool false si le poste n'a aucune session active
     */
    public function suspendre(Poste $poste): bool
    {
        $session = $poste->sessionActive();
        if ($session === null) {
            return false;
        }

        if ($session->etat === 'en_cours') {
            $session->update(['etat' => 'en_pause', 'date_heure_suspension' => now()]);
        }
        $poste->update(['etat' => Poste::ETAT_SUSPENDU]);

        return true;
    }

    /**
     * Reprend la session active (si elle était en pause) et passe le poste « utilisé ».
     * Base de données uniquement ; la reconnexion réseau se fait ensuite via connecterReseau().
     *
     * @return bool false si le poste n'a aucune session active
     */
    public function reprendre(Poste $poste): bool
    {
        $session = $poste->sessionActive();
        if ($session === null) {
            return false;
        }

        if ($session->etat === 'en_pause') {
            $pause = $session->date_heure_suspension
                ? max(0, now()->getTimestamp() - $session->date_heure_suspension->getTimestamp())
                : 0;

            $session->update([
                'etat'                  => 'en_cours',
                'date_heure_reprise'    => now(),
                'duree_suspension'      => (int) $session->duree_suspension + $pause,
                'date_heure_suspension' => null,
            ]);
        }

        $poste->update(['etat' => Poste::ETAT_OCCUPE]);

        return true;
    }

    /**
     * Clôture la session active : calcule et fige le montant, poste « disponible ».
     * Base de données uniquement (à appeler dans une transaction) ; la coupure réseau
     * se fait ensuite via deconnecterReseau().
     *
     * @param CarbonInterface|null $arretA Instant (passé) où la consommation s'est réellement arrêtée
     *                                     (expiration d'un forfait, poste injoignable). Ignoré si la session est en pause.
     */
    public function cloturer(Poste $poste, string $motif = 'client', ?CarbonInterface $arretA = null): ?Session
    {
        $session = $poste->sessionActive();
        if ($session === null) {
            return null;
        }

        $brut     = $session->secondesConsommees();   // avant le changement d'état
        $secondes = $brut;
        $fin      = now();

        if ($arretA !== null && $session->etat === 'en_cours' && $session->date_heure_debut !== null) {
            $arret = $arretA->copy();
            if ($arret->lt($session->date_heure_debut)) {
                $arret = $session->date_heure_debut->copy();
            }
            if ($arret->lt($fin)) {
                // Le chrono comptait en continu : on retire le temps écoulé depuis l'arrêt réel
                $secondes = max(0, $brut - ($fin->getTimestamp() - $arret->getTimestamp()));
                $fin      = $arret;
            }
        }

        $montant = $this->montant($session, $secondes);

        $session->update([
            'etat'                  => 'terminee',
            'date_heure_fin_reelle' => $fin,
            'motif_fin'             => $motif,
            'duree_consommee'       => intdiv($secondes, 60),
            'montant_consomme'      => $montant,
            'montant_total'         => $montant,
        ]);

        $poste->update(['etat' => Poste::ETAT_DISPONIBLE]);

        return $session;
    }

    // ------------------------------------------------------------------
    // Expiration des forfaits
    // ------------------------------------------------------------------

    /** Vrai si la session est un forfait en cours dont le temps est épuisé. */
    public function sessionExpiree(Session $session): bool
    {
        return $session->etat === 'en_cours'
            && ($session->mode_facturation ?? 'forfait') === 'forfait'
            && (int) $session->duree_prevue > 0
            && $session->secondesConsommees() >= (int) $session->duree_prevue * 60;
    }

    /**
     * Clôture la session du poste si son forfait est épuisé, puis coupe Internet.
     * Idempotent et sûr en concurrence (verrou de ligne) : appelable depuis le heartbeat
     * ET depuis la maintenance planifiée.
     */
    public function expirerSiNecessaire(Poste $poste): ?Session
    {
        $apercu = $poste->sessionActive();
        if ($apercu === null || !$this->sessionExpiree($apercu)) {
            return null;
        }

        $session = DB::transaction(function () use ($poste) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || !$this->sessionExpiree($s)) {
                return null;
            }

            // Instant exact où le forfait a été épuisé (en tenant compte des temps non facturés)
            $deductions = (int) $s->duree_suspension + (int) $s->duree_hors_ligne_secondes + (int) $s->duree_pause_secondes;
            $arret      = $s->date_heure_debut->copy()->addSeconds((int) $s->duree_prevue * 60 + $deductions);

            return $this->cloturer($p, 'expiration', $arret->isFuture() ? null : $arret);
        });

        if ($session === null) {
            return null;
        }

        Log::info("Forfait expiré : session #{$session->id} du poste {$poste->nom_poste} clôturée ({$session->montant_total} Ar).");
        $this->deconnecterReseau($poste);

        return $session;
    }

    // ------------------------------------------------------------------
    // Poste injoignable / retour en ligne
    // ------------------------------------------------------------------

    /**
     * Le poste redonne signe de vie après une panne : le temps d'injoignabilité n'est pas facturé.
     *
     * @return int Secondes déduites
     */
    public function deduireTempsHorsLigne(Poste $poste, ?CarbonInterface $dernierSigne): int
    {
        if ($dernierSigne === null) {
            return 0;
        }

        return DB::transaction(function () use ($poste, $dernierSigne) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || $s->etat !== 'en_cours') {
                return 0;
            }

            // La panne ne peut pas avoir commencé avant le début ou la dernière reprise de la session
            $debutPanne = max(
                $dernierSigne->getTimestamp(),
                $s->date_heure_debut?->getTimestamp() ?? 0,
                $s->date_heure_reprise?->getTimestamp() ?? 0
            );
            $panne = now()->getTimestamp() - $debutPanne;
            if ($panne <= Poste::DELAI_HORS_LIGNE) {
                return 0;
            }

            $s->update(['duree_hors_ligne_secondes' => (int) $s->duree_hors_ligne_secondes + $panne]);
            Log::info("Poste {$poste->nom_poste} de retour après {$panne} s : temps de panne déduit de la session #{$s->id}.");

            return $panne;
        });
    }

    /**
     * Retour en ligne d'un poste (redémarrage, coupure) : déduit la panne, applique une éventuelle
     * expiration, puis rétablit Internet si la session est toujours en cours.
     */
    public function traiterRetourEnLigne(Poste $poste, ?CarbonInterface $dernierSigne): void
    {
        $this->deduireTempsHorsLigne($poste, $dernierSigne);

        if ($this->expirerSiNecessaire($poste) !== null) {
            return;
        }

        $session = $poste->sessionActive();
        if ($session !== null && $session->etat === 'en_cours') {
            $this->connecterReseau($poste);
        }
    }

    /**
     * Clôture une session « en cours » dont le poste est injoignable depuis plus longtemps que la tolérance.
     * Facturée jusqu'au dernier signe de vie ; Internet coupé.
     */
    public function cloturerSiInjoignable(Poste $poste, int $toleranceMinutes): ?Session
    {
        $session = DB::transaction(function () use ($poste, $toleranceMinutes) {
            $p = Poste::whereKey($poste->id)->lockForUpdate()->first();
            $s = $p?->sessionActive();
            if ($s === null || $s->etat !== 'en_cours') {
                return null;
            }

            $dernier = $p->dernier_heartbeat;
            if ($dernier !== null && $dernier->gt(now()->subMinutes($toleranceMinutes))) {
                return null;   // revenu entre-temps ou tolérance pas encore écoulée
            }

            $arret = ($dernier === null || $dernier->lt($s->date_heure_debut)) ? $s->date_heure_debut : $dernier;

            return $this->cloturer($p, 'hors_ligne', $arret);
        });

        if ($session === null) {
            return null;
        }

        Log::warning("Poste {$poste->nom_poste} injoignable depuis plus de {$toleranceMinutes} min : session #{$session->id} clôturée automatiquement ({$session->montant_total} Ar, facturée jusqu'au dernier signe de vie).");
        $this->deconnecterReseau($poste);

        return $session;
    }



    // ------------------------------------------------------------------
    // Réseau (MikroTik) — mode dégradé : ne lève jamais d'exception
    // ------------------------------------------------------------------

    /**
     * Le compte HotSpot existe-t-il sur MikroTik ?
     *
     * @return bool|null true/false = vérifié, null = impossible à vérifier (MikroTik désactivé ou injoignable)
     */
    public function compteHotspotExiste(string $username): ?bool
    {
        if (!$this->mikrotikActif()) {
            return null;
        }

        try {
            return $this->mikroTik->findHotspotUser($username) !== null;
        } catch (\Throwable $e) {
            Log::warning('MikroTik : vérification du compte HotSpot impossible (' . $e->getMessage() . ').');
            return null;
        }
    }

    public function mikrotikActif(): bool
    {
        return (bool) config('mikrotik.enabled', true);
    }

    /**
     * Connecte le poste à Internet via MikroTik.
     *
     * @return bool|null true = connecté, false = échec (mode dégradé), null = non applicable
     *                   (MikroTik désactivé, identifiants HotSpot ou MAC absents)
     */
    public function connecterReseau(Poste $poste): ?bool
    {
        if (!$this->mikrotikActif()) {
            Log::info("MikroTik désactivé (MIKROTIK_ENABLED=false) : connexion Internet de {$poste->nom_poste} ignorée.");
            return null;
        }
        if (!$poste->hotspot_username || !$poste->adresse_mac) {
            Log::warning("Poste {$poste->nom_poste} : identifiants HotSpot ou adresse MAC absents, Internet non activé.");
            return null;
        }

        try {
            $this->mikroTik->connecterPosteHotspot(
                $poste->hotspot_username,
                $poste->hotspot_password ?? '',
                $poste->adresse_mac,
                $poste->adresse_ip
            );
            Log::info("MikroTik : poste {$poste->nom_poste} connecté à Internet.");
            return true;
        } catch (\Throwable $e) {
            Log::warning("MikroTik : connexion de {$poste->nom_poste} échouée (mode dégradé, la session continue) : " . $e->getMessage());
            return false;
        }
    }

    /**
     * Coupe l'accès Internet du poste.
     *
     * @return bool|null true = traité, false = échec (mode dégradé), null = non applicable
     */
    public function deconnecterReseau(Poste $poste): ?bool
    {
        if (!$this->mikrotikActif()) {
            Log::info("MikroTik désactivé (MIKROTIK_ENABLED=false) : coupure Internet de {$poste->nom_poste} ignorée.");
            return null;
        }
        if (!$poste->adresse_mac) {
            Log::info("Poste {$poste->nom_poste} : MAC absente, coupure réseau ignorée.");
            return null;
        }

        try {
            $coupe = $this->mikroTik->deconnecterPosteHotspot($poste->adresse_mac, $poste->hotspot_username);
            Log::info($coupe
                ? "MikroTik : Internet coupé pour {$poste->nom_poste}."
                : "MikroTik : aucune session HotSpot active pour {$poste->nom_poste} (déjà coupé).");
            return true;
        } catch (\Throwable $e) {
            Log::warning("MikroTik : coupure Internet de {$poste->nom_poste} échouée (mode dégradé, la session est tout de même terminée) : " . $e->getMessage());
            return false;
        }
    }

    /**
     * Aligne MikroTik sur l'état réel des sessions (auto-réparation) :
     *  - poste avec session « en cours » mais sans session HotSpot → reconnecté ;
     *  - poste sans session « en cours » mais encore connecté → coupé (YouTube résiduel).
     * Rattrape un MikroTik tombé pendant une fermeture, ou redémarré en pleine session.
     *
     * @return array{connectes:int, coupes:int, erreurs:int}
     */
    public function reconcilierReseau(): array
    {
        $bilan = ['connectes' => 0, 'coupes' => 0, 'erreurs' => 0];

        if (!$this->mikrotikActif()) {
            return $bilan;
        }

        try {
            $actives = $this->mikroTik->listHotspotActive();
        } catch (\Throwable $e) {
            Log::warning('MikroTik injoignable : réconciliation réseau ignorée (' . $e->getMessage() . ').');
            $bilan['erreurs']++;
            return $bilan;
        }

        $macsActives = [];
        foreach ($actives as $ligne) {
            $mac = strtoupper((string) ($ligne['mac-address'] ?? ''));
            if ($mac !== '') {
                $macsActives[$mac] = true;
            }
        }

        $cache = Cache::store('file');

        foreach (Poste::whereNotNull('adresse_mac')->get() as $poste) {
            $session          = $poste->etat === Poste::ETAT_DESACTIVE ? null : $poste->sessionActive();
            $doitEtreConnecte = $session !== null && $session->etat === 'en_cours';
            $estConnecte      = isset($macsActives[strtoupper($poste->adresse_mac)]);

            if ($doitEtreConnecte && !$estConnecte) {
                if (!$poste->estEnLigne() || !$poste->hotspot_username) {
                    continue;   // PC éteint ou poste non configuré : rien à reconnecter
                }
                $cle = "reseau:reessai:{$poste->id}";
                if ($cache->has($cle)) {
                    continue;   // évite de répéter un échec toutes les 30 s
                }
                $res = $this->connecterReseau($poste);
                if ($res === true) {
                    $bilan['connectes']++;
                } elseif ($res === false) {
                    $bilan['erreurs']++;
                    $cache->put($cle, true, 120);
                }
            } elseif (!$doitEtreConnecte && $estConnecte) {
                $res = $this->deconnecterReseau($poste);
                if ($res === true) {
                    $bilan['coupes']++;
                } elseif ($res === false) {
                    $bilan['erreurs']++;
                }
            }
        }

        return $bilan;
    }
}

<?php

namespace App\Services;

use App\Services\MikroTik\MikrotikConnection;
use RuntimeException;

/**
 * Façade haut niveau vers l'API RouterOS.
 *
 * Responsabilités :
 *  - gérer la connexion + login ;
 *  - exposer des méthodes métier simples (listHotspotUsers, ...) ;
 *  - isoler toute la cuisine du protocole dans MikrotikConnection.
 *
 * Ne contient AUCUN accès aux Controllers, AUCUN SQL.
 */
class MikroTikService
{
    private ?MikrotikConnection $connection = null;
    private bool $loggedIn = false;

    public function __construct(
        private array $config
    ) {
    }

    public function connection(): MikrotikConnection
    {
        if ($this->connection === null) {
            $this->connection = new MikrotikConnection(
                host:   $this->config['host'],
                port:   (int) $this->config['port'],
                timeout: (float) $this->config['timeout'],
                useSsl: (bool) $this->config['ssl'],
            );
        }
        return $this->connection;
    }

    public function login(): void
    {
        if ($this->loggedIn) {
            return;
        }

        $this->connection()->connect();

        $reply = $this->connection()->sendSentence([
            '/login',
            '=name='     . $this->config['user'],
            '=password=' . $this->config['password'],
        ]);

        if (!$this->replyHasDone($reply)) {
            $msg = $this->extractTrapMessage($reply) ?? 'Login MikroTik échoué';
            $this->disconnect();
            throw new RuntimeException($msg);
        }

        $this->loggedIn = true;
    }

    public function disconnect(): void
    {
        if ($this->connection !== null) {
            $this->connection->disconnect();
        }
        $this->connection = null;
        $this->loggedIn   = false;
    }

    /**
     * Vérifie la connexion et retourne les infos de base du routeur.
     *
     * @return array{identity: ?string, version: ?string, board_name: ?string, uptime: ?string}
     */
    public function testConnection(): array
    {
        $identity = $this->query('/system/identity/print');
        $resource = $this->query('/system/resource/print');

        return [
            'identity'   => $identity[0]['name']       ?? null,
            'version'    => $resource[0]['version']    ?? null,
            'board_name' => $resource[0]['board-name'] ?? null,
            'uptime'     => $resource[0]['uptime']     ?? null,
        ];
    }

    /**
     * Envoie une commande /... et retourne uniquement les enregistrements !re.
     *
     * @param  array<string, scalar>  $arguments
     * @return array<int, array<string,string>>
     */
    public function query(string $command, array $arguments = []): array
    {
        $words = [$command];
        foreach ($arguments as $key => $value) {
            $words[] = "={$key}={$value}";
        }
        return $this->send($words);
    }

    public function queryFiltered(string $command, array $filters = []): array
    {
        $words = [$command];
        foreach ($filters as $key => $value) {
            $words[] = "?{$key}={$value}";
        }
        return $this->send($words);
    }

    public function queryWithId(string $command, string $id, array $arguments = []): array
    {
        $words = [$command, "=.id={$id}"];
        foreach ($arguments as $key => $value) {
            $words[] = "={$key}={$value}";
        }
        return $this->send($words);
    }

    /**
     * @param  string[]  $words
     * @return array<int, array<string,string>>
     */
    private function send(array $words): array
    {
        $this->login();

        $reply = $this->connection()->sendSentence($words);

        if ($this->replyHasTrap($reply)) {
            $msg = $this->extractTrapMessage($reply);
            throw new RuntimeException(
                'MikroTik ' . ($words[0] ?? '?') . ' : ' . $msg
            );
        }

        $records = [];
        foreach ($reply as $item) {
            if ($item['type'] === '!re') {
                $records[] = $item['attributes'];
            }
        }
        return $records;
    }

        /**
     * Normalise une adresse MAC au format RouterOS (AA:BB:CC:DD:EE:FF).
     */
    private function normaliserMac(string $mac): string
    {
        return strtoupper(str_replace('-', ':', trim($mac)));
    }

    /**
     * Retire toutes les sessions HotSpot actives correspondant aux filtres.
     *
     * @param  array<string, scalar>  $filtres
     * @return int Nombre de sessions retirées
     */
    private function retirerSessionsActives(array $filtres): int
    {
        $retirees = 0;

        foreach ($this->queryFiltered('/ip/hotspot/active/print', $filtres) as $ligne) {
            if (isset($ligne['.id'])) {
                $this->queryWithId('/ip/hotspot/active/remove', $ligne['.id']);
                $retirees++;
            }
        }

        return $retirees;
    }

    /**
     * Retrouve l'IP actuelle d'un appareil à partir de sa MAC (table des hôtes HotSpot,
     * puis baux DHCP). Source de vérité : le routeur lui-même.
     */
    public function trouverIpParMac(string $mac): ?string
    {
        $mac = $this->normaliserMac($mac);

        foreach ($this->queryFiltered('/ip/hotspot/host/print', ['mac-address' => $mac]) as $hote) {
            if (!empty($hote['address'])) {
                return $hote['address'];
            }
        }

        foreach ($this->queryFiltered('/ip/dhcp-server/lease/print', ['mac-address' => $mac]) as $bail) {
            $ip = $bail['active-address'] ?? $bail['address'] ?? null;
            if (!empty($ip)) {
                return $ip;
            }
        }

        return null;
    }

    /**
     * Connecte un poste à Internet : ouvre une session HotSpot pour le compte du poste.
     * Idempotent : si le poste est déjà connecté avec ce compte et la bonne IP, ne fait rien.
     *
     * @param  string|null $ipIndice IP connue côté Laravel, utilisée seulement si le routeur ne la trouve pas
     * @return bool true = poste connecté
     *
     * @throws RuntimeException si le routeur est injoignable ou refuse la connexion
     */
    public function connecterPosteHotspot(string $username, string $password, string $mac, ?string $ipIndice = null): bool
    {
        $mac = $this->normaliserMac($mac);
        $ip  = $this->trouverIpParMac($mac) ?? ($ipIndice !== '' ? $ipIndice : null);

        $actives = $this->queryFiltered('/ip/hotspot/active/print', ['mac-address' => $mac]);
        foreach ($actives as $ligne) {
            $memeCompte = ($ligne['user'] ?? null) === $username;
            $memeIp     = $ip === null || ($ligne['address'] ?? null) === $ip;
            if ($memeCompte && $memeIp) {
                return true;   // déjà connecté
            }
        }

        // Autre compte / ancienne IP sur cette MAC, ou ce compte déjà utilisé ailleurs (shared-users) : on libère
        $this->retirerSessionsActives(['mac-address' => $mac]);
        $this->retirerSessionsActives(['user' => $username]);

        if ($ip === null) {
            throw new RuntimeException("Adresse IP introuvable sur MikroTik pour la MAC {$mac}.");
        }

        $this->query('/ip/hotspot/active/login', [
            'user'        => $username,
            'password'    => $password,
            'ip'          => $ip,
            'mac-address' => $mac,
        ]);

        return true;
    }

    /**
     * Coupe Internet pour un poste : retire ses sessions HotSpot actives (par MAC et par compte)
     * et purge ses cookies HotSpot pour empêcher une reconnexion automatique.
     *
     * @return bool true si au moins une session active a été retirée
     */
    public function deconnecterPosteHotspot(string $mac, ?string $username = null): bool
    {
        $mac = $this->normaliserMac($mac);

        $retirees = $this->retirerSessionsActives(['mac-address' => $mac]);
        if ($username !== null && $username !== '') {
            $retirees += $this->retirerSessionsActives(['user' => $username]);
        }

        // Cookies HotSpot : sans eux, le navigateur ne peut pas se reconnecter tout seul
        try {
            foreach ($this->queryFiltered('/ip/hotspot/cookie/print', ['mac-address' => $mac]) as $cookie) {
                if (isset($cookie['.id'])) {
                    $this->queryWithId('/ip/hotspot/cookie/remove', $cookie['.id']);
                }
            }
        } catch (\Throwable) {
            // non bloquant : la session active est déjà coupée
        }

        return $retirees > 0;
    }

    /**
     * Convertit une durée en minutes au format uptime RouterOS.
     * Ex: 30 → "30m", 90 → "1h30m", 1500 → "1d1h".
     */
    public static function formatUptime(int $minutes): string
    {
        if ($minutes <= 0) {
            return '1m';
        }

        $days  = intdiv($minutes, 1440);
        $hours = intdiv($minutes % 1440, 60);
        $mins  = $minutes % 60;

        $out = '';
        if ($days  > 0) { $out .= "{$days}d"; }
        if ($hours > 0) { $out .= "{$hours}h"; }
        if ($mins  > 0) { $out .= "{$mins}m"; }

        return $out ?: '1m';
    }

    private function replyHasDone(array $reply): bool
    {
        foreach ($reply as $item) {
            if ($item['type'] === '!done') {
                return true;
            }
        }
        return false;
    }

    private function replyHasTrap(array $reply): bool
    {
        foreach ($reply as $item) {
            if ($item['type'] === '!trap' || $item['type'] === '!fatal') {
                return true;
            }
        }
        return false;
    }

    private function extractTrapMessage(array $reply): ?string
    {
        foreach ($reply as $item) {
            if ($item['type'] === '!trap' || $item['type'] === '!fatal') {
                return $item['attributes']['message']
                    ?? $item['attributes']['category']
                    ?? 'Erreur inconnue';
            }
        }
        return null;
    }

        /**
     * Autorise l'adresse IP de ce serveur à être vue par les clients non connectés.
     */
    public function configurerWalledGarden(string $ip, string $port = '80'): void
    {
        // On vérifie si la règle exacte (IP + Port) existe déjà
        $existantes = $this->queryFiltered('/ip/hotspot/walled-garden/ip/print', [
            'dst-address' => $ip,
            'dst-port'    => $port
        ]);
        
        if (empty($existantes)) {
            $this->query('/ip/hotspot/walled-garden/ip/add', [
                'dst-address' => $ip,
                'dst-port'    => $port,
                'protocol'    => 'tcp',
                'action'      => 'accept',
                'comment'     => 'Portail Captif Laravel (Auto)'
            ]);
        }
    }

    /**
     * Ajoute ou met à jour un nom de domaine local sur le routeur.
     * Ex: portail.cyber pointe vers l'IP de Laravel.
     */
    public function configurerDnsLocal(string $domaine, string $ip): void
    {
        $ip = trim($ip);
        $existantes = $this->queryFiltered('/ip/dns/static/print', ['name' => $domaine]);

        if (!empty($existantes)) {
            $id = $existantes[0]['.id'];
            $this->queryWithId('/ip/dns/static/set', $id, ['address' => $ip]);
        } else {
            $this->query('/ip/dns/static/add', [
                'name'    => $domaine,
                'address' => $ip,
                'comment' => 'Portail Captif Laravel (Auto)'
            ]);
        }
    }

        /**
     * Configure le nom de domaine (DNS Name) dans le profil HotSpot.
     */
    public function configurerDnsNameHotspot(string $nomDeDomaine): void
    {
        $profils = $this->query('/ip/hotspot/profile/print');
        
        if (empty($profils)) {
            throw new \RuntimeException("Aucun profil HotSpot trouvé sur le routeur.");
        }

        // On boucle sur TOUS les profils pour être sûr que le bon soit configuré
        foreach ($profils as $profil) {
            $this->queryWithId('/ip/hotspot/profile/set', $profil['.id'], [
                'dns-name' => $nomDeDomaine,
            ]);
        }
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;

class Poste extends Model
{
    public const ETAT_DISPONIBLE  = 'disponible';
    public const ETAT_OCCUPE      = 'occupe';
    public const ETAT_SUSPENDU    = 'suspendu';
    public const ETAT_VERROUILLE  = 'verrouille';
    public const ETAT_DESACTIVE   = 'desactive';
    public const ETAT_HORS_LIGNE  = 'hors_ligne';

    /** Délai (secondes) sans heartbeat avant de considérer le poste hors ligne. */
    public const DELAI_HORS_LIGNE = 15;

    protected $fillable = [
        'nom_poste', 'nom_windows', 'adresse_mac', 'adresse_ip',
        'type_connexion', 'etat', 'actif', 'derniere_communication',
        'agent_url', 'identifiant_unique', 'agent_token',
        'dernier_heartbeat', 'agent_version',
        'hotspot_username', 'hotspot_password',
    ];

    protected $casts = [
        'actif' => 'boolean',
        'derniere_communication' => 'datetime',
        'dernier_heartbeat' => 'datetime',
    ];

    protected $hidden = ['agent_token', 'hotspot_password'];

    public function sessions(): HasMany
    {
        return $this->hasMany(Session::class);
    }

    public function commandes(): HasMany
    {
        return $this->hasMany(PosteCommande::class);
    }

    public function messages(): HasMany
    {
        return $this->hasMany(PosteMessage::class);
    }

    /** Session active (en_cours ou en_pause) */
    public function sessionActive(): ?Session
    {
        return $this->sessions()
            ->whereIn('etat', ['en_cours', 'en_pause'])
            ->latest()
            ->first();
    }

    public function estEnLigne(): bool
    {
        return $this->dernier_heartbeat
            && $this->dernier_heartbeat->gt(now()->subSeconds(self::DELAI_HORS_LIGNE));
    }

    public function estConnecte(): bool
    {
        return $this->estEnLigne()
            && $this->etat !== self::ETAT_HORS_LIGNE
            && $this->etat !== self::ETAT_DESACTIVE;
    }

    /**
     * État à afficher : calculé, jamais écrasé en base, afin de ne pas perdre
     * l'état logique (occupé, suspendu...) quand l'agent coupe temporairement.
     */
    public function etatAffiche(): string
    {
        if ($this->etat === self::ETAT_DESACTIVE) {
            return self::ETAT_DESACTIVE;
        }
        if (!$this->estEnLigne()) {
            return self::ETAT_HORS_LIGNE;
        }
        // Anciennes lignes où "hors_ligne" a été persisté
        if ($this->etat === self::ETAT_HORS_LIGNE) {
            return $this->sessionActive() ? self::ETAT_OCCUPE : self::ETAT_DISPONIBLE;
        }
        return $this->etat;
    }
}

    public function estTermine(): bool    { return $this->etat_wifi === self::WIFI_TERMINE; }

    {
        return $this->etat_wifi === self::WIFI_CONNECTE;
    }

    /**
     * Secondes facturables écoulées (hors périodes déconnecté / en pause).
     *
     * On travaille sur des timestamps entiers : avec Carbon 3, diffInSeconds()
     * renvoie un float signé, ce qui provoquait des conversions implicites
     * float → int (dépréciées en PHP 8.1+).
     */
    public function secondesConsommees(): int
    {
        if (!$this->date_heure_debut) return 0;

        if ($this->estTermine()) {
            return max(0, (int) $this->duree_consommee * 60);
        }

        if ($this->estEnPause() && $this->pause_at) {
            $fin = $this->pause_at;
        } elseif ($this->estDeconnecte() && $this->deconnecte_at) {
            $fin = $this->deconnecte_at;
        } elseif ($this->etat === 'en_pause' && $this->date_heure_suspension) {
            $fin = $this->date_heure_suspension;      // postes Ethernet : chrono figé pendant la suspension
        } elseif ($this->etat === 'terminee' && $this->date_heure_fin_reelle) {
            $fin = $this->date_heure_fin_reelle;      // postes Ethernet : session clôturée
        } else {
            $fin = now();
        }

        $total = $fin->getTimestamp() - $this->date_heure_debut->getTimestamp();
        $total -= (int) $this->duree_hors_ligne_secondes;
        $total -= (int) $this->duree_pause_secondes;
        $total -= (int) $this->duree_suspension;      // pauses Ethernet déjà terminées

        return max(0, (int) $total);
    }
}

<?php

namespace App\Console\Commands;

use App\Models\Session;
use App\Services\PosteSessionService;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Log;

/**
 * Maintenance des postes clients.
 *
 *  postes:maintenance           → expire les forfaits terminés + clôture les sessions de postes
 *                                  injoignables au-delà de la tolérance (voir config/postes.php)
 *  postes:maintenance --reseau  → réconcilie MikroTik avec l'état réel des sessions
 *
 * Les verrous utilisent le cache « file » : aucune dépendance à la table cache de la base.
 */
class MaintenancePostes extends Command
{
    protected $signature = 'postes:maintenance {--reseau : Réconcilie l\'état réseau MikroTik (connecté / coupé)}';

    protected $description = 'Expire les forfaits, clôture les sessions de postes injoignables, réconcilie MikroTik.';

    public function handle(PosteSessionService $sessions): int
    {
        $reseau = (bool) $this->option('reseau');

        // Un seul exemplaire à la fois (évite les doubles clôtures si un passage traîne)
        $verrou = Cache::store('file')->lock($reseau ? 'postes:maintenance:reseau' : 'postes:maintenance', 120);
        if (!$verrou->get()) {
            return self::SUCCESS;
        }

        try {
            if ($reseau) {
                $this->reconcilierReseau($sessions);
            } else {
                $this->expirerLesForfaits($sessions);
                $this->cloturerLesPostesInjoignables($sessions);
            }
        } catch (\Throwable $e) {
            Log::error('Maintenance des postes : erreur inattendue : ' . $e->getMessage());
            return self::FAILURE;
        } finally {
            $verrou->release();
        }

        return self::SUCCESS;
    }

    /** Forfaits dont le temps est écoulé (postes en ligne ou non). */
    private function expirerLesForfaits(PosteSessionService $sessions): void
    {
        $candidates = Session::query()
            ->where('etat', 'en_cours')
            ->where('mode_facturation', 'forfait')
            ->where('duree_prevue', '>', 0)
            ->whereNotNull('poste_id')
            ->with('poste')
            ->get();

        foreach ($candidates as $session) {
            $poste = $session->poste;
            if ($poste === null || !$sessions->sessionExpiree($session)) {
                continue;
            }
            $sessions->expirerSiNecessaire($poste);
        }
    }

    /** Sessions « en cours » dont le poste ne donne plus signe de vie depuis plus que la tolérance. */
    private function cloturerLesPostesInjoignables(PosteSessionService $sessions): void
    {
        $magasin    = Cache::store('file');
        $maintenant = now()->getTimestamp();

        // Garde de redémarrage : si le planificateur était arrêté, on laisse les agents se reconnecter
        $dernierPassage = $magasin->get('postes:maintenance:dernier_passage');
        $magasin->put('postes:maintenance:dernier_passage', $maintenant, 86400);
        if ($dernierPassage === null || $maintenant - (int) $dernierPassage > 60) {
            $garde = (int) config('postes.garde_redemarrage_secondes', 90);
            $magasin->put('postes:maintenance:garde_jusqua', $maintenant + $garde, 86400);
            Log::info("Maintenance des postes (re)démarrée : pas de clôture pour injoignabilité pendant {$garde} s.");
        }
        if ((int) $magasin->get('postes:maintenance:garde_jusqua', 0) > $maintenant) {
            return;
        }

        $tolerance = (int) config('postes.hors_ligne_tolerance_minutes', 10);
        if ($tolerance <= 0) {
            return;   // clôture automatique désactivée : le gérant ferme à la main
        }

        $sessionsEnCours = Session::query()
            ->where('etat', 'en_cours')
            ->whereNotNull('poste_id')
            ->with('poste')
            ->get();

        foreach ($sessionsEnCours as $session) {
            $poste = $session->poste;
            if ($poste === null || $poste->estEnLigne()) {
                continue;
            }
            if ($poste->dernier_heartbeat && $poste->dernier_heartbeat->gt(now()->subMinutes($tolerance))) {
                continue;   // encore dans la tolérance
            }
            $sessions->cloturerSiInjoignable($poste, $tolerance);
        }
    }

    private function reconcilierReseau(PosteSessionService $sessions): void
    {
        $bilan = $sessions->reconcilierReseau();

        if ($bilan['connectes'] > 0 || $bilan['coupes'] > 0) {
            Log::info("Réconciliation MikroTik : {$bilan['connectes']} poste(s) reconnecté(s), {$bilan['coupes']} coupé(s).");
        }
    }
}

@extends('layouts.cyber-manager', ['title' => 'Postes — Cyber Manager'])

@section('content')
<script>
const POSTES_URL = @json(url('/postes'));

const COULEURS = {
    disponible: '#22c55e', occupe: '#f59e0b', suspendu: '#8b5cf6',
    verrouille: '#ef4444', desactive: '#94a3b8', hors_ligne: '#94a3b8',
};
const LIBELLES = {
    disponible: 'Libre', occupe: 'Utilisé', suspendu: 'En pause',
    verrouille: 'Verrouillé', desactive: 'Désactivé', hors_ligne: 'Non communiqué',
};
const MODES = { forfait: 'Forfait', postpaye: 'Postpayé', libre: 'Débloqué (sans facturation)' };

function svgIcone(etat, taille = 'h-9 w-9') {
    const c = COULEURS[etat] || '#94a3b8';
    let badge = '';
    if (etat === 'hors_ligne') {
        badge = '<circle cx="31" cy="31" r="8" fill="#dc2626" stroke="#fff" stroke-width="1.5"/>'
              + '<rect x="30" y="26" width="2" height="6" fill="#fff"/><rect x="30" y="33" width="2" height="2" fill="#fff"/>';
    } else if (etat === 'desactive') {
        badge = '<line x1="6" y1="30" x2="30" y2="6" stroke="#475569" stroke-width="3" stroke-linecap="round"/>';
    } else if (etat === 'verrouille') {
        badge = '<rect x="26" y="25" width="10" height="8" rx="1.5" fill="#fff" stroke="#b91c1c" stroke-width="1.5"/>'
              + '<path d="M28 25v-3a3 3 0 0 1 6 0v3" fill="none" stroke="#b91c1c" stroke-width="1.5"/>';
    } else if (etat === 'suspendu') {
        badge = '<rect x="12" y="10" width="3.5" height="11" fill="#fff"/><rect x="20.5" y="10" width="3.5" height="11" fill="#fff"/>';
    }
    return `<svg viewBox="0 0 38 38" class="${taille}">
        <rect x="3" y="4" width="26" height="19" rx="2.5" fill="${c}" stroke="#1e293b" stroke-width="1.5"/>
        <rect x="6" y="7" width="20" height="13" rx="1" fill="#0f172a" opacity=".35"/>
        <rect x="13" y="23" width="6" height="4" fill="#64748b"/><rect x="9" y="27" width="14" height="2.5" rx="1" fill="#475569"/>
        ${badge}</svg>`;
}

function fmtDuree(s) {
    if (s === null || s === undefined) return '—';
    s = Math.max(0, Math.floor(s));
    const p = n => String(n).padStart(2, '0');
    return `${p(Math.floor(s / 3600))}:${p(Math.floor((s % 3600) / 60))}:${p(s % 60)}`;
}
const fmtAr = n => (Number(n) || 0).toLocaleString('fr-FR') + ' Ar';

const BOUTONS = {
    vert:   'bg-emerald-600 hover:bg-emerald-700 text-white',
    rouge:  'bg-rose-600 hover:bg-rose-700 text-white',
    ambre:  'bg-amber-500 hover:bg-amber-600 text-white',
    violet: 'bg-violet-600 hover:bg-violet-700 text-white',
    bleu:   'bg-sky-600 hover:bg-sky-700 text-white',
    gris:   'bg-slate-600 hover:bg-slate-700 text-white',
};

function postesApp() {
    return {
        postes: @json($postes),
        selId: null,
        menu: { ouvert: false, x: 0, y: 0, id: null },
        modal: null,
        form: { duree: 60, type: 'forfait', minutes: 30 },
        hs: { username: '', password: '', voir: false, erreur: '' },
        chat: [], chatDernier: 0, texte: '',
        nonLusVus: {},
        maintenant: Date.now(), recuLe: Date.now(),
        horsServeur: false, notif: null, occupe: false,

        init() {
            this.selId = this.postes[0]?.id ?? null;
            this.postes.forEach(p => this.nonLusVus[p.id] = p.non_lus ?? 0);
            if (this.selId) this.chargerChat(true);
            setInterval(() => this.maintenant = Date.now(), 1000);
            setInterval(() => this.rafraichir(), 3000);
            document.addEventListener('click', () => this.menu.ouvert = false);
            document.addEventListener('keydown', e => { if (e.key === 'Escape') { this.menu.ouvert = false; this.modal = null; } });
        },

        get sel() { return this.postes.find(p => p.id === this.selId) || null; },
        get bilan() {
            const n = e => this.postes.filter(p => p.etat === e).length;
            return `${this.postes.length} poste(s) — ${n('occupe') + n('suspendu') + n('verrouille')} utilisé(s), ${n('hors_ligne')} non communiqué(s)`;
        },
        // Messages non lus des postes autres que celui affiché
        get totalNonLus() {
            return this.postes.filter(p => p.id !== this.selId).reduce((t, p) => t + (p.non_lus || 0), 0);
        },

        icone: svgIcone,
        libelle(e) { return LIBELLES[e] || e; },
        mode(m) { return MODES[m] || m; },
        classeBouton(c) { return BOUTONS[c] || BOUTONS.gris; },
        fmtAr,
        titreModal() {
            const t = { demarrer: 'Démarrer — ', recharger: 'Recharger — ', hotspot: 'Identifiants HotSpot — ' };
            return (t[this.modal] || '') + (this.sel?.nom ?? '');
        },

        // Le chrono avance localement entre deux polls
        decalage(p) {
            if (!p.session || !['occupe', 'verrouille'].includes(p.etat)) return 0;
            return Math.floor((this.maintenant - this.recuLe) / 1000);
        },
        ecoule(p) { return p.session ? fmtDuree(p.session.ecoule_s + this.decalage(p)) : '—'; },
        restant(p) {
            if (!p.session || p.session.restant_s === null) return '—';
            return fmtDuree(p.session.restant_s - this.decalage(p));
        },
        tempsListe(p) {
            if (!p.session) return '';
            return p.session.restant_s !== null ? this.restant(p) : this.ecoule(p);
        },

        async rafraichir() {
            try {
                const r = await cmFetch(POSTES_URL + '/etat');
                const d = await r.json();
                // Nouveau message d'un client sur un poste non affiché → notification
                for (const p of d.postes) {
                    const avant = this.nonLusVus[p.id] ?? 0;
                    if ((p.non_lus || 0) > avant && p.id !== this.selId) {
                        this.afficher(`Nouveau message de ${p.nom}`, true);
                    }
                    this.nonLusVus[p.id] = p.non_lus || 0;
                }
                this.postes = d.postes;
                this.recuLe = Date.now();
                this.horsServeur = false;
                if (this.selId && !this.sel) this.selId = null;
                if (this.selId) this.chargerChat(false);
            } catch (e) { this.horsServeur = true; }
        },

        choisir(id) { this.selId = id; this.chat = []; this.chatDernier = 0; this.chargerChat(true); },

        ouvrirMenu(e, id) {
            this.selId = id;
            this.menu = { ouvert: true, x: Math.min(e.clientX, window.innerWidth - 220), y: Math.min(e.clientY, window.innerHeight - 320), id };
            this.chat = []; this.chatDernier = 0; this.chargerChat(true);
        },

        async lancer(id, action) {
            this.menu.ouvert = false;
            this.selId = id;
            const def = (this.postes.find(p => p.id === id)?.actions || []).find(a => a.cle === action);
            if (!def) return;
            if (def.modal) { this.modal = action; return; }
            if (def.confirm && !confirm(def.confirm)) return;
            await this.envoyer(id, action, {});
        },

        async envoyer(id, action, corps) {
            if (this.occupe) return;
            this.occupe = true;
            try {
                const r = await cmFetch(`${POSTES_URL}/${id}/${action}`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify(corps),
                });
                const d = await r.json().catch(() => ({}));
                this.afficher(d.message || (r.ok ? 'Action effectuée.' : 'Erreur.'), r.ok);
                if (r.ok) this.modal = null;
            } catch (e) { this.afficher('Serveur injoignable.', false); }
            this.occupe = false;
            await this.rafraichir();
        },

        validerModal() {
            const id = this.selId;
            if (this.modal === 'demarrer') {
                this.envoyer(id, 'demarrer', { type: this.form.type, duree_minutes: this.form.type === 'forfait' ? Number(this.form.duree) : null });
            } else if (this.modal === 'recharger') {
                this.envoyer(id, 'recharger', { minutes: Number(this.form.minutes) });
            } else if (this.modal === 'hotspot') {
                this.enregistrerHotspot();
            }
        },

        // --- Identifiants HotSpot ---
        ouvrirHotspot() {
            if (!this.sel) return;
            this.hs = { username: this.sel.hotspot_username || '', password: '', voir: false, erreur: '' };
            this.modal = 'hotspot';
        },

        async enregistrerHotspot() {
            if (this.occupe) return;
            const u = this.hs.username.trim(), p = this.hs.password.trim();
            if (/\s/.test(u) || /\s/.test(p)) { this.hs.erreur = "Aucun espace n'est autorisé."; return; }
            if (u && u.length < 3) { this.hs.erreur = "L'identifiant doit contenir au moins 3 caractères."; return; }
            if (p && p.length < 4) { this.hs.erreur = 'Le mot de passe doit contenir au moins 4 caractères.'; return; }

            this.hs.erreur = '';
            this.occupe = true;
            try {
                const r = await cmFetch(`${POSTES_URL}/${this.selId}/hotspot`, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ hotspot_username: u || null, hotspot_password: p || null }),
                });
                const d = await r.json().catch(() => ({}));
                if (r.ok) {
                    this.modal = null;
                    this.afficher(d.message || 'Identifiants enregistrés.', true, 7000);
                } else {
                    this.hs.erreur = (d.errors && Object.values(d.errors)[0]?.[0]) || d.message || 'Erreur.';
                }
            } catch (e) { this.hs.erreur = 'Serveur injoignable.'; }
            this.occupe = false;
            await this.rafraichir();
        },

        async chargerChat(reset) {
            const id = this.selId;
            if (!id) return;
            try {
                const r = await cmFetch(`${POSTES_URL}/${id}/chat?apres=${reset ? 0 : this.chatDernier}`);
                const d = await r.json();
                if (id !== this.selId) return;
                if (reset) this.chat = [];
                if (d.messages?.length) {
                    this.chat.push(...d.messages);
                    this.chatDernier = d.messages[d.messages.length - 1].id;
                    this.$nextTick(() => { const b = this.$refs.chatBox; if (b) b.scrollTop = b.scrollHeight; });
                }
            } catch (e) {}
        },

        async envoyerMessage() {
            const t = this.texte.trim();
            if (!t || !this.selId) return;
            this.texte = '';
            await this.envoyer(this.selId, 'message', { message: t });
            this.chargerChat(false);
        },

        afficher(msg, ok, ms = 3000) {
            this.notif = { msg, ok };
            setTimeout(() => this.notif = null, ms);
        },
    };
}
