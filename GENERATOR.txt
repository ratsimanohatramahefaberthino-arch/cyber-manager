    /**
     * "Arrêter" : fige le compteur, calcule le montant final (arrondi),
     * déconnecte le client du MikroTik, met la session en "terminé".
     * Le client ne peut plus utiliser ce voucher tant que le responsable
     * n'a pas cliqué sur "Démarrer" (rare) ou "Supprimer".
     */
    public function arreter(Session $session): JsonResponse
    {
        if ($session->type_session !== 'wifi') {
            return response()->json(['ok' => false, 'message' => 'Session non Wi-Fi.'], 422);
        }

        if ($session->estTermine()) {
            return response()->json(['ok' => false, 'message' => 'Déjà terminée.'], 422);
        }

        $final = $this->montant->montantFinal($session);

        $session->update([
            'etat'                  => 'terminee',
            'etat_wifi'             => Session::WIFI_TERMINE,
            'date_heure_fin_reelle' => now(),
            'montant_consomme'      => $final['final'],
            'montant_total'         => $final['final'],
            'montant_restant'       => 0,
            'alarme_active'         => false,
            'motif_fin'             => 'arret_manuel',
            'duree_consommee'       => $final['minutes_reelles'],
        ]);

        // Déconnecter côté MikroTik
        if ($session->client_mac) {
            try {
                $this->mikroTik->disconnectHotspotActiveByMac($session->client_mac);
            } catch (\Throwable $e) {}
        }

        return response()->json([
            'ok'      => true,
            'message' => "Session arrêtée. Montant à encaisser : "
                . number_format($final['final'], 0, ',', ' ') . " Ar.",
            'final'   => $final,
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    /**
     * "Démarrer" : reprend une session arrêtée. Le client se reconnecte,
     * on remet l'état à "connecté", le compteur reprend à la seconde près.
     */
    public function demarrer(Session $session): JsonResponse
    {
        if ($session->type_session !== 'wifi') {
            return response()->json(['ok' => false, 'message' => 'Session non Wi-Fi.'], 422);
        }

        if (!$session->estTermine()) {
            return response()->json(['ok' => false, 'message' => 'La session n\'est pas terminée.'], 422);
        }

        $session->update([
            'etat'                  => 'en_cours',
            'etat_wifi'             => Session::WIFI_CONNECTE,
            'date_heure_fin_reelle' => null,
            'motif_fin'             => null,
        ]);

        return response()->json([
            'ok'      => true,
            'message' => 'Session reprise.',
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    /**
     * Suppression définitive : déconnecte, efface le voucher du MikroTik,
     * supprime la session. Autorisé même si connecté (déconnexion auto).
     */
    public function supprimer(Session $session): JsonResponse
    {
        if ($session->type_session !== 'wifi') {
            return response()->json(['ok' => false, 'message' => 'Session non Wi-Fi.'], 422);
        }

        // Voucher protégé → redirection HotSpot (sécurité)
        if ($session->voucher && $session->voucher->protege) {
            return response()->json([
                'ok'       => false,
                'redirect' => route('hotspot.index', ['q' => $session->voucher->username]),
                'message'  => "Cet identifiant est protégé. Supprimez-le depuis HotSpot.",
            ], 403);
        }

        // Déconnecter le client du MikroTik si encore actif
        if ($session->client_mac) {
            try {
                $this->mikroTik->disconnectHotspotActiveByMac($session->client_mac);
            } catch (\Throwable $e) {}
        }

        // Supprimer le voucher temporaire associé
        if ($session->voucher && !$session->voucher->estReutilisable()) {
            try {
                $this->mikroTik->deleteHotspotUser($session->voucher->username);
            } catch (\Throwable $e) {}
            $session->voucher->delete();
        }

        $session->delete();

        return response()->json([
            'ok'      => true,
            'message' => 'Appareil et identifiant supprimés.',
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    /** Suppression en masse. */
    public function supprimerMasse(Request $request): JsonResponse
    {
        $data = $request->validate([
            'ids'   => ['required', 'array', 'min:1', 'max:100'],
            'ids.*' => ['integer'],
        ]);

        $sessions = Session::whereIn('id', $data['ids'])
            ->where('type_session', 'wifi')
            ->get();

        $supprimes = 0;
        $proteges  = 0;

        foreach ($sessions as $s) {
            if ($s->voucher && $s->voucher->protege) { $proteges++; continue; }

            if ($s->client_mac) {
                try { $this->mikroTik->disconnectHotspotActiveByMac($s->client_mac); } catch (\Throwable $e) {}
            }
            if ($s->voucher && !$s->voucher->estReutilisable()) {
                try { $this->mikroTik->deleteHotspotUser($s->voucher->username); } catch (\Throwable $e) {}
                $s->voucher->delete();
            }
            $s->delete();
            $supprimes++;
        }

        $msg = "{$supprimes} supprimé(s)";
        if ($proteges > 0) $msg .= ", {$proteges} protégé(s) ignoré(s)";

        return response()->json([
            'ok'      => true,
            'message' => $msg . '.',
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    public function definirAlarme(Request $request, Session $session): JsonResponse
    {
        if ($session->type_session !== 'wifi') {
            return response()->json(['ok' => false, 'message' => 'Session non Wi-Fi.'], 422);
        }

        $data = $request->validate([
            'type'    => ['required', 'in:duree,montant'],
            'valeur'  => ['required', 'integer', 'min:1', 'max:1000000'],
            'depart'  => ['required', 'in:debut,maintenant'],
        ]);

        $secondesDeja = $session->secondesConsommees();

        if ($data['type'] === 'duree') {
            $minutes = (int) $data['valeur'];
            $montant = null;
        } else {
            $montant = (int) $data['valeur'];
            $tarif = $session->tarif ?: $this->tarifService->tarifWifi();
            $minutes = $tarif
                ? $this->tarifService->calculerDuree($montant, $tarif)
                : (int) floor($montant / 20);
        }

        $secondesTotal = $minutes * 60;
        if ($data['depart'] === 'debut') {
            if ($secondesDeja >= $secondesTotal) {
                return response()->json([
                    'ok' => false,
                    'message' => "Cette limite est déjà dépassée. Choisissez une valeur plus grande ou 'À partir de maintenant'.",
                ], 422);
            }
            $secondesRestantes = $secondesTotal - $secondesDeja;
        } else {
            $secondesRestantes = $secondesTotal;
        }

        $session->update([
            'alarme_active'        => true,
            'alarme_type'          => $data['type'],
            'alarme_duree_minutes' => $minutes,
            'alarme_montant'       => $montant,
            'alarme_fin_prevue'    => now()->addSeconds($secondesRestantes),
            'alarme_declenchee_at' => null,
            'alarme_message'       => null,
            'alarme_historique'    => array_merge($session->alarme_historique ?? [], [[
                'action' => 'pose',
                'type'   => $data['type'],
                'valeur' => (int) $data['valeur'],
                'minutes'=> $minutes,
                'montant'=> $montant,
                'depart' => $data['depart'],
                'at'     => now()->toIso8601String(),
            ]]),
        ]);

        return response()->json([
            'ok'      => true,
            'message' => "Alarme activée : {$minutes} min.",
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    public function prolongerAlarme(Request $request, Session $session): JsonResponse
    {
        if ($session->type_session !== 'wifi' || !$session->alarme_active) {
            return response()->json(['ok' => false, 'message' => 'Aucune alarme active.'], 422);
        }

        $data = $request->validate([
            'type'   => ['required', 'in:duree,montant'],
            'valeur' => ['required', 'integer', 'min:1', 'max:1000000'],
        ]);

        $tarif = $session->tarif ?: $this->tarifService->tarifWifi();

        if ($data['type'] === 'duree') {
            $minutesAjoutees = (int) $data['valeur'];
            $montantAjoute = $tarif ? $minutesAjoutees * $tarif->montant_par_minute : 0;
        } else {
            $montantAjoute = (int) $data['valeur'];
            $minutesAjoutees = $tarif
                ? $this->tarifService->calculerDuree($montantAjoute, $tarif)
                : (int) floor($montantAjoute / 20);
        }

        $nouvelleDuree = (int) $session->alarme_duree_minutes + $minutesAjoutees;
        $nouveauMontant = $session->alarme_type === 'montant'
            ? (int) $session->alarme_montant + ($data['type'] === 'montant' ? $montantAjoute : 0)
            : null;

        $nouvelleFin = now()->addSeconds(
            max(0, $nouvelleDuree * 60 - $session->secondesConsommees())
        );

        $session->update([
            'alarme_duree_minutes' => $nouvelleDuree,
            'alarme_montant'       => $nouveauMontant,
            'alarme_fin_prevue'    => $nouvelleFin,
            'alarme_declenchee_at' => null,
            'alarme_message'       => null,
            'alarme_historique'    => array_merge($session->alarme_historique ?? [], [[
                'action'           => 'prolongation',
                'type'             => $data['type'],
                'valeur'           => (int) $data['valeur'],
                'minutes_ajoutees' => $minutesAjoutees,
                'at'               => now()->toIso8601String(),
            ]]),
        ]);

        return response()->json([
            'ok'      => true,
            'message' => "Alarme prolongée de {$minutesAjoutees} min.",
            'lignes'  => $this->lignesSerialisees(),
            'stats'   => $this->stats(),
        ]);
    }

    public function annulerAlarme(Session $session): JsonResponse

    public function comptes()
    {
        try {
            $users = $this->mikroTik->listHotspotUsers();
            $erreur = null;
        } catch (\Throwable $e) {
            $users = [];
            $erreur = $e->getMessage();
        }

        $prefix = config('mikrotik.hotspot_user_prefix', 'cm-');

        $comptes = collect($users)->map(function ($u) use ($prefix) {
            $u['est_cyber_manager'] = str_starts_with($u['name'] ?? '', $prefix);
            return $u;
        })->sortBy([
            fn ($a, $b) => ($b['est_cyber_manager'] ?? false) <=> ($a['est_cyber_manager'] ?? false),
            fn ($a, $b) => strcmp($a['name'] ?? '', $b['name'] ?? ''),
        ])->values();

        return view('wifi.comptes', compact('comptes', 'erreur'));
    }

    public function syncNow()
    {
        $r = $this->wifiSessions->synchroniser();
        if ($r['erreur']) return back()->withErrors(['global' => 'MikroTik injoignable : ' . $r['erreur']]);
        return back()->with('success', "Sync : {$r['ouvertes']} ouverte(s), {$r['maj']} mise(s) à jour, {$r['fermees']} déconnectée(s).");
    }

    public function deconnecter(Request $request)
    {
        $mac = (string) $request->input('mac');
        if (!$mac) return back()->withErrors(['global' => 'MAC manquante.']);

        try {
            $ok = $this->mikroTik->disconnectHotspotActiveByMac($mac);
        } catch (\Throwable $e) {
            return back()->withErrors(['global' => $e->getMessage()]);
        }

        if (!$ok) return back()->withErrors(['global' => "Aucune session active avec la MAC {$mac}."]);
        return back()->with('success', 'Client déconnecté.');
    }

    public function toggleCompte(string $username)
    {
        try {
            $user = $this->mikroTik->findHotspotUser($username);
            if (!$user) return back()->withErrors(['global' => "Compte {$username} introuvable."]);

            $disabled = ($user['disabled'] ?? 'false') === 'true';
            $this->mikroTik->setHotspotUserDisabled($username, !$disabled);

            return back()->with('success', "Compte {$username} " . (!$disabled ? 'désactivé' : 'réactivé') . '.');
        } catch (\Throwable $e) {
            return back()->withErrors(['global' => $e->getMessage()]);
        }
    }

    public function supprimerCompte(string $username)
    {
        try {
            $ok = $this->mikroTik->deleteHotspotUser($username);
            if (!$ok) return back()->withErrors(['global' => "Compte {$username} introuvable."]);
            return back()->with('success', "Compte {$username} supprimé.");
        } catch (\Throwable $e) {
            return back()->withErrors(['global' => $e->getMessage()]);
        }
    }

    private function lignesSerialisees(): array
    {
        $sessions = Session::query()
            ->where('type_session', 'wifi')
            ->whereIn('etat_wifi', [
                Session::WIFI_CONNECTE,
                Session::WIFI_DECONNECTE,
                Session::WIFI_EN_PAUSE,
                Session::WIFI_TERMINE,
            ])
            ->with(['appareilWifi', 'voucher', 'tarif'])
            ->orderByDesc('date_heure_debut')
            ->get();

        return $sessions->map(fn ($s) => $this->serialiser($s))->values()->all();
    }

    private function serialiser(Session $s): array
    {
        $tarif = $s->tarif ?: $this->tarifService->tarifWifi();
        $protege = (bool) ($s->voucher?->protege ?? false);

        $etatAppareil = match ($s->etat_wifi) {
            Session::WIFI_CONNECTE   => 'connecte',
            Session::WIFI_DECONNECTE => 'deconnecte',
            Session::WIFI_EN_PAUSE   => 'en_pause',
            Session::WIFI_TERMINE    => 'termine',
            default                  => 'inconnu',
        };

        return [
            'id'               => $s->id,
            'etat_wifi'        => $s->etat_wifi,
            'etat_appareil'    => $etatAppareil,
            'appareil'         => $s->appareilWifi?->libelle() ?? 'Appareil inconnu',
            'appareil_type'    => $s->appareilWifi?->type_appareil ?? 'inconnu',
            'mac'              => $s->client_mac,
            'ip'               => $s->client_ip,
            'ssid'             => $s->ssid,
            'voucher'          => $s->voucher ? [
                'username' => $s->voucher->username,
                'protege'  => $protege,
                'permanent'=> $s->voucher->estReutilisable(),
                'montant'  => (int) $s->voucher->montant,
            ] : null,
            'date_debut'       => $s->date_heure_debut?->toIso8601String(),
            'deconnecte_at'    => $s->deconnecte_at?->toIso8601String(),
            'pause_at'         => $s->pause_at?->toIso8601String(),
            'secondes_hors_ligne' => (int) $s->duree_hors_ligne_secondes,
            'secondes_pause'   => (int) $s->duree_pause_secondes,
            'volume_entree'    => (int) $s->volume_entree,
            'volume_sortie'    => (int) $s->volume_sortie,
            'volume_total'     => (int) $s->volume_total,
            'tarif'            => $tarif ? [
                'montant_par_minute' => $tarif->montant_par_minute,
                'montant_minimum'    => $tarif->montant_minimum,
                'arrondi_actif'      => (bool) $tarif->arrondi_actif,
                'unite_arrondi'      => $tarif->unite_arrondi,
                'seuil_arrondi'      => $tarif->seuil_arrondi,
            ] : null,
            'alarme' => [
                'active'      => (bool) $s->alarme_active,
                'type'        => $s->alarme_type,
                'minutes'     => $s->alarme_duree_minutes,
                'montant'     => $s->alarme_montant,
                'fin'         => $s->alarme_fin_prevue?->toIso8601String(),
                'declenchee'  => (bool) $s->alarme_declenchee_at,
                'acquittee'   => ($s->alarme_message === 'acquittee'),
            ],
        ];
    }

    private function stats(): array
    {
        $sessions = Session::where('type_session', 'wifi')
            ->whereIn('etat_wifi', [
                Session::WIFI_CONNECTE,
                Session::WIFI_DECONNECTE,
                Session::WIFI_EN_PAUSE,
                Session::WIFI_TERMINE,
            ])->get();

        return [
            'connectes'    => $sessions->where('etat_wifi', Session::WIFI_CONNECTE)->count(),
            'deconnectes'  => $sessions->where('etat_wifi', Session::WIFI_DECONNECTE)->count(),
            'termines'     => $sessions->where('etat_wifi', Session::WIFI_TERMINE)->count(),
            'volume_total' => (int) $sessions->sum('volume_total'),
        ];
    }

    /**
     * Vouchers disponibles = temporaires, état "disponible", sans session
     * active en cours. On EXCLUT explicitement tout voucher dont une session
     * Wi-Fi est actuellement "en_cours".
     */
    private function vouchersDisponibles(): array
    {
        $usernamesActifs = Session::where('type_session', 'wifi')
            ->where('etat_wifi', Session::WIFI_CONNECTE)
            ->whereNotNull('hotspot_username')
            ->pluck('hotspot_username')
            ->all();

        $query = Voucher::query()
            ->where('etat', 'disponible')
            ->where('type_voucher', Voucher::TYPE_TEMPORAIRE)
            ->where('source', 'cyber_manager');

        if (!empty($usernamesActifs)) {
            $query->whereNotIn('username', $usernamesActifs);
        }

        return $query->orderBy('id')
            ->limit(5)
            ->get()
            ->map(fn($v) => [
                'id'       => $v->id,
                'username' => $v->username,
                'password' => $v->password,
            ])->values()->all();
    }

    private function vouchersWarning(): ?string
    {
        $n = Voucher::query()
            ->where('etat', 'disponible')
            ->where('type_voucher', Voucher::TYPE_TEMPORAIRE)
            ->where('source', 'cyber_manager')
            ->count();

        if ($n === 0) return "Aucun identifiant disponible. Générez un lot depuis HotSpot.";
        if ($n < 5)   return "Plus que {$n} identifiant(s) disponible(s).";
        return null;
    }

    private function alarmeSettings(): array

            'duree_secondes' => (int) Parametre::get('wifi_alarm_duration', '0'),
            'taille_max_mo'  => (int) Parametre::get('wifi_alarm_max_size', '10'),
            'son_nom'        => $son ?: null,
            'son_url'        => $son ? '/sounds/' . $son : null,
        ];
    }

<?php

namespace App\Http\Controllers;

use Illuminate\Http\JsonResponse;
use Illuminate\Http\Request;

class WifiAlarmController extends Controller
{
    private const CLE_ENABLED = 'wifi_alarm_enabled';
    private const CLE_VOLUME  = 'wifi_alarm_volume';
    private const CLE_SOUND   = 'wifi_alarm_sound';
    private const CLE_DUREE   = 'wifi_alarm_duration';
    private const CLE_TAILLE  = 'wifi_alarm_max_size';   // stocké en Mo (1 à 10)

    private const DOSSIER = 'sounds';

    /**
     * Retourne les réglages actuels.
     * Gère la migration automatique de l'ancienne valeur Ko → Mo.
     */
    public function index(): JsonResponse
    {
        $tailleBrute = (int) Parametre::get(self::CLE_TAILLE, '10');

        // Migration à la volée : si l'ancienne valeur était en Ko (ex. 10240),
        // on la convertit en Mo (10) et on la réenregistre.
        if ($tailleBrute > 10) {
            $tailleBrute = max(1, (int) round($tailleBrute / 1024));
            Parametre::set(self::CLE_TAILLE, (string) $tailleBrute);
        }

        return response()->json([
            'enabled'        => Parametre::get(self::CLE_ENABLED, '1') === '1',
            'volume'         => (int) Parametre::get(self::CLE_VOLUME, '70'),
            'duree_secondes' => (int) Parametre::get(self::CLE_DUREE, '0'),
            'taille_max_mo'  => $tailleBrute,
            'son_nom'        => Parametre::get(self::CLE_SOUND, '') ?: null,
            'son_url'        => ($s = Parametre::get(self::CLE_SOUND, '')) ? '/sounds/' . $s : null,
            'sons_uploades'  => $this->listerSonsUploades(),
        ]);
    }

    /**
     * Enregistre les réglages généraux (activation, volume, durée, taille max).
     */
    public function reglages(Request $request): JsonResponse
    {
        $data = $request->validate([
            'enabled'        => ['required', 'boolean'],
            'volume'         => ['required', 'integer', 'min:0', 'max:100'],
            'duree_secondes' => ['required', 'integer', 'min:0', 'max:600'],
            'taille_max_mo'  => ['required', 'integer', 'min:1', 'max:10'],
        ]);

        Parametre::set(self::CLE_ENABLED, $data['enabled'] ? '1' : '0');
        Parametre::set(self::CLE_VOLUME,  (string) $data['volume']);
        Parametre::set(self::CLE_DUREE,   (string) $data['duree_secondes']);
        Parametre::set(self::CLE_TAILLE,  (string) $data['taille_max_mo']);

        return response()->json(['ok' => true]);
    }

    /**
     * Upload d'un fichier son personnalisé.
     */
    public function uploader(Request $request): JsonResponse
    {
        $tailleMaxMo = (int) Parametre::get(self::CLE_TAILLE, '10');

        // Sécurité : si la valeur stockée est encore l'ancienne (Ko), on borne à 10 Mo
        if ($tailleMaxMo > 10) {
            $tailleMaxMo = 10;
        }

        $tailleMaxKo = $tailleMaxMo * 1024;

        $request->validate([
            'son' => ['required', 'file', 'max:' . $tailleMaxKo, 'mimes:mp3,wav,ogg,m4a,aac'],
        ], [
            'son.max'   => "Le fichier dépasse la taille maximale de {$tailleMaxMo} Mo.",
            'son.mimes' => 'Format non accepté. Utilisez MP3, WAV, OGG, M4A ou AAC.',
        ]);

        $fichier = $request->file('son');
        $nomOriginal = pathinfo($fichier->getClientOriginalName(), PATHINFO_FILENAME);
        $nomOriginal = preg_replace('/[^a-zA-Z0-9_\-]/', '_', $nomOriginal) ?: 'son';
        $ext = strtolower($fichier->getClientOriginalExtension());
        $nom = $nomOriginal . '-' . now()->format('YmdHis') . '.' . $ext;

        if (!is_dir(public_path(self::DOSSIER))) {
            mkdir(public_path(self::DOSSIER), 0755, true);
        }

        $fichier->move(public_path(self::DOSSIER), $nom);

        return response()->json([
            'ok'  => true,
            'nom' => $nom,
            'url' => '/' . self::DOSSIER . '/' . $nom,
        ]);
    }

    /**
     * Choisit le son actif.
     * Cas spécial '__default__' → revient au bip par défaut (synthétisé côté client).
     */
    public function choisirSon(Request $request): JsonResponse
    {
        $data = $request->validate(['nom' => ['required', 'string', 'max:255']]);
        $nom = basename($data['nom']);

        // Retour au son par défaut : on vide le paramètre
        if ($nom === '__default__') {
            Parametre::set(self::CLE_SOUND, '');
            return response()->json(['ok' => true]);
        }

        if (!file_exists(public_path(self::DOSSIER . '/' . $nom))) {
            return response()->json(['ok' => false, 'message' => 'Fichier introuvable.'], 404);
        }

        Parametre::set(self::CLE_SOUND, $nom);

        return response()->json(['ok' => true]);
    }

    /**
     * Supprime un fichier son.
     */
    public function supprimerSon(Request $request): JsonResponse
    {
        $data = $request->validate(['nom' => ['required', 'string', 'max:255']]);
        $nom = basename($data['nom']);

        $chemin = public_path(self::DOSSIER . '/' . $nom);
        if (file_exists($chemin)) {
            @unlink($chemin);
        }

        // Si c'était le son actif, on retombe sur le son par défaut
        if (Parametre::get(self::CLE_SOUND) === $nom) {
            Parametre::set(self::CLE_SOUND, '');
        }

        return response()->json(['ok' => true]);
    }

    /**
     * Réinitialise les réglages (ne touche PAS aux fichiers de public/sounds/).
     */
    public function reinitialiser(): JsonResponse
    {
        Parametre::set(self::CLE_ENABLED, '1');
        Parametre::set(self::CLE_VOLUME,  '70');
        Parametre::set(self::CLE_DUREE,   '0');
        Parametre::set(self::CLE_TAILLE,  '10');
        Parametre::set(self::CLE_SOUND,   '');

        return response()->json(['ok' => true]);
    }

    /**
     * Liste les fichiers audio présents dans public/sounds/.
     */
    private function listerSonsUploades(): array
    {
        $dossier = public_path(self::DOSSIER);
        if (!is_dir($dossier)) return [];

        $fichiers = scandir($dossier);
        $sons = [];
        foreach ($fichiers as $f) {
            if (in_array($f, ['.', '..', '.gitignore', '.gitkeep'])) continue;
            $ext = strtolower(pathinfo($f, PATHINFO_EXTENSION));
            if (!in_array($ext, ['mp3','wav','ogg','m4a','aac'])) continue;
            $sons[] = [
                'nom' => $f,
                'url' => '/' . self::DOSSIER . '/' . $f,
            ];
        }
        return $sons;
    }
}

<?php

namespace App\Services;

use App\Models\AppareilWifi;
use App\Models\Session;
use App\Models\Voucher;
use Illuminate\Support\Facades\Log;

class WifiSessionService
{
    private ?array $baux = null;
    private ?array $hotes = null;
    private ?array $regTable = null;
    private ?array $wireless = null;

    public function __construct(private MikroTikService $mikroTik) {}

    public function synchroniser(): array
    {
        $this->baux     = null;
        $this->hotes    = null;
        $this->regTable = null;
        $this->wireless = null;

        try {
            $actives = $this->mikroTik->listHotspotActive();
        } catch (\Throwable $e) {
            Log::warning('Sync Wi-Fi : MikroTik injoignable', ['error' => $e->getMessage()]);
            return ['ouvertes' => 0, 'maj' => 0, 'fermees' => 0, 'erreur' => $e->getMessage()];
        }

        $ouvertes = 0;
        $maj      = 0;

        foreach ($actives as $active) {
            try {
                $r = $this->traiterActive($active);
                if ($r === 'ouverte') $ouvertes++;
                elseif ($r === 'maj') $maj++;
            } catch (\Throwable $e) {
                Log::warning('Sync Wi-Fi : entrée ignorée', [
                    'active_id' => $active['.id'] ?? null,
                    'error'     => $e->getMessage(),
                ]);
            }
        }

        $fermees = $this->marquerSessionsDisparues($actives);

        return ['ouvertes' => $ouvertes, 'maj' => $maj, 'fermees' => $fermees, 'erreur' => null];
    }

    private function traiterActive(array $active): ?string
    {
        $activeId = $active['.id'] ?? null;
        if (!$activeId) return null;

        $session = Session::where('mikrotik_active_id', $activeId)
            ->whereIn('etat_wifi', [
                Session::WIFI_CONNECTE,
                Session::WIFI_DECONNECTE,
                Session::WIFI_EN_PAUSE,
            ])
            ->first();

        if ($session) {
            $this->mettreAJour($session, $active);
            return 'maj';
        }

        $this->creer($active);
        return 'ouverte';
    }

    private function creer(array $active): Session
    {
        $username = $active['user'] ?? null;
        $voucher  = $username ? Voucher::where('username', $username)->first() : null;

        $appareil = $this->resoudreAppareil($active);
        $ssid     = $this->resoudreSsid($active);

        $uptimeSecondes = $this->parseUptime($active['uptime'] ?? '0s');
        $debut = now()->subSeconds($uptimeSecondes);

        $montantTotal = $voucher?->montant ?? 0;

        $session = Session::create(array_merge([
            'poste_id'              => null,
            'voucher_id'            => $voucher?->id,
            'type_session'          => 'wifi',
            'date_heure_debut'      => $debut,
            'date_heure_fin_prevue' => null,
            'duree_prevue'          => 0,
            'duree_consommee'       => 0,
            'temps_restant'         => 0,
            'duree_suspension'      => 0,
            'montant_initial'       => $montantTotal,
            'montant_recharge'      => 0,
            'montant_total'         => $montantTotal,
            'montant_consomme'      => 0,
            'montant_restant'       => $montantTotal,
            'montant_a_reverser'    => 0,
            'etat'                  => 'en_cours',
            'etat_wifi'             => Session::WIFI_CONNECTE,
            'mikrotik_active_id'    => $active['.id'] ?? null,
            'hotspot_username'      => $username,
            'client_mac'            => $active['mac-address'] ?? null,
            'client_ip'             => $active['address']     ?? null,
            'ssid'                  => $ssid,
            'appareil_wifi_id'      => $appareil?->id,
            'derniere_sync_at'      => now(),
        ], $this->calculerVolumes($active)));

        // ⚡ FIX CLÉ : marquer le voucher comme "en cours" pour qu'il
        // disparaisse IMMÉDIATEMENT de la liste des identifiants disponibles.
        if ($voucher) {
            try {
                $voucher->marquerEnCours($appareil);
            } catch (\Throwable $e) {
                Log::warning('Impossible de marquer le voucher en cours', [
                    'voucher_id' => $voucher->id,
                    'error'      => $e->getMessage(),
                ]);
            }
        }

        return $session;
    }

    private function mettreAJour(Session $session, array $active): void
    {
        if ($session->etat_wifi === Session::WIFI_DECONNECTE) {
            $dureeHorsLigne = $session->deconnecte_at
                ? $session->deconnecte_at->diffInSeconds(now())
                : 0;

            $session->duree_hors_ligne_secondes = (int) $session->duree_hors_ligne_secondes + $dureeHorsLigne;
            $session->etat_wifi     = Session::WIFI_CONNECTE;
            $session->reconnecte_at = now();
            $session->deconnecte_at = null;
        }

        $session->client_ip        = $active['address']     ?? $session->client_ip;
        $session->client_mac       = $active['mac-address'] ?? $session->client_mac;
        $session->derniere_sync_at = now();

        if ($app = $this->resoudreAppareil($active)) {
            $session->appareil_wifi_id = $app->id;
        }

        if (empty($session->ssid)) {
            $session->ssid = $this->resoudreSsid($active);
        }

        if ($session->etat_wifi === Session::WIFI_CONNECTE) {
            $vol = $this->calculerVolumes($active);
            $session->volume_entree = $vol['volume_entree'];
            $session->volume_sortie = $vol['volume_sortie'];
            $session->volume_total  = $vol['volume_total'];
            $session->duree_consommee = (int) floor($session->secondesConsommees() / 60);
        }

        $session->save();
    }

    private function marquerSessionsDisparues(array $actives): int
    {
        $idsActifs = array_filter(array_map(fn($a) => $a['.id'] ?? null, $actives));

        $sessions = Session::where('type_session', 'wifi')
            ->where('etat_wifi', Session::WIFI_CONNECTE)
            ->whereNotNull('mikrotik_active_id')
            ->whereNotIn('mikrotik_active_id', $idsActifs)
            ->get();

        $n = 0;
        foreach ($sessions as $s) {
            $s->etat_wifi     = Session::WIFI_DECONNECTE;
            $s->deconnecte_at = now();
            $s->derniere_sync_at = now();
            $s->duree_consommee = (int) floor($s->secondesConsommees() / 60);
            $s->save();
            $n++;
        }

        return $n;
    }

    private function resoudreSsid(array $active): ?string
    {
        $mac = AppareilWifi::normaliserMac($active['mac-address'] ?? null);

        if ($mac) {
            $reg = $this->registrationPourMac($mac);
            $iface = $reg['interface'] ?? null;

            if ($iface) {
                $wifi = $this->wirelessPourInterface($iface);
                if (!empty($wifi['ssid'])) {
                    return mb_substr($wifi['ssid'], 0, 64);
                }
            }
        }

        $serveur = $active['server'] ?? null;
        if ($serveur) {
            $table = config('mikrotik.ssid_map', []);
            if (isset($table[$serveur])) {
                return mb_substr($table[$serveur], 0, 64);
            }
        }

        $defaut = config('mikrotik.ssid_default');
        return $defaut ? mb_substr($defaut, 0, 64) : null;
    }

    private function registrationPourMac(string $mac): array
    {
        if ($this->regTable === null) {
            $this->regTable = [];
            try {
                foreach ($this->mikroTik->listWirelessRegistrationTable() as $r) {
                    $m = AppareilWifi::normaliserMac($r['mac-address'] ?? null);
                    if ($m) $this->regTable[$m] = $r;
                }
            } catch (\Throwable $e) {}
        }
        return $this->regTable[$mac] ?? [];
    }

    private function wirelessPourInterface(string $iface): array
    {
        if ($this->wireless === null) {
            $this->wireless = [];
            try {
                foreach ($this->mikroTik->listWirelessInterfaces() as $w) {
                    $name = $w['name'] ?? null;
                    if ($name) $this->wireless[$name] = $w;
                }
            } catch (\Throwable $e) {}
        }
        return $this->wireless[$iface] ?? [];
    }

    private function resoudreAppareil(array $active): ?AppareilWifi
    {
        $mac = AppareilWifi::normaliserMac($active['mac-address'] ?? null);
        if (!$mac) return null;

        $bail = $this->bailPourMac($mac);

        $appareil = AppareilWifi::trouverOuCreerParMac($mac, [
            'adresse_ip' => $active['address'] ?? ($bail['address'] ?? null),
            'host_name'  => $bail['host-name'] ?? null,
        ]);

        if (!$appareil->wasRecentlyCreated) {
            $appareil->toucherDerniereConnexion();
        }

        return $appareil;
    }

    private function calculerVolumes(array $active): array
    {
        $bytesIn  = (int) ($active['bytes-in']  ?? 0);
        $bytesOut = (int) ($active['bytes-out'] ?? 0);

        return [
            'volume_entree' => $bytesOut,
            'volume_sortie' => $bytesIn,
            'volume_total'  => $bytesIn + $bytesOut,
        ];
    }

    private function bailPourMac(string $mac): array
    {
        if ($this->baux === null) {
            $this->baux = [];
            try {
                foreach ($this->mikroTik->listDhcpLeases() as $bail) {
                    $m = AppareilWifi::normaliserMac($bail['mac-address'] ?? null);
                    if ($m) $this->baux[$m] = $bail;
                }
            } catch (\Throwable $e) {}
        }
        return $this->baux[$mac] ?? [];
    }

    private function parseUptime(string $uptime): int
    {
        $t = 0;
        if (preg_match('/(\d+)w/', $uptime, $m)) $t += (int)$m[1] * 604800;
        if (preg_match('/(\d+)d/', $uptime, $m)) $t += (int)$m[1] * 86400;
        if (preg_match('/(\d+)h/', $uptime, $m)) $t += (int)$m[1] * 3600;
        if (preg_match('/(\d+)m(?!s)/', $uptime, $m)) $t += (int)$m[1] * 60;
        if (preg_match('/(\d+)s/', $uptime, $m)) $t += (int)$m[1];
        return $t;
    }
}

<?php

namespace App\Services;

use App\Models\Session;
use App\Models\Tarif;

/**
 * Montants Wi-Fi :
 *   - montantTempsReel() : brut (sans arrondi), augmente en continu.
 *   - montantFinal()     : minimum + arrondi appliqués, à l'arrêt.
 */
class WifiMontantService
{
    public function __construct(private TarifService $tarifService) {}

    /** Montant brut affiché pendant la session (pas d'arrondi). */
    public function montantTempsReel(Session $s): int
    {
        $tarif = $this->tarifPour($s);
        if (!$tarif) return 0;

        $secondes = $s->secondesConsommees();
        $minutes  = $secondes / 60;

        return (int) floor($minutes * $tarif->montant_par_minute);
    }

    /** Montant à payer : minimum + arrondi appliqués. */
    public function montantFinal(Session $s): array
    {
        $tarif = $this->tarifPour($s);
        if (!$tarif) {
            return ['brut' => 0, 'final' => 0, 'minutes_reelles' => 0];
        }

        $secondes = $s->secondesConsommees();
        $minutes  = (int) ceil($secondes / 60);

        $brut = $minutes * $tarif->montant_par_minute;
        $brut = $this->tarifService->appliquerMinimum($brut, $tarif);
        $final = $this->tarifService->appliquerArrondi($brut, $tarif);

        return [
            'brut'            => $brut,
            'final'           => $final,
            'minutes_reelles' => $minutes,
        ];
    }

    private function tarifPour(Session $s): ?Tarif
    {
        if ($s->tarif) return $s->tarif;
        try {
            return $this->tarifService->tarifWifi();
        } catch (\Throwable $e) {
            return null;
        }
    }
}

<?php

namespace App\Services;

use App\Services\MikroTik\MikrotikConnection;
use RuntimeException;

/**
 * Façade haut niveau vers l'API RouterOS.
 *
 * Responsabilités :
 *  - gérer la connexion + login ;
 *  - exposer des méthodes métier simples (listHotspotUsers, ...) ;
 *  - isoler toute la cuisine du protocole dans MikrotikConnection.
 *
 * Ne contient AUCUN accès aux Controllers, AUCUN SQL.
 */
class MikroTikService
{
    private ?MikrotikConnection $connection = null;
    private bool $loggedIn = false;

    public function __construct(
        private array $config
    ) {
    }

    public function connection(): MikrotikConnection
    {
        if ($this->connection === null) {
            $this->connection = new MikrotikConnection(
                host:   $this->config['host'],
                port:   (int) $this->config['port'],
                timeout: (float) $this->config['timeout'],
                useSsl: (bool) $this->config['ssl'],
            );
        }
        return $this->connection;
    }

    public function login(): void
    {
        if ($this->loggedIn) {
            return;
        }

        $this->connection()->connect();

        $reply = $this->connection()->sendSentence([
            '/login',
            '=name='     . $this->config['user'],
            '=password=' . $this->config['password'],
        ]);

        if (!$this->replyHasDone($reply)) {
            $msg = $this->extractTrapMessage($reply) ?? 'Login MikroTik échoué';
            $this->disconnect();
            throw new RuntimeException($msg);
        }

        $this->loggedIn = true;
    }

    public function disconnect(): void
    {
        if ($this->connection !== null) {
            $this->connection->disconnect();
        }
        $this->connection = null;
        $this->loggedIn   = false;
    }

    /**
     * Vérifie la connexion et retourne les infos de base du routeur.
     *
     * @return array{identity: ?string, version: ?string, board_name: ?string, uptime: ?string}
     */
    public function testConnection(): array
    {
        $identity = $this->query('/system/identity/print');
        $resource = $this->query('/system/resource/print');

        return [
            'identity'   => $identity[0]['name']       ?? null,
            'version'    => $resource[0]['version']    ?? null,
            'board_name' => $resource[0]['board-name'] ?? null,
            'uptime'     => $resource[0]['uptime']     ?? null,
        ];
    }

    /**
     * Envoie une commande /... et retourne uniquement les enregistrements !re.
     *
     * @param  array<string, scalar>  $arguments
     * @return array<int, array<string,string>>
     */
    public function query(string $command, array $arguments = []): array
    {
        $words = [$command];
        foreach ($arguments as $key => $value) {
            $words[] = "={$key}={$value}";
        }
        return $this->send($words);
    }

    public function queryFiltered(string $command, array $filters = []): array
    {
        $words = [$command];
        foreach ($filters as $key => $value) {
            $words[] = "?{$key}={$value}";
        }
        return $this->send($words);
    }

    public function queryWithId(string $command, string $id, array $arguments = []): array
    {
        $words = [$command, "=.id={$id}"];
        foreach ($arguments as $key => $value) {
            $words[] = "={$key}={$value}";
        }
        return $this->send($words);
    }

    /**
     * @param  string[]  $words
     * @return array<int, array<string,string>>
     */
    private function send(array $words): array
    {
        $this->login();

        $reply = $this->connection()->sendSentence($words);

        if ($this->replyHasTrap($reply)) {
            $msg = $this->extractTrapMessage($reply);
            throw new RuntimeException(
                'MikroTik ' . ($words[0] ?? '?') . ' : ' . $msg
            );
        }

        $records = [];
        foreach ($reply as $item) {
            if ($item['type'] === '!re') {
                $records[] = $item['attributes'];
            }
        }
        return $records;
    }

    /**
     * Liste les utilisateurs HotSpot configurés.
     */
    public function listHotspotUsers(): array
    {
        return $this->query('/ip/hotspot/user/print');
    }

    /**
     * Liste les sessions HotSpot actuellement actives.
     */
    public function listHotspotActive(): array
    {
        return $this->query('/ip/hotspot/active/print');
    }

    /**
     * Liste les profils HotSpot.
     */

     * Ex: 30 → "30m", 90 → "1h30m", 1500 → "1d1h".
     */
    public static function formatUptime(int $minutes): string
    {
        if ($minutes <= 0) {
            return '1m';
        }

        $days  = intdiv($minutes, 1440);
        $hours = intdiv($minutes % 1440, 60);
        $mins  = $minutes % 60;

        $out = '';
        if ($days  > 0) { $out .= "{$days}d"; }
        if ($hours > 0) { $out .= "{$hours}h"; }
        if ($mins  > 0) { $out .= "{$mins}m"; }

        return $out ?: '1m';
    }

                return $item['attributes']['message']
                    ?? $item['attributes']['category']

        }
        return null;

<?php

namespace App\Services\MikroTik;

use RuntimeException;

/**
 * Connexion TCP bas niveau à l'API RouterOS (port 8728 / 8729 SSL).
 *
 * Ne contient AUCUNE logique métier : uniquement l'encodage/décodage
 * du protocole "sentences" de RouterOS.
 *
 * Compatible RouterOS 6.43+ (login simple) et 7.x.
 */
class MikrotikConnection
{
    /** @var resource|null */
    private $socket = null;

    public function __construct(
        private string $host,
        private int $port = 8728,
        private float $timeout = 5.0,
        private bool $useSsl = false,
    ) {
    }

    public function connect(): void
    {
        if ($this->socket !== null) {
            return;
        }

        $transport = $this->useSsl ? 'ssl' : 'tcp';
        $address   = "{$transport}://{$this->host}:{$this->port}";

        // Pour SSL auto-signé (RouterOS par défaut). À durcir plus tard.
        $context = stream_context_create([
            'ssl' => [
                'verify_peer'      => false,
                'verify_peer_name' => false,
            ],
        ]);

        $errno  = 0;
        $errstr = '';

        $socket = @stream_socket_client(
            $address,
            $errno,
            $errstr,
            $this->timeout,
            STREAM_CLIENT_CONNECT,
            $context
        );

        if ($socket === false) {
            throw new RuntimeException(
                "Connexion MikroTik impossible ({$this->host}:{$this->port}) : "
                . "{$errstr} (errno {$errno})"
            );
        }

        stream_set_timeout($socket, (int) $this->timeout);
        $this->socket = $socket;
    }

    public function disconnect(): void
    {
        if ($this->socket !== null) {
            @fclose($this->socket);
            $this->socket = null;
        }
    }

    public function isConnected(): bool
    {
        return $this->socket !== null;
    }

    /**
     * Envoie une phrase (liste de mots) et retourne la réponse complète
     * jusqu'à !done ou !fatal.
     *
     * @param  string[]  $words
     * @return array<int, array{type: string, attributes: array<string,string>}>
     */
    public function sendSentence(array $words): array
    {
        if ($this->socket === null) {
            throw new RuntimeException('Non connecté au MikroTik.');
        }

        foreach ($words as $word) {
            $this->writeWord($word);
        }
        // Fin de phrase = mot vide
        $this->writeWord('');

        return $this->readReply();
    }

    // -----------------------------------------------------------------
    // Encodage / décodage
    // -----------------------------------------------------------------

    private function writeWord(string $word): void
    {
        fwrite($this->socket, $this->encodeLength(strlen($word)) . $word);
    }

    private function encodeLength(int $length): string
    {
        if ($length < 0x80) {
            return chr($length);
        }
        if ($length < 0x4000) {
            $length |= 0x8000;
            return chr(($length >> 8) & 0xFF) . chr($length & 0xFF);
        }
        if ($length < 0x200000) {
            $length |= 0xC00000;
            return chr(($length >> 16) & 0xFF)
                 . chr(($length >> 8) & 0xFF)
                 . chr($length & 0xFF);
        }
        if ($length < 0x10000000) {
            $length |= 0xE0000000;
            return chr(($length >> 24) & 0xFF)
                 . chr(($length >> 16) & 0xFF)
                 . chr(($length >> 8) & 0xFF)
                 . chr($length & 0xFF);
        }
        return chr(0xF0)
             . chr(($length >> 24) & 0xFF)
             . chr(($length >> 16) & 0xFF)
             . chr(($length >> 8) & 0xFF)
             . chr($length & 0xFF);
    }

    private function readLength(): int
    {
        $byte = ord($this->readBytes(1));

        if (($byte & 0x80) === 0x00) {
            return $byte;
        }
        if (($byte & 0xC0) === 0x80) {
            $b2 = ord($this->readBytes(1));
            return (($byte & 0x3F) << 8) | $b2;
        }
        if (($byte & 0xE0) === 0xC0) {
            $rest = $this->readBytes(2);
            return (($byte & 0x1F) << 16) | unpack('n', $rest)[1];
        }
        if (($byte & 0xF0) === 0xE0) {
            $rest = $this->readBytes(3);
            return (($byte & 0x0F) << 24) | unpack('N', "\x00" . $rest)[1];
        }
        if (($byte & 0xF8) === 0xF0) {
            return unpack('N', $this->readBytes(4))[1];
        }

        throw new RuntimeException(
            'Longueur MikroTik invalide : 0x' . dechex($byte)
        );
    }

    private function readBytes(int $n): string
    {
        $buffer = '';
        while (strlen($buffer) < $n) {
            $chunk = fread($this->socket, $n - strlen($buffer));

            if ($chunk === false || $chunk === '') {
                $info = stream_get_meta_data($this->socket);
                if (!empty($info['timed_out'])) {
                    throw new RuntimeException('Timeout de lecture MikroTik.');
                }
                if (feof($this->socket)) {
                    throw new RuntimeException('Connexion MikroTik fermée.');
                }
                throw new RuntimeException('Lecture MikroTik échouée.');
            }
            $buffer .= $chunk;
        }
        return $buffer;
    }

    /**
     * Lit une réponse complète jusqu'à !done ou !fatal.
     */
    private function readReply(): array
    {
        $reply = [];

        while (true) {
            $sentence = $this->readSentence();

            if (empty($sentence)) {
                continue;
            }

            $type       = array_shift($sentence);
            $attributes = [];

            foreach ($sentence as $word) {
                if ($word === '' || $word[0] !== '=') {
                    continue;
                }
                $rest  = substr($word, 1);
                $eqPos = strpos($rest, '=');
                if ($eqPos === false) {
                    $attributes[$rest] = '';
                } else {
                    $attributes[substr($rest, 0, $eqPos)] = substr($rest, $eqPos + 1);
                }
            }

            $reply[] = [
                'type'       => $type,
                'attributes' => $attributes,
            ];

            if ($type === '!done' || $type === '!fatal') {
                return $reply;
            }
        }
    }

    /**
     * @return string[]
     */
    private function readSentence(): array
    {
        $words = [];
        while (true) {
            $length = $this->readLength();
            if ($length === 0) {
                return $words;
            }
            $words[] = $this->readBytes($length);
        }
    }
}

                this.selected = this.selected.filter(id => ids.includes(id));
            } catch (e) {}
        },

        mettreAJourVouchers(nouveaux) {
            const idsNouveaux = new Set(nouveaux.map(v => v.id));

            // Retirer ceux qui ne sont plus dans la réponse (utilisés depuis)
            this.vouchersDisponibles.forEach(v => {
                if (!idsNouveaux.has(v.id) && !v.transition) {
                    v.transition = true;
                    setTimeout(() => {
                        this.vouchersDisponibles = this.vouchersDisponibles.filter(x => x.id !== v.id);
                        // Ajouter les nouveaux qui ne sont pas encore là
                        nouveaux.forEach(nv => {
                            if (!this.vouchersDisponibles.find(x => x.id === nv.id)) {
                                this.vouchersDisponibles.push({...nv, masque: true, transition: false});
                            }
                        });
                    }, 2000);
                }
            });

            // Ajouter les nouveaux immédiatement
            nouveaux.forEach(nv => {
                if (!this.vouchersDisponibles.find(x => x.id === nv.id)) {
                    this.vouchersDisponibles.push({...nv, masque: true, transition: false});
                }
            });
        },

        verifierAlarmes() {
            this.lignes.forEach(l => {
                if (!l.alarme.active || !l.alarme.fin) {
                    if (this.fenetresAlarme.find(f => f.id === l.id)) {
                        this.fermerFenetre(l.id);
                    }
                    return;
                }
                const fin = new Date(l.alarme.fin).getTime();
                const dejaDansListe = this.fenetresAlarme.find(f => f.id === l.id);

                if (this.now >= fin && !dejaDansListe && !this._alarmesFermees.has(l.id)) {
                    this.ouvrirFenetreAlarme(l);
                }
                if (this.now < fin && dejaDansListe) {
                    this.fermerFenetre(l.id);
                }
            });

            this.fenetresAlarme.forEach(f => {
                const l = this.lignes.find(x => x.id === f.id);
                if (l && l.alarme.fin) {
                    const fin = new Date(l.alarme.fin).getTime();
                    f.depassementSecondes = Math.max(0, Math.floor((this.now - fin) / 1000));
                    if (l.tarif) {
                        f.depassementMontant = Math.floor((f.depassementSecondes / 60) * l.tarif.montant_par_minute);
                    }
                }
            });

            if (this.fenetresAlarme.length > 0 && this.alarme.enabled) {
                if (this.now - this._lastBeep > 2500) {
                    this._lastBeep = this.now;
                    this.jouerSon();
                }
            }
        },

        ouvrirFenetreAlarme(l) {
            this.fenetresAlarme.push({
                id: l.id,
                appareil: l.appareil,
                voucher: l.voucher?.username || null,
                forfait: l.alarme.type === 'duree'
                    ? this.formatDuree(l.alarme.minutes)
                    : this.formatAr(l.alarme.montant),
                heureFin: this.heureExacte(l.alarme.fin),
                depassementSecondes: 0,
                depassementMontant: 0,
                prolongationsRapides: [
                    { label: '+15 min', type: 'duree', h: 0, m: 15 },
                    { label: '+30 min', type: 'duree', h: 0, m: 30 },
                    { label: '+500 Ar', type: 'montant', montant: 500 },
                    { label: '+1000 Ar', type: 'montant', montant: 1000 },
                ],
            });

            this.marquerDeclenchee(l.id);
            if (l.alarme) l.alarme.declenchee = true;

            if (typeof Notification !== 'undefined' && Notification.permission === 'granted') {
                try {
                    new Notification('Cyber Manager — Alarme', {
                        body: `${l.appareil} a atteint son forfait.`,
                        icon: '/favicon.ico',
                        tag: 'wifi-alarm-' + l.id,
                        requireInteraction: true,
                    });
                } catch (e) {}
            }
            this.clignoterTitre(true);
        },

        fermerFenetre(id) {
            this.fenetresAlarme = this.fenetresAlarme.filter(f => f.id !== id);
            if (this.fenetresAlarme.length === 0) {

                this.clignoterTitre(false);
            }

        majApercu() {
            const l = this.alarmeModal.ligne;
            if (!l) return;
            const secDeja = this.secondesConsommees(l);
            const minutesDeja = Math.floor(secDeja / 60);
            let minutesCible = 0;
            if (this.alarmeModal.type === 'duree') {
                minutesCible = this.alarmeModal.h * 60 + this.alarmeModal.m;
            } else {
                minutesCible = l.tarif
                    ? Math.floor(this.alarmeModal.montant / l.tarif.montant_par_minute)
                    : Math.floor(this.alarmeModal.montant / 20);
            }
            const minutesRestantes = this.alarmeModal.depart === 'debut'
                ? minutesCible - minutesDeja
                : minutesCible;
            if (minutesRestantes <= 0) {
                this.alarmeModal.apercu = "Cette limite est déjà dépassée.";
                return;
            }
            const fin = new Date(this.now + minutesRestantes * 60000);
            const heureFin = String(fin.getHours()).padStart(2,'0') + ':' +
                             String(fin.getMinutes()).padStart(2,'0');
            this.alarmeModal.apercu = `Déjà consommé : ${this.formatDureeSec(secDeja)} (${this.formatAr(this.montantTempsReel(l))}). L'alarme sonnera à ${heureFin}, dans ${this.formatDureeSec(minutesRestantes * 60)}.`;
        },

            this.reglages.enCours = true;
            try {
                let dureeSec = 0;
                if (this.reglages.form.modeDuree === 'manuel') {
                    dureeSec = (this.reglages.form.dureeMin || 0) * 60 + (this.reglages.form.dureeSec || 0);
                    if (dureeSec < 10) dureeSec = 10;
                    if (dureeSec > 600) dureeSec = 600;
                }
                let tailleMo = parseInt(this.reglages.form.tailleMax ?? 10, 10);
                if (isNaN(tailleMo) || tailleMo < 1) tailleMo = 1;
                if (tailleMo > 10) tailleMo = 10;
                let volume = parseInt(this.reglages.form.volume ?? 70, 10);
                if (isNaN(volume) || volume < 0) volume = 0;
                if (volume > 100) volume = 100;

                const r1 = await window.cmFetch(cfg.urlReglagesSave, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        enabled: this.reglages.form.enabled ? 1 : 0,
                        volume: volume,
                        duree_secondes: dureeSec,
                        taille_max_mo: tailleMo,
                    }),
                });
                if (!r1.ok) {
                    window.dispatchEvent(new CustomEvent('toast', { detail: 'Erreur serveur (réglages).' }));
                    this.reglages.enCours = false;
                    return;
                }

                // Choix du son actif
                if (this.reglages.form.sonUrl && this.reglages.form.sonUrl.startsWith('/sounds/')) {
                    const nom = this.reglages.form.sonUrl.split('/').pop();
                    await window.cmFetch(cfg.urlChoisirSon, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ nom: nom }),
                    });
                } else if (this.reglages.form.sonId) {
                    await window.cmFetch(cfg.urlChoisirSon, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify({ nom: '__default__' }),
                    }).catch(() => {});
                }

                this.alarme.enabled = this.reglages.form.enabled;
                this.alarme.volume = volume;
                this.alarme.son_url = this.reglages.form.sonUrl;
                this.alarme.son_nom = this.reglages.form.sonNom;

                    this.$refs.audio.src = this.alarme.son_url;
                }

                window.dispatchEvent(new CustomEvent('toast', { detail: 'Réglages enregistrés.' }));
                this.reglages.ouvert = false;
                await this.chargerReglages();
            } catch (e) {
                window.dispatchEvent(new CustomEvent('toast', { detail: 'Erreur réseau.' }));
            }
            this.reglages.enCours = false;

        async envoyerSon(file) {
            this.reglages.uploadErreur = '';
            this.reglages.uploadOk = '';
            this.reglages.uploadEnCours = true;

            const tailleMaxMo = this.reglages.form.tailleMax ?? 10;
            const tailleMaxOctets = tailleMaxMo * 1024 * 1024;

            if (file.size > tailleMaxOctets) {
                this.reglages.uploadErreur = `Fichier trop volumineux (max ${tailleMaxMo} Mo).`;
                this.reglages.uploadEnCours = false;
                return;
            }
            const ext = file.name.split('.').pop().toLowerCase();
            if (!['mp3','wav','ogg','m4a','aac'].includes(ext)) {
                this.reglages.uploadErreur = 'Format non accepté (MP3, WAV, OGG, M4A, AAC).';
                this.reglages.uploadEnCours = false;
                return;
            }

            const fd = new FormData();
            fd.append('son', file);

            try {
                const r = await fetch(cfg.urlUploaderSon, {
                    method: 'POST',
                    headers: { 'X-CSRF-TOKEN': cfg.csrf, 'Accept': 'application/json' },
                    body: fd,
                });
                const d = await r.json();
                if (!r.ok) {
                    this.reglages.uploadErreur = d.message || (d.errors?.son?.[0]) || 'Erreur d\'upload.';
                    this.reglages.uploadEnCours = false;
                    return;
                }
                this.reglages.sonsUploades.push({ nom: d.nom, url: d.url });
                this.reglages.uploadOk = 'Fichier ajouté. Choisissez-le dans la liste pour l\'utiliser.';
            } catch (e) {
                this.reglages.uploadErreur = 'Erreur réseau.';
            }
            this.reglages.uploadEnCours = false;
        },

        demanderSupprSon(s) {
            this.supprSonModal = { ouvert: true, nom: s.nom, son: s };
        },

        async confirmerSupprSon() {
            const s = this.supprSonModal.son;
            if (!s) return;
            try {
                await window.cmFetch(cfg.urlSupprimerSon, {
                    method: 'DELETE',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ nom: s.nom }),
                });
                this.reglages.sonsUploades = this.reglages.sonsUploades.filter(x => x.nom !== s.nom);
                if (this.reglages.form.sonUrl === s.url) {
                    this.reglages.form.sonId = 'bip-classique';
                    this.reglages.form.sonUrl = null;
                    this.reglages.form.sonNom = 'Bip par défaut';
                }
                this.supprSonModal.ouvert = false;
            } catch (e) {}
        },

        // Son
        // -----------------------------------------------------------------
        jouerSon() {
            if (this.alarme.son_url) {
                const a = this.$refs.audio;
                if (!a) return;
                try {
                    a.src = this.alarme.son_url;
                    a.volume = Math.max(0, Math.min(1, (this.alarme.volume ?? 70) / 100));
                    a.currentTime = 0;
                    a.play().catch(() => {});
                } catch (e) {}
                return;
            }
            this.bipParDefaut('bip-classique', this.alarme.volume ?? 70);
        },

        arreterSon() {
            try {
                if (this.$refs.audio) {
                    this.$refs.audio.pause();
                    this.$refs.audio.currentTime = 0;
                }
                if (this._ctxAudio) {
                    this._ctxAudio.close().catch(() => {});
                    this._ctxAudio = null;
                }
                this.testEnCours = false;
                this.testId = null;
                clearTimeout(this._audioTimeout);
            } catch (e) {}
        },

        bipParDefaut(id = 'bip-classique', volumePct = 70) {
            try {
                const Ctx = window.AudioContext || window.webkitAudioContext;
                if (!Ctx) return;
                if (this._ctxAudio) this._ctxAudio.close().catch(() => {});
                const ctx = new Ctx();
                this._ctxAudio = ctx;
                const vol = Math.max(0, Math.min(1, volumePct / 100));
                const gain = ctx.createGain();
                gain.gain.value = vol * 0.35;
                gain.connect(ctx.destination);
                const sequences = {
                    'bip-classique': [880, 880, 880],
                    'carillon': [523, 659, 784, 1047],
                    'urgent': [1200, 1200, 1200, 1200, 1200],
                };
                const freqs = sequences[id] || sequences['bip-classique'];
                freqs.forEach((f, i) => {
                    const osc = ctx.createOscillator();
                    osc.type = id === 'urgent' ? 'sawtooth' : 'sine';
                    osc.frequency.value = f;
                    osc.connect(gain);
                    const start = ctx.currentTime + i * 0.28;
                    osc.start(start);
                    osc.stop(start + 0.20);
                });
            } catch (e) {}
        },

        testerSon(idOrUrl) {
            if (this.testEnCours && this.testId === idOrUrl) {
                this.arreterSon();
                return;
            }
            this.arreterSon();
            this.testEnCours = true;
            this.testId = idOrUrl;
            const volumePct = this.reglages.form.volume ?? 70;
            let source = idOrUrl;
            if (idOrUrl === 'actif') {
                source = this.reglages.form.sonUrl || this.reglages.form.sonId || 'bip-classique';
            }
            if (typeof source === 'string' && source.startsWith('/')) {
                const a = this.$refs.audio;
                if (a) {
                    a.src = source;
                    a.volume = Math.max(0, Math.min(1, volumePct / 100));
                    a.currentTime = 0;
                    a.play().catch(() => {});
                }
            } else {
                this.bipParDefaut(source, volumePct);
            }
            this._audioTimeout = setTimeout(() => this.arreterSon(), 6000);
        },

                if (!Ctx) return;
                const ctx = new Ctx();
                ctx.resume().then(() => ctx.close()).catch(() => {});

        // -----------------------------------------------------------------

        // -----------------------------------------------------------------
        formatAr(n) { return Number(n || 0).toLocaleString('fr-FR') + ' Ar'; },
        formatDuree(m) {
            m = parseInt(m, 10) || 0;
            if (m < 60) return m + ' min';
            const h = Math.floor(m / 60), r = m % 60;
            return r === 0 ? h + ' h' : h + ' h ' + String(r).padStart(2, '0') + ' min';
        },
        formatDureeSec(sec) {
            sec = Math.max(0, Math.floor(sec));
            const h = Math.floor(sec / 3600);
            const m = Math.floor((sec % 3600) / 60);
            const s = sec % 60;
            const pad = n => String(n).padStart(2, '0');
            return h > 0 ? `${pad(h)}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`;
        },
        formatOctets(o) {
            o = parseInt(o, 10) || 0;
            if (o < 1024) return o + ' o';
            if (o < 1048576) return (o / 1024).toFixed(1) + ' Ko';
            if (o < 1073741824) return (o / 1048576).toFixed(1) + ' Mo';
            return (o / 1073741824).toFixed(2) + ' Go';
        },
        formatDebit(mbps) {
            if (!mbps || mbps < 0.001) return '0 Ko/s';
            if (mbps < 1) return (mbps * 125).toFixed(1) + ' Ko/s';
            return mbps.toFixed(2) + ' Mo/s';
        },
        heureExacte(iso) {
            if (!iso) return '—';
            const d = new Date(iso);
            return String(d.getHours()).padStart(2,'0') + ':' +
                   String(d.getMinutes()).padStart(2,'0') + ':' +
                   String(d.getSeconds()).padStart(2,'0');
        },
        dateLongueFr(iso) {
            if (!iso) return '—';
            const d = new Date(iso);
            const mois = ['janvier','février','mars','avril','mai','juin','juillet','août','septembre','octobre','novembre','décembre'];
            return String(d.getDate()).padStart(2,'0') + ' ' + mois[d.getMonth()] + ' ' + d.getFullYear();
        },

        secondesConsommees(l) {
            if (!l.date_debut) return 0;
            const debut = new Date(l.date_debut).getTime();
            let fin;
            if (l.etat_appareil === 'en_pause' && l.pause_at) {
                fin = new Date(l.pause_at).getTime();
            } else if (l.etat_appareil === 'deconnecte' && l.deconnecte_at) {
                fin = new Date(l.deconnecte_at).getTime();
            } else if (l.etat_appareil === 'termine') {
                fin = l.date_fin_reelle ? new Date(l.date_fin_reelle).getTime() : this.now;
            } else {
                fin = this.now;
            }
            let total = Math.floor((fin - debut) / 1000);
            total -= (l.secondes_hors_ligne || 0);
            total -= (l.secondes_pause || 0);
            return Math.max(0, total);
        },

        montantTempsReel(l) {
            if (!l.tarif) return 0;
            const sec = this.secondesConsommees(l);
            return Math.floor((sec / 60) * l.tarif.montant_par_minute);
        },

        montantFinalCalcule(l) {
            if (!l.tarif) return 0;
            const sec = this.secondesConsommees(l);
            const minutes = Math.ceil(sec / 60);
            let brut = minutes * l.tarif.montant_par_minute;
            if (brut < l.tarif.montant_minimum) brut = l.tarif.montant_minimum;
            if (l.tarif.arrondi_actif && l.tarif.unite_arrondi > 0) {
                const u = l.tarif.unite_arrondi, s = l.tarif.seuil_arrondi, r = brut % u;
                if (r !== 0) brut = r < s ? brut - r : brut + (u - r);
            }
            return brut;
        },

        chronoRestantTexte(l) {
            if (!l.alarme.fin) return '00:00';
            const fin = new Date(l.alarme.fin).getTime();
            const sec = Math.max(0, Math.floor((fin - this.now) / 1000));
            return this.formatDureeSec(sec);
        },

        progressionAlarme(l) {
            if (!l.alarme.minutes || !l.date_debut) return 0;
            const secDeja = this.secondesConsommees(l);
            const secTotal = l.alarme.minutes * 60;
            return Math.max(0, Math.min(100, (secDeja / secTotal) * 100));
        },

        debitLigne(l, sens) {
            const key = l.id + '-' + sens;
            const volume = sens === 'entree' ? l.volume_entree : l.volume_sortie;
            const now = Date.now();
            const c = this.volumeCache[key];
            if (!c) { this.volumeCache[key] = { vol: volume, ts: now, mbps: 0 }; return '0 Ko/s'; }
            const dOctets = volume - c.vol;
            const dSec = (now - c.ts) / 1000;
            if (dSec >= 1 && dOctets >= 0) {
                c.mbps = (dOctets * 8) / (dSec * 1_000_000);
                c.vol = volume;
                c.ts = now;
            }
            return this.formatDebit(c.mbps);
        },
        get debitTotalTotal() {
            return Object.values(this.volumeCache).reduce((a, c) => a + (c.mbps || 0), 0);
        },

        libelleEtat(l) {

                'bg-amber-500': l.etat_appareil === 'deconnecte',

                'bg-slate-400': l.etat_appareil === 'termine',
            };
        },
        classeLigne(l) {
            if (l.alarme.declenchee && !l.alarme.acquittee) return 'bg-rose-50 border-l-4 border-rose-500';
            return {
                'bg-emerald-50/60 border-l-4 border-emerald-500': l.etat_appareil === 'connecte',

                'bg-slate-100 border-l-4 border-slate-400': l.etat_appareil === 'termine',

        get lignesAffichees() {

        basculerTous() {

        escapeHtml(s) {
            if (s == null) return '';

        },
    };

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class Session extends Model
{
    protected $table = 'cyber_sessions';

    public const WIFI_CONNECTE   = 'connecte';
    public const WIFI_DECONNECTE = 'deconnecte';
    public const WIFI_TERMINE    = 'termine';
    public const WIFI_EN_PAUSE   = 'en_pause';

    protected $fillable = [
        'poste_id', 'type_session',
        'date_heure_debut', 'date_heure_fin_prevue', 'date_heure_fin_reelle',
        'date_heure_suspension', 'date_heure_reprise', 'duree_suspension',
        'duree_prevue', 'duree_consommee', 'temps_restant',
        'montant_initial', 'montant_recharge', 'montant_total',
        'montant_consomme', 'montant_restant', 'montant_a_reverser',
        'description', 'etat', 'motif_fin', 'tarif_id', 'voucher_id',
        'volume_entree', 'volume_sortie', 'volume_total',
        'mikrotik_active_id', 'hotspot_username', 'client_mac', 'client_ip',
        'ssid', 'appareil_wifi_id', 'derniere_sync_at',
        'etat_wifi', 'deconnecte_at', 'reconnecte_at', 'duree_hors_ligne_secondes',
        'pause_at', 'duree_pause_secondes',
        'alarme_active', 'alarme_type', 'alarme_duree_minutes', 'alarme_montant',
        'alarme_fin_prevue', 'alarme_declenchee_at', 'alarme_message',
        'alarme_historique',
    ];

    protected $casts = [
        'date_heure_debut' => 'datetime',
        'date_heure_fin_prevue' => 'datetime',
        'date_heure_fin_reelle' => 'datetime',
        'date_heure_suspension' => 'datetime',
        'date_heure_reprise' => 'datetime',
        'derniere_sync_at' => 'datetime',
        'deconnecte_at' => 'datetime',
        'reconnecte_at' => 'datetime',
        'pause_at' => 'datetime',
        'alarme_fin_prevue' => 'datetime',
        'alarme_declenchee_at' => 'datetime',
        'alarme_active' => 'boolean',
        'duree_hors_ligne_secondes' => 'integer',
        'duree_pause_secondes' => 'integer',
        'alarme_duree_minutes' => 'integer',
        'alarme_montant' => 'integer',
        'alarme_historique' => 'array',
    ];

    // Relations
    // -----------------------------------------------------------------

    public function poste(): BelongsTo { return $this->belongsTo(Poste::class); }
    public function recharges(): HasMany { return $this->hasMany(Recharge::class); }
    public function restitutions(): HasMany { return $this->hasMany(Restitution::class); }
    public function tarif(): BelongsTo { return $this->belongsTo(Tarif::class); }
    public function voucher(): BelongsTo { return $this->belongsTo(Voucher::class); }
    public function appareilWifi(): BelongsTo { return $this->belongsTo(AppareilWifi::class, 'appareil_wifi_id'); }

    // -----------------------------------------------------------------
    // Helpers
    // -----------------------------------------------------------------

    public function estConnecte(): bool
    {
        return $this->etat_wifi === self::WIFI_CONNECTE;
    }

    public function estDeconnecte(): bool
    {
        return $this->etat_wifi === self::WIFI_DECONNECTE;
    }

    public function estEnPause(): bool
    {
        return $this->etat_wifi === self::WIFI_EN_PAUSE;
    }

    public function estTermine(): bool
    {
        return $this->etat_wifi === self::WIFI_TERMINE;
    }

    /**
     * Vrai si le compteur doit tourner maintenant.
     */
    public function compteurActif(): bool
    {
        return $this->etat_wifi === self::WIFI_CONNECTE;
    }

    /**
     * Secondes consommées au moment présent.
     *
     * Décrémente du temps de :
     *   - duree_hors_ligne_secondes (déconnexions)
     *   - duree_pause_secondes (mises en pause manuelles)
     *   - du temps depuis pause_at si en pause actuellement
     */
    public function secondesConsommees(): int
    {
        if (!$this->date_heure_debut) return 0;

        // Terminé = figé
        if ($this->estTermine()) {
            return max(0, (int) $this->duree_consommee * 60);
        }

        // Fin virtuelle : maintenant, ou pause_at, ou deconnecte_at
        if ($this->estEnPause() && $this->pause_at) {
            $fin = $this->pause_at;
        } elseif ($this->estDeconnecte() && $this->deconnecte_at) {
            $fin = $this->deconnecte_at;
        } else {
            $fin = now();
        }

        $total = $this->date_heure_debut->diffInSeconds($fin);
        $total -= (int) $this->duree_hors_ligne_secondes;
        $total -= (int) $this->duree_pause_secondes;

        return max(0, $total);
    }
}

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\QueryException;
use Illuminate\Support\Arr;
use InvalidArgumentException;

class AppareilWifi extends Model
{
    protected $table = 'appareils_wifi';

    public const TYPES = ['telephone', 'ordinateur', 'tablette', 'inconnu'];

    protected $fillable = [
        'adresse_mac',
        'adresse_ip',
        'host_name',
        'nom_affichage',   // ← AJOUTÉ : nom personnalisé choisi par le personnel
        'type_appareil',
        'fabricant',
        'modele',
        'systeme',
        'user_agent',
        'premiere_connexion',
        'derniere_connexion',
    ];

    protected $casts = [
        'premiere_connexion' => 'datetime',
        'derniere_connexion' => 'datetime',
    ];

    public function sessions(): HasMany
    {
        return $this->hasMany(Session::class, 'appareil_wifi_id');
    }

    /**
     * Normalise une MAC au format AA:BB:CC:DD:EE:FF.
     * Retourne null si la valeur n'est pas une MAC valide (12 hexa).
     */
    public static function normaliserMac(?string $mac): ?string
    {
        if ($mac === null) {
            return null;
        }

        $hex = strtoupper(preg_replace('/[^0-9A-Fa-f]/', '', $mac));

        if (strlen($hex) !== 12) {
            return null;
        }

        return implode(':', str_split($hex, 2));
    }

    /**
     * Retrouve l'appareil par sa MAC, ou le crée.
     *
     * - Les attributs vides/null ne PEUVENT PAS écraser une valeur existante.
     * - Un type "inconnu" n'écrase jamais un type déjà connu.
     * - Ne touche pas derniere_connexion sur un appareil existant
     *   (voir toucherDerniereConnexion()).
     *
     * @param  array<string,mixed>  $attrs  adresse_ip, host_name, type_appareil,
     *                                      fabricant, modele, systeme, user_agent
     */
    public static function trouverOuCreerParMac(string $mac, array $attrs = []): self
    {
        $macNormalisee = self::normaliserMac($mac);

        if ($macNormalisee === null) {
            throw new InvalidArgumentException("Adresse MAC invalide : {$mac}");
        }

        $attrs = Arr::only($attrs, [
            'adresse_ip', 'host_name', 'type_appareil',
            'fabricant', 'modele', 'systeme', 'user_agent',
            'nom_affichage',   // ← AJOUTÉ : on autorise aussi ce champ à être passé
        ]);

        $attrs = array_filter($attrs, fn ($v) => $v !== null && $v !== '');

        if (
            isset($attrs['type_appareil'])
            && (
                $attrs['type_appareil'] === 'inconnu'
                || !in_array($attrs['type_appareil'], self::TYPES, true)
            )
        ) {
            unset($attrs['type_appareil']);
        }

        $appareil = self::where('adresse_mac', $macNormalisee)->first();

        if ($appareil === null) {
            try {
                return self::create(array_merge(
                    ['type_appareil' => 'inconnu'],
                    $attrs,
                    [
                        'adresse_mac'        => $macNormalisee,
                        'premiere_connexion' => now(),
                        'derniere_connexion' => now(),
                    ]
                ));
            } catch (QueryException $e) {
                // Course critique : un autre process vient de le créer
                $appareil = self::where('adresse_mac', $macNormalisee)->first();

                if ($appareil === null) {
                    throw $e;
                }
            }
        }

        $appareil->fill($attrs);

        if ($appareil->isDirty()) {
            $appareil->save();
        }

        return $appareil;
    }

    /**
     * Met à jour la date de dernière connexion (et la première si absente).
     */
    public function toucherDerniereConnexion(): void
    {
        $maintenant = now();

        $this->derniere_connexion = $maintenant;

        if ($this->premiere_connexion === null) {
            $this->premiere_connexion = $maintenant;
        }

        $this->save();
    }

    /**
     * Nom affiché dans l'interface, par ordre de priorité :
     *  1. nom_affichage : choisi manuellement par le personnel (fiable à 100%) ;
     *  2. host_name : gardé seulement s'il semble parlant (on écarte les
     *     identifiants génériques type "android-1339b7ea..." qui ne disent rien) ;
     *  3. repli générique + 4 derniers caractères de la MAC, en attendant
     *     qu'on le renomme depuis le tableau.
     */
    public function libelle(): string
    {
        if (!empty($this->nom_affichage)) {
            return $this->nom_affichage;
        }

        $h = $this->host_name;
        $generique = $h !== null && preg_match('/^(android|localhost|iphone|ipad|unknown)[-_]?[0-9a-f]{4,}$/i', $h);

        if ($h !== null && $h !== '' && !$generique && mb_strlen($h) >= 3) {
            return str_replace(['-', '_'], ' ', $h);
        }

        $libelleType = match ($this->type_appareil) {
            'telephone'  => 'Téléphone',
            'tablette'   => 'Tablette',
            'ordinateur' => 'Ordinateur',
            default      => 'Appareil',
        };

        $suffixe = $this->adresse_mac ? substr(str_replace(':', '', $this->adresse_mac), -4) : '????';

        return "{$libelleType} · {$suffixe}";
    }
}

        if ($reutilisable === 'oui') {

        if ($protege === 'oui') {
            $query->where('protege', true);

            $query->where('protege', false);
        }

        $sync = $f['sync'] ?? null;
        if ($sync === 'synced') {
            $query->whereNotNull('mikrotik_synced_at');
        } elseif ($sync === 'not_synced') {
            $query->whereNull('mikrotik_synced_at');
        }

        $lot = $f['lot'] ?? null;
        if ($lot !== null && $lot !== '') {
            $query->where('lot_voucher_id', (int) $lot);

    public function volumeLisible(): string
    {
        $octets = (int) ($this->volume_total_octets ?? 0);

        if ($octets <= 0) {
            return '—';
        }

        $unites = ['o', 'Ko', 'Mo', 'Go'];
        $i = 0;
        $valeur = $octets;

        while ($valeur >= 1024 && $i < count($unites) - 1) {
            $valeur /= 1024;
            $i++;
        }

        return ($i === 0 ? (int) $valeur : round($valeur, 1)) . ' ' . $unites[$i];

                : $this->appareil_wifi_id !== $appareil->id;

            if ($rattacher) {
                $maj['appareil_wifi_id'] = $appareil->id;
            }
        }

        $this->update($maj);

        session(['hotspot.lot_genere' => $lot->id]);

    // Génération en masse (façon Mikhmon : quantité + paramètres techniques)
    // -----------------------------------------------------------------

    /**
     * @param array{
     *   quantite:int, mode:string, longueur:int, prefixe?:string, jeu:string,
     *   serveur?:string, profil?:string, duree?:int, limite_data_mo?:int, commentaire?:string
     * } $p
     */
    public function genererEnMasse(array $p): LotVoucher
    {
        $quantite = (int) $p['quantite'];
        if ($quantite < 1 || $quantite > 200) {
            throw new InvalidArgumentException('La quantité doit être comprise entre 1 et 200.');
        }

        $mode     = $p['mode'] ?? 'distinct';
        $longueur = max(3, min(8, (int) ($p['longueur'] ?? 4)));
        $prefixe  = preg_replace('/\s+/', '', trim((string) ($p['prefixe'] ?? '')));
        $jeu      = $this->jeuCaracteres($p['jeu'] ?? 'minuscules');
        $serveur  = trim((string) ($p['serveur'] ?? '')) ?: null;
        $profil   = trim((string) ($p['profil'] ?? '')) ?: null;
        $duree    = (int) ($p['duree'] ?? 0);
        $dataMo   = isset($p['limite_data_mo']) && $p['limite_data_mo'] !== '' ? (int) $p['limite_data_mo'] : null;
        $commentaire = trim((string) ($p['commentaire'] ?? '')) ?: null;

        $lot = DB::transaction(function () use (
            $quantite, $mode, $longueur, $prefixe, $jeu, $serveur, $profil, $duree, $dataMo, $commentaire
        ) {
            $lot = LotVoucher::create([
                'nom'              => 'Génération du ' . now()->format('d/m/Y H:i'),
                'quantite_prevue'  => $quantite,
                'montant_unitaire' => 0,
                'duree_unitaire'   => $duree,
                'quantite_generee' => 0,
                'source'           => 'cyber_manager',
            ]);

            for ($i = 0; $i < $quantite; $i++) {
                do {
                    $username = $prefixe . $this->genererChaine($longueur, $jeu);
                } while (Voucher::where('username', $username)->exists());

                $password = $mode === 'identique' ? $username : $this->genererChaine($longueur, $jeu);

                Voucher::create([
                    'lot_voucher_id'      => $lot->id,
                    'username'            => $username,
                    'password'            => $password,
                    'code'                => $username,
                    'montant'             => 0,
                    'duree'               => $duree,
                    'etat'                => 'disponible',
                    'source'              => 'cyber_manager',
                    'type_voucher'        => Voucher::TYPE_TEMPORAIRE,
                    'protege'             => false,
                    'serveur'             => $serveur,
                    'profil'              => $profil,
                    'limite_data_mo'      => $dataMo,
                    'description'         => $commentaire,
                    'date_heure_creation' => now(),
                ]);
            }

            $lot->update(['quantite_generee' => $quantite]);

            return $lot->fresh();
        });

        foreach (Voucher::where('lot_voucher_id', $lot->id)->get() as $voucher) {
            $this->synchroniserVersMikroTik($voucher);
        }

        return $lot;
    }

    private function jeuCaracteres(string $cle): string
    {
        return match ($cle) {
            'majuscules'          => 'ABCDEFGHIJKLMNOPQRSTUVWXYZ',
            'mixte'                => 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ',
            'minuscules_chiffres'  => 'abcdefghijklmnopqrstuvwxyz0123456789',
            'majuscules_chiffres'  => 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789',
            'mixte_chiffres'       => 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789',
            default                => 'abcdefghijklmnopqrstuvwxyz',
        };

    private function genererChaine(int $longueur, string $car): string
    {
        $chaine = '';
        for ($i = 0; $i < $longueur; $i++) {

        }
        return $chaine;
    }

    public function synchroniserVersMikroTik(Voucher $voucher): bool
    {
        try {
            $existing = $this->mikroTik->findHotspotUser($voucher->username);

            if ($existing === null) {
                $limiteUptime = (int) $voucher->duree > 0
                    ? MikroTikService::formatUptime((int) $voucher->duree)
                    : null;

                $limiteData = (int) $voucher->limite_data_mo > 0
                    ? (string) ((int) $voucher->limite_data_mo * 1024 * 1024)
                    : null;

                $this->mikroTik->createHotspotUser(
                    username:        $voucher->username,
                    password:        $voucher->password,
                    profile:         $voucher->profil ?: config('mikrotik.hotspot_profile', 'default'),
                    comment:         $voucher->description ?: (($voucher->estReutilisable() ? 'CM réutilisable #' : 'CM voucher #') . $voucher->id),
                    limitUptime:     $limiteUptime,
                    server:          $voucher->serveur,
                    limitBytesTotal: $limiteData,
                );

                $existing = $this->mikroTik->findHotspotUser($voucher->username);
            }

            $voucher->update([
                'mikrotik_id'         => $existing['.id'] ?? null,
                'mikrotik_synced_at'  => now(),
                'mikrotik_sync_error' => null,
            ]);

            return true;
        } catch (\Throwable $e) {
            Log::warning('Sync MikroTik échouée pour l’identifiant ' . $voucher->id, [
                'username' => $voucher->username,
                'error'    => $e->getMessage(),
            ]);

            $voucher->update(['mikrotik_sync_error' => $e->getMessage()]);

            return false;
        }
    }

    /** @return array{ok:int, ko:int, restants:int} */
    public function synchroniserVouchersEnAttente(?int $limite = null): array
    {
        $query = Voucher::whereNull('mikrotik_synced_at')->orderBy('id');

        if ($limite !== null && $limite > 0) {
            $query->limit($limite);
        }

        $ok = 0;
        $ko = 0;

        foreach ($query->get() as $voucher) {
            $this->synchroniserVersMikroTik($voucher) ? $ok++ : $ko++;
        }

        $restants = Voucher::whereNull('mikrotik_synced_at')->count();

        return ['ok' => $ok, 'ko' => $ko, 'restants' => $restants];
    }

    /**
     * @param  int[]  $ids
     * @param  bool   $inclureProteges  true après saisie du username (déprotection explicite)
     * @return array{supprimes:int, proteges_ignores:int, echecs:int}
     */
    public function supprimerEnMasse(array $ids, bool $inclureProteges = false): array
    {
        $ids = array_values(array_unique(array_map('intval', $ids)));
        $resultat = ['supprimes' => 0, 'proteges_ignores' => 0, 'echecs' => 0];

        foreach (Voucher::whereIn('id', $ids)->get() as $voucher) {
            if ($voucher->protege && !$inclureProteges) {
                $resultat['proteges_ignores']++;
                continue;
            }

            try {
                $this->supprimer($voucher);
                $resultat['supprimes']++;
            } catch (\Throwable $e) {
                Log::warning('Suppression en masse échouée', [
                    'voucher_id' => $voucher->id,
                    'error'      => $e->getMessage(),
                ]);
                $resultat['echecs']++;
            }
        }

        return $resultat;
    }

<?php

namespace App\Services;

use App\Models\Voucher;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Log;

/**
 * Maintient un stock d'identifiants usage unique disponibles (le "pool").
 * Ne compte, ne crée et ne touche que des identifiants usage unique non
 * protégés ; ne supprime jamais rien ; ne fait rien si MikroTik est injoignable.
 */
class VoucherReplenishmentService
{
    private const VERROU = 'vouchers:reapprovisionnement';

    public function __construct(
        private VoucherService $vouchers,
        private MikroTikService $mikroTik,
    ) {
    }

       public function verifierEtReapprovisionner(bool $simuler = false, bool $force = false): array
    {
        $verrou = null;

        if (!$simuler) {
            $verrou = Cache::lock(self::VERROU, 600);

            if (!$verrou->get()) {
                return $this->resultat('ignore', 'Un réapprovisionnement est déjà en cours.');
            }
        }

        try {
            return $this->executer($simuler, $force);
        } finally {
            $verrou?->release();
        }
    }

     private function executer(bool $simuler, bool $force = false): array
    {
        $mode = \App\Models\Parametre::get('hotspot_pool_mode', 'auto');

        if ($mode === 'manuel' && !$force) {
            return $this->resultat('ignore', 'Mode manuel actif : réapprovisionnement automatique désactivé.');
        }

        $seuil       = max(0, (int) config('mikrotik.voucher_pool_seuil', 10));
        $taille      = max(0, (int) config('mikrotik.voucher_pool_taille', 40));
        $disponibles = $this->compterDisponibles();

        if ($disponibles >= $seuil) {
            return $this->resultat('ok', "Pool suffisant : {$disponibles} disponible(s) pour un seuil de {$seuil}.");
        }

        $aGenerer = $taille - $disponibles;

        if ($aGenerer < 1) {
            return $this->resultat('ignore', 'Configuration incohérente : la taille du pool doit dépasser le seuil.');
        }

        if ($simuler) {
            return $this->resultat('simulation', "Générerait {$aGenerer} identifiant(s).", ['a_generer' => $aGenerer]);
        }

        try {
            $this->mikroTik->testConnection();
        } catch (\Throwable $e) {
            Log::warning('Réapprovisionnement reporté : MikroTik injoignable', ['error' => $e->getMessage()]);
            return $this->resultat('ignore', 'MikroTik injoignable, réapprovisionnement reporté.');
        }

        try {
            $lot = $this->vouchers->genererEnMasse([
                'quantite' => $aGenerer,
                'mode'     => 'distinct',
                'longueur' => 4,
                'jeu'      => 'minuscules',
            ]);
        } catch (\Throwable $e) {
            Log::error('Réapprovisionnement des identifiants échoué', ['error' => $e->getMessage()]);
            return $this->resultat('erreur', $e->getMessage());
        }

        $nonSync = Voucher::where('lot_voucher_id', $lot->id)->whereNull('mikrotik_synced_at')->count();

        $message = "{$lot->quantite_generee} identifiant(s) généré(s).";
        if ($nonSync > 0) {
            $message .= " {$nonSync} non synchronisé(s) avec MikroTik.";
        }

        return $this->resultat('genere', $message, [
            'generes'          => (int) $lot->quantite_generee,
            'lot_id'           => $lot->id,
            'non_synchronises' => $nonSync,
        ]);
    }
    
    private function compterDisponibles(): int
    {
        return Voucher::porteeTemporaires()->porteeDisponibles()->count();
    }

    private function resultat(string $statut, string $message, array $extra = []): array
    {
        return array_merge([
            'statut'           => $statut,
            'message'          => $message,
            'seuil'            => max(0, (int) config('mikrotik.voucher_pool_seuil', 10)),

<?php

namespace App\Services;

use App\Models\Tarif;
use InvalidArgumentException;

class TarifService
{
    // -----------------------------------------------------------------
    // Résolution de la configuration active
    // -----------------------------------------------------------------

    /** Y a-t-il un tarif unifié ou deux tarifs séparés ? */
    public function mode(): string
    {
        return Tarif::where('cible', Tarif::CIBLE_UNIFIE)->exists()
            ? 'unifie'
            : 'separe';
    }

    /** Tarif unique (mode unifié). Null si mode séparé. */
    public function tarifUnifie(): ?Tarif
    {
        return Tarif::where('cible', Tarif::CIBLE_UNIFIE)->first();
    }

    public function tarifEthernet(): ?Tarif
    {
        if ($u = $this->tarifUnifie()) {
            return $u;
        }

        return Tarif::where('cible', Tarif::CIBLE_ETHERNET)->first();
    }

    public function tarifWifi(): ?Tarif
    {
        if ($u = $this->tarifUnifie()) {
            return $u;
        }

        return Tarif::where('cible', Tarif::CIBLE_WIFI)->first();
    }

    /** Compatibilité avec l'ancien code (SessionService). */
    public function tarifStandard(): Tarif
    {
        return $this->tarifPour(Tarif::CIBLE_ETHERNET);
    }

    public function tarifPour(string $cible): Tarif
    {
        if ($cible === Tarif::CIBLE_ETHERNET) {
            $t = $this->tarifEthernet();
        } elseif ($cible === Tarif::CIBLE_WIFI) {
            $t = $this->tarifWifi();
        } else {
            $t = $this->tarifUnifie() ?? $this->tarifEthernet() ?? $this->tarifWifi();
        }

        if (!$t) {
            throw new InvalidArgumentException("Aucun tarif configuré pour « {$cible} ».");
        }

        return $t;
    }

    // -----------------------------------------------------------------
    // Arrondi et minimum
    // -----------------------------------------------------------------

    public function appliquerArrondi(int $montant, Tarif $tarif): int
    {
        if (!$tarif->arrondi_actif || $tarif->unite_arrondi <= 0) {
            return $montant;
        }

        $unite = max(1, (int) $tarif->unite_arrondi);
        $seuil = max(0, min($unite - 1, (int) $tarif->seuil_arrondi));
        $reste = $montant % $unite;

        if ($reste === 0) {
            return $montant;
        }

        return $reste < $seuil
            ? $montant - $reste
            : $montant + ($unite - $reste);
    }

    public function appliquerMinimum(int $montant, Tarif $tarif): int
    {
        if ($tarif->montant_minimum > 0 && $montant < $tarif->montant_minimum) {
            return $tarif->montant_minimum;
        }

        return $montant;
    }

    // -----------------------------------------------------------------
    // Conversions
    // -----------------------------------------------------------------

    public function calculerDuree(int $montant, ?Tarif $tarif = null): int
    {
        $tarif ??= $this->tarifStandard();

        if ($tarif->montant_par_minute <= 0) {
            throw new InvalidArgumentException("Prix/minute invalide.");
        }

        $montant = $this->appliquerMinimum($montant, $tarif);
        $montant = $this->appliquerArrondi($montant, $tarif);

        return intdiv($montant, $tarif->montant_par_minute);
    }

    public function montantPourMinutes(int $minutes, ?Tarif $tarif = null): int
    {
        $tarif ??= $this->tarifStandard();

        $brut = max(0, $minutes) * $tarif->montant_par_minute;

        $brut = $this->appliquerMinimum($brut, $tarif);
        return $this->appliquerArrondi($brut, $tarif);
    }

    // -----------------------------------------------------------------
    // Simulateur
    // -----------------------------------------------------------------

    /**
     * @return array{
     *   entree:int, mode:string, tarif:array, etapes:array, resultat:array
     * }
     */
    public function simuler(int $entree, Tarif $tarif, string $mode = 'montant'): array
    {
        $etapes = [];

        if ($mode === 'montant') {
            $brut = $entree;
            $etapes[] = ['label' => 'Montant saisi', 'valeur' => $this->formatAr($brut)];

            $min = $this->appliquerMinimum($brut, $tarif);
            if ($min !== $brut) {
                $etapes[] = ['label' => 'Minimum appliqué', 'valeur' => $this->formatAr($min)];
                $brut = $min;
            }

            $arr = $this->appliquerArrondi($brut, $tarif);
            if ($arr !== $brut) {
                $etapes[] = ['label' => 'Arrondi appliqué', 'valeur' => $this->formatAr($arr)];
                $brut = $arr;
            }

            $minutes = $tarif->montant_par_minute > 0
                ? intdiv($brut, $tarif->montant_par_minute)
                : 0;

            $etapes[] = [
                'label' => "Durée = {$brut} ÷ {$tarif->montant_par_minute} Ar/min",
                'valeur' => $this->formatDuree($minutes),
            ];

            $resultat = [
                'minutes'  => $minutes,
                'lisible'  => $this->formatDuree($minutes),
                'montant'  => $brut,
            ];
        } else {
            $minutes = max(0, $entree);
            $etapes[] = ['label' => 'Durée saisie', 'valeur' => $this->formatDuree($minutes)];

            $brut = $minutes * $tarif->montant_par_minute;
            $etapes[] = [
                'label' => "Montant brut = {$minutes} × {$tarif->montant_par_minute} Ar",
                'valeur' => $this->formatAr($brut),
            ];

            $min = $this->appliquerMinimum($brut, $tarif);
            if ($min !== $brut) {
                $etapes[] = ['label' => 'Minimum appliqué', 'valeur' => $this->formatAr($min)];
                $brut = $min;
            }

            $arr = $this->appliquerArrondi($brut, $tarif);
            if ($arr !== $brut) {
                $etapes[] = ['label' => 'Arrondi appliqué', 'valeur' => $this->formatAr($arr)];
                $brut = $arr;
            }

            $resultat = [
                'minutes' => $minutes,
                'lisible' => $this->formatDuree($minutes),
                'montant' => $brut,
            ];
        }

        return [
            'entree'   => $entree,
            'mode'     => $mode,
            'tarif'    => [
                'id'                 => $tarif->id,
                'cible'              => $tarif->cible,
                'libelle'            => $tarif->libelle(),
                'montant_par_minute' => $tarif->montant_par_minute,
                'montant_minimum'    => $tarif->montant_minimum,
                'arrondi_actif'      => (bool) $tarif->arrondi_actif,
                'unite_arrondi'      => $tarif->unite_arrondi,
                'seuil_arrondi'      => $tarif->seuil_arrondi,
            ],
            'etapes'   => $etapes,
            'resultat' => $resultat,
        ];
    }

    // -----------------------------------------------------------------

    public function formatAr(int $n): string
    {
        return number_format($n, 0, ',', ' ') . ' Ar';
    }

    public function formatDuree(int $minutes): string
    {
        $minutes = max(0, $minutes);

        if ($minutes < 60) {
            return "{$minutes} min";
        }

        $h = intdiv($minutes, 60);
        $r = $minutes % 60;

        return $r === 0
            ? "{$h} h ({$minutes} min)"
            : "{$h} h " . str_pad((string) $r, 2, '0', STR_PAD_LEFT) . " min ({$minutes} min)";
    }

        /**
     * Recalcule les durées des raccourcis d'un tarif selon le prix/minute.
     * Appelé après tout changement de tarif.
     * Le MONTANT est la référence (fixe) ; la DURÉE s'adapte.
     */
    public function resynchroniserRaccourcis(Tarif $tarif): void
    {
        if ($tarif->montant_par_minute <= 0) {
            return;
        }

        foreach ($tarif->raccourcis()->get() as $r) {
            $montant = (int) $r->montant;

            // Applique minimum puis arrondi, puis divise par le prix/minute
            $montantNet = $this->appliquerMinimum($montant, $tarif);
            $montantNet = $this->appliquerArrondi($montantNet, $tarif);

            $nouvelleDuree = intdiv($montantNet, $tarif->montant_par_minute);

            if ($nouvelleDuree !== (int) $r->duree) {
                $r->update(['duree' => $nouvelleDuree]);
            }
        }
    }
}

    public function storeRaccourci(Request $request, Tarif $tarif): RedirectResponse
    {

            'montant' => ['nullable', 'integer', 'min:1', 'max:1000000'],
            'h'       => ['nullable', 'integer', 'min:0', 'max:999'],
            'm'       => ['nullable', 'integer', 'min:0', 'max:59'],
            's'       => ['nullable', 'integer', 'min:0', 'max:59'],
            'libelle' => ['nullable', 'string', 'max:100'],
        ]);

        $data['mode'] = $data['mode'] ?? 'montant';

        if ($data['mode'] === 'montant') {
            if (empty($data['montant'])) {
                return back()->withErrors(['montant' => 'Indiquez un montant.']);
            }

            $montant = (int) $data['montant'];
            $duree   = $this->tarifService->calculerDuree($montant, $tarif);
        } else {
            $h = (int) ($data['h'] ?? 0);
            $m = (int) ($data['m'] ?? 0);
            $s = (int) ($data['s'] ?? 0);
            $minutes = $h * 60 + $m + ($s > 0 ? 1 : 0); // arrondi à la minute sup.

            if ($minutes < 1) {
                return back()->withErrors(['h' => 'Indiquez une durée supérieure à 0.']);
            }

            $montant = $this->tarifService->montantPourMinutes($minutes, $tarif);
            $duree   = $minutes;
        }

        if ($tarif->raccourcis()->where('montant', $montant)->exists()) {
            return back()
                ->withInput()
                ->withErrors([
                    "raccourci_tarif_{$tarif->id}" => "Un raccourci à " . number_format($montant, 0, ',', ' ') . " Ar existe déjà.",
                ]);
        }

        $tarif->raccourcis()->create([
            'montant' => $montant,
            'duree'   => $duree,
            'libelle' => $data['libelle'] ?? null,
            'ordre'   => ($tarif->raccourcis()->max('ordre') ?? 0) + 10,
        ]);

        return redirect()
            ->route('tarifs.index')
            ->with('success', "Raccourci ajouté : {$montant} Ar → {$duree} min.");
    }

    public function destroyRaccourci(Tarif $tarif, TarifRaccourci $raccourci): RedirectResponse

        return redirect()
            ->route('tarifs.index')
            ->with('success', 'Raccourci supprimé.');
    }

    // Simulateur
    // -----------------------------------------------------------------

    public function simuler(Request $request): JsonResponse
    {
        $data = $request->validate([
            'cible'  => ['required', Rule::in(['unifie', 'ethernet', 'wifi'])],
            'mode'   => ['required', Rule::in(['montant', 'duree'])],
            'entree' => ['required', 'integer', 'min:0', 'max:10000000'],
        ]);

        $tarif = $this->tarifService->tarifPour($data['cible']);

        return response()->json(
            $this->tarifService->simuler((int) $data['entree'], $tarif, $data['mode'])
        );
    }
