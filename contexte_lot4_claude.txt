    // ========================================================================
    //  Installateur Cyber Manager : à lancer en administrateur.
    //  Mode silencieux : CyberManagerInstall.exe /silent --server=URL --code=CODE [--nom="Poste 5"] [--sans-compte] [--redemarrer]
    // ========================================================================
    static class Program
    {
        [STAThread]
        static int Main(string[] args)
        {
            var o = Options.Lire(args);
            if (o.Silencieux)
            {
                try
                {
                    var r = Installation.Executer(o, (m, p) => Journal.Ecrire(m));
                    Journal.Ecrire("Installation terminée.");
                    if (!r.EcranAffiche)
                    {
                        Journal.Ecrire("Un redémarrage est nécessaire pour afficher l'écran de verrouillage.");
                        if (o.Redemarrer)
                        {
                            Installation.ProgrammerRedemarrage(120);
                            Journal.Ecrire("Redémarrage programmé dans 120 s (annulable avec : shutdown /a).");
                        }
                    }
                    return 0;
                }
                catch (Exception ex) { Journal.Ecrire("ÉCHEC : " + ex.Message); return 1; }
            }

            Application.EnableVisualStyles();
            Application.Run(new Fenetre(o));
            return 0;
        }
    }

    class Options
    {
        public string Serveur = "", Code = "", Nom = "";
        public bool SansCompte, Silencieux, Redemarrer;

        public static Options Lire(string[] args)
        {
            var o = new Options();

            // Indice d'adresse écrit au téléchargement (serveur.json à côté de l'exe) : un indice, jamais une condition
            try
            {
                var f = Path.Combine(AppDomain.CurrentDomain.BaseDirectory, "serveur.json");
                if (File.Exists(f))
                {
                    var d = new JavaScriptSerializer().DeserializeObject(File.ReadAllText(f)) as Dictionary<string, object>;
                    if (d != null && d.ContainsKey("server_url")) o.Serveur = Convert.ToString(d["server_url"]);
                }
            }
            catch { }

            foreach (var a in args)
            {
                if (a.Equals("/silent", StringComparison.OrdinalIgnoreCase) || a.Equals("--silent", StringComparison.OrdinalIgnoreCase)) o.Silencieux = true;
                else if (a.Equals("--sans-compte", StringComparison.OrdinalIgnoreCase)) o.SansCompte = true;
                else if (a.Equals("--redemarrer", StringComparison.OrdinalIgnoreCase)) o.Redemarrer = true;
                else if (a.StartsWith("--server=", StringComparison.OrdinalIgnoreCase)) o.Serveur = a.Substring(9).Trim().TrimEnd('/');
                else if (a.StartsWith("--code=", StringComparison.OrdinalIgnoreCase)) o.Code = a.Substring(7).Trim();
                else if (a.StartsWith("--nom=", StringComparison.OrdinalIgnoreCase)) o.Nom = a.Substring(6).Trim();
            }
            return o;
        }
    }

    static class Journal
    {
        static readonly string Dossier = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.CommonApplicationData), "CyberManager", "logs");

        public static void Ecrire(string m)
        {
            try
            {
                Directory.CreateDirectory(Dossier);
                File.AppendAllText(Path.Combine(Dossier, "installateur.log"), "[" + DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss") + "] " + m + Environment.NewLine);
            }
            catch { }
        }
    }

    /// <summary>Résultat de l'installation : l'Écran est-il déjà affiché, ou faut-il redémarrer ?</summary>
    class ResultatInstallation
    {
        public bool EcranAffiche;
        public bool UtilisateurEstClient;
        public bool AutoLogon;
        public string Utilisateur = "";

        public string Explication()
        {
            if (UtilisateurEstClient)
                return "L'écran de verrouillage ne s'est pas encore affiché. Redémarrez le poste pour terminer.";
            if (AutoLogon)
                return "Vous êtes connecté avec un autre compte que « Client » : l'écran de verrouillage ne s'affiche que sur le compte « Client ». Au redémarrage, le poste ouvrira la session « Client » tout seul.";
            return "Vous êtes connecté avec un autre compte que « Client » : l'écran de verrouillage ne s'affiche que sur le compte « Client ». Au redémarrage, ouvrez la session « Client ».";
        }
    }

    static class SessionWindows
    {
        [DllImport("kernel32.dll")] static extern uint WTSGetActiveConsoleSessionId();
        [DllImport("wtsapi32.dll", CharSet = CharSet.Unicode, SetLastError = true)]
        static extern bool WTSQuerySessionInformation(IntPtr serveur, uint session, int classe, out IntPtr tampon, out int octets);
        [DllImport("wtsapi32.dll")] static extern void WTSFreeMemory(IntPtr tampon);

        public static uint Console() { return WTSGetActiveConsoleSessionId(); }

        /// <summary>Nom du compte connecté sur la console (vide si personne).</summary>
        public static string UtilisateurConsole()
        {
            uint s = WTSGetActiveConsoleSessionId();
            if (s == 0xFFFFFFFF) return "";
            IntPtr p; int n;
            if (!WTSQuerySessionInformation(IntPtr.Zero, s, 5, out p, out n)) return "";   // 5 = WTSUserName
            try { return Marshal.PtrToStringUni(p) ?? ""; }
            finally { WTSFreeMemory(p); }
        }
    }

    static class Installation
    {
        const string NomService = "CyberManagerGardien";
        const string CompteClient = "Client";

        static readonly string Racine = Path.Combine(
            Environment.GetEnvironmentVariable("ProgramW6432") ?? Environment.GetFolderPath(Environment.SpecialFolder.ProgramFiles), "CyberManager");
        static readonly string DossierGardien = Path.Combine(Racine, "Gardien");
        static readonly string DossierEcran = Path.Combine(Racine, "Ecran");
        static readonly string Temp = Path.Combine(Path.GetTempPath(), "CyberManagerInstall");

        public static ResultatInstallation Executer(Options o, Action<string, int> progres)
        {
            if (string.IsNullOrWhiteSpace(o.Serveur)) throw new Exception("Adresse du serveur manquante.");
            if (string.IsNullOrWhiteSpace(o.Code)) throw new Exception("Code d'installation manquant.");
            var serveur = o.Serveur.Trim().TrimEnd('/');
            Uri u;
            if (!Uri.TryCreate(serveur, UriKind.Absolute, out u) || (u.Scheme != "http" && u.Scheme != "https"))
                throw new Exception("Adresse du serveur invalide (exemple : http://192.168.1.10:8000).");

            using (var http = new HttpClient { Timeout = TimeSpan.FromMinutes(15) })
            {
                http.DefaultRequestHeaders.Add("X-Install-Code", o.Code.Trim());

                // 1. Le serveur
                progres("Recherche du serveur…", 5);
                Dictionary<string, object> info;
                try { info = new JavaScriptSerializer().DeserializeObject(http.GetStringAsync(serveur + "/api/agent/installation/info").GetAwaiter().GetResult()) as Dictionary<string, object>; }
                catch (Exception ex) { throw new Exception("Serveur injoignable à l'adresse " + serveur + " (" + Court(ex) + "). Vérifiez l'adresse et le pare-feu du serveur."); }
                if (info == null || !info.ContainsKey("cyber_manager") || !(info["cyber_manager"] is bool) || !(bool)info["cyber_manager"])
                    throw new Exception("Ce serveur n'est pas un serveur Cyber Manager.");

                var version = info.ContainsKey("version_ecran") ? info["version_ecran"] as string : null;
                var shaAttendu = info.ContainsKey("sha256_ecran") ? info["sha256_ecran"] as string : null;
                if (string.IsNullOrEmpty(version)) throw new Exception("Aucune version de l'Écran n'est disponible sur le serveur (onglet Versions).");
                bool webviewDispo = info.ContainsKey("webview2") && info["webview2"] is bool && (bool)info["webview2"];

                Directory.CreateDirectory(Temp);

                // 2. WebView2 seulement s'il est absent (version hors ligne, téléchargée depuis le serveur)
                if (!WebView2Present())
                {
                    if (!webviewDispo) throw new Exception("WebView2 est absent du poste et le serveur n'en fournit pas (placez webview2.exe dans storage/app/private/installateur).");
                    progres("Installation de WebView2…", 15);
                    var wv = Path.Combine(Temp, "webview2.exe");
                    Telecharger(http, serveur + "/api/agent/installation/webview2", wv);
                    Lancer(wv, "/silent /install", 600000);
                    if (!WebView2Present()) throw new Exception("L'installation de WebView2 a échoué.");
                }

                // 3. Arrêt d'une éventuelle installation précédente
                progres("Arrêt de l'ancienne version…", 25);
                Cmd("sc.exe", "stop " + NomService);
                System.Threading.Thread.Sleep(3000);
                foreach (var p in Process.GetProcessesByName("cyber-manager-agent")) { try { p.Kill(); } catch { } }

                // 4. Gardien
                progres("Téléchargement du Gardien…", 35);
                var zipG = Path.Combine(Temp, "gardien.zip");
                Telecharger(http, serveur + "/api/agent/installation/gardien", zipG);
                Extraire(zipG, DossierGardien);
                var exeGardien = Path.Combine(DossierGardien, "cyber-manager-gardien.exe");
                if (!File.Exists(exeGardien)) throw new Exception("gardien.zip ne contient pas cyber-manager-gardien.exe.");

                // 5. Écran (empreinte vérifiée)
                progres("Téléchargement de l'Écran " + version + "…", 50);
                var zipE = Path.Combine(Temp, "ecran.zip");
                Telecharger(http, serveur + "/api/agent/installation/ecran", zipE);
                if (!string.IsNullOrEmpty(shaAttendu) && !Sha256(zipE).Equals(shaAttendu, StringComparison.OrdinalIgnoreCase))
                    throw new Exception("Empreinte SHA-256 de l'Écran incorrecte : fichier altéré ou incomplet.");
                var dossierVersion = Path.Combine(DossierEcran, "versions", version);
                progres("Installation de l'Écran…", 65);
                Extraire(zipE, dossierVersion);
                // Courante = Validée : l'Écran installé à la main est considéré comme bon
                File.WriteAllText(Path.Combine(DossierEcran, "courante.txt"), version);
                File.WriteAllText(Path.Combine(DossierEcran, "validee.txt"), version);

                // 6. Configuration du Gardien (ProgramData)
                progres("Configuration du poste…", 75);
                var args = "--configurer --server=" + serveur + " --code=" + o.Code.Trim();
                if (!string.IsNullOrWhiteSpace(o.Nom)) args += " \"--nom=" + o.Nom.Trim().Replace("\"", "") + "\"";
                var rc = Lancer(exeGardien, args, 60000);
                if (rc != 0) throw new Exception("La configuration du Gardien a échoué (code " + rc + ").");

                // 7. Compte Client
                if (!o.SansCompte) { progres("Compte « Client »…", 85); CreerCompteClient(); }

                // 8. Service Windows (démarrage automatique + relance en cas de panne)
                progres("Démarrage du service…", 92);
                var bin = "\\\"" + exeGardien + "\\\"";
                if (Cmd("sc.exe", "query " + NomService) == 0) Cmd("sc.exe", "config " + NomService + " binPath= \"" + bin + "\" start= auto");
                else if (Cmd("sc.exe", "create " + NomService + " binPath= \"" + bin + "\" start= auto DisplayName= \"Cyber Manager Gardien\"") != 0)
                    throw new Exception("Création du service impossible (droits administrateur ?).");
                Cmd("sc.exe", "failure " + NomService + " reset= 86400 actions= restart/5000/restart/5000/restart/5000");
                if (Cmd("sc.exe", "start " + NomService) != 0) Journal.Ecrire("Démarrage du service : à vérifier (peut-être déjà démarré).");

                // 9. L'Écran s'affiche-t-il tout de suite ? Oui si la session ouverte est celle du compte « Client » :
                //    le Gardien l'y lance de lui-même. Sinon (session administrateur) : redémarrage assisté.
                progres("Vérification de l'écran de verrouillage…", 96);
                var res = new ResultatInstallation();
                res.Utilisateur = SessionWindows.UtilisateurConsole();
                res.UtilisateurEstClient = string.Equals(res.Utilisateur, CompteClient, StringComparison.OrdinalIgnoreCase);
                res.AutoLogon = AutoLogonClient();
                res.EcranAffiche = EcranAffiche(res.UtilisateurEstClient ? 30 : 12);

                try { Directory.Delete(Temp, true); } catch { }
                progres(res.EcranAffiche
                    ? "Terminé. L'écran de verrouillage est affiché, aucun redémarrage n'est nécessaire. Le poste apparaît dans l'interface web (désactivé) : activez-le."
                    : "Installation terminée. Un redémarrage du poste est nécessaire pour afficher l'écran de verrouillage.", 100);
                return res;
            }
        }

        /// <summary>Un Écran tourne-t-il dans la session affichée à l'écran ? (attente maximale en secondes)</summary>
        static bool EcranAffiche(int secondes)
        {
            var fin = DateTime.UtcNow.AddSeconds(secondes);
            while (DateTime.UtcNow < fin)
            {
                var console = SessionWindows.Console();
                foreach (var p in Process.GetProcessesByName("cyber-manager-agent"))
                {
                    try { if ((uint)p.SessionId == console) return true; }
                    catch { }
                    finally { p.Dispose(); }
                }
                System.Threading.Thread.Sleep(1000);
            }
            return false;
        }

        static bool AutoLogonClient()
        {
            try
            {
                using (var k = Registry.LocalMachine.OpenSubKey(@"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon"))
                {
                    return k != null
                        && Convert.ToString(k.GetValue("AutoAdminLogon")) == "1"
                        && string.Equals(Convert.ToString(k.GetValue("DefaultUserName")), CompteClient, StringComparison.OrdinalIgnoreCase);
                }
            }
            catch { return false; }
        }

        /// <summary>Redémarrage annulable (shutdown /a). Jamais forcé : pas de /f.</summary>
        public static void ProgrammerRedemarrage(int secondes)
        {
            Lancer("shutdown.exe", "/r /t " + secondes + " /c \"Cyber Manager : redémarrage pour terminer l'installation\"", 10000);
        }

        static void CreerCompteClient()
        {
            if (Cmd("net.exe", "user " + CompteClient) == 0)
            {
                Journal.Ecrire("Le compte « " + CompteClient + " » existe déjà : non modifié, ouverture automatique non configurée.");
                return;
            }
            // Compte standard (jamais administrateur), sans mot de passe
            if (Cmd("net.exe", "user " + CompteClient + " \"\" /add /active:yes /expires:never /passwordchg:no") != 0)
            { Journal.Ecrire("Création du compte « " + CompteClient + " » impossible."); return; }

            using (var k = Registry.LocalMachine.CreateSubKey(@"SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon"))
            {
                k.SetValue("AutoAdminLogon", "1");
                k.SetValue("DefaultUserName", CompteClient);
                k.SetValue("DefaultPassword", "");
                k.SetValue("DefaultDomainName", Environment.MachineName);
            }
        }

        static bool WebView2Present()
        {
            const string guid = @"SOFTWARE\Microsoft\EdgeUpdate\Clients\{F3017226-FE2A-4295-8BDF-00C3A9A7E4C5}";
            foreach (var chemin in new[] { @"SOFTWARE\WOW6432Node\Microsoft\EdgeUpdate\Clients\{F3017226-FE2A-4295-8BDF-00C3A9A7E4C5}", guid })
            {
                using (var k = Registry.LocalMachine.OpenSubKey(chemin))
                {
                    var v = k == null ? null : k.GetValue("pv") as string;
                    if (!string.IsNullOrEmpty(v) && v != "0.0.0.0") return true;
                }
            }
            return false;
        }

        static void Telecharger(HttpClient http, string url, string dest)
        {
            using (var r = http.GetAsync(url, HttpCompletionOption.ResponseHeadersRead).GetAwaiter().GetResult())
            {
                if (!r.IsSuccessStatusCode)
                    throw new Exception(((int)r.StatusCode == 403 ? "Code d'installation refusé par le serveur." : "Téléchargement refusé (" + (int)r.StatusCode + ")."));
                using (var s = r.Content.ReadAsStreamAsync().GetAwaiter().GetResult())
                using (var f = File.Create(dest)) s.CopyTo(f);
            }
        }

        static void Extraire(string zip, string dossier)
        {
            try { if (Directory.Exists(dossier)) Directory.Delete(dossier, true); } catch { }
            Directory.CreateDirectory(dossier);
            ZipFile.ExtractToDirectory(zip, dossier);
        }

        static string Sha256(string f)
        {
            using (var s = File.OpenRead(f))
            {
                var h = SHA256.Create().ComputeHash(s);
                var sb = new StringBuilder();
                foreach (var b in h) sb.Append(b.ToString("x2"));
                return sb.ToString();
            }
        }

        static int Cmd(string exe, string args) { return Lancer(exe, args, 30000); }

        static int Lancer(string exe, string args, int attenteMs)
        {
            var psi = new ProcessStartInfo(exe, args) { UseShellExecute = false, CreateNoWindow = true, RedirectStandardOutput = true, RedirectStandardError = true };
            using (var p = Process.Start(psi))
            {
                p.StandardOutput.ReadToEnd();   // le code d'installation n'est jamais écrit dans le journal
                p.StandardError.ReadToEnd();
                if (!p.WaitForExit(attenteMs)) { try { p.Kill(); } catch { } return -1; }
                return p.ExitCode;
            }
        }

        static string Court(Exception ex) { var e = ex; while (e.InnerException != null) e = e.InnerException; return e.Message; }
    }

    class Fenetre : Form
    {
        readonly TextBox serveur = new TextBox(), code = new TextBox(), nom = new TextBox();
        readonly CheckBox compte = new CheckBox { Text = "Créer le compte Windows « Client » (standard, sans mot de passe) avec ouverture automatique", Checked = true, AutoSize = true };
        readonly Button installer = new Button { Text = "Installer", Height = 36 };
        readonly ProgressBar barre = new ProgressBar { Dock = DockStyle.Top, Height = 18 };
        readonly Label etat = new Label { AutoSize = false, Height = 48, ForeColor = Color.DimGray };

        // Redémarrage assisté : décompte visible, « Plus tard », l'utilisateur garde la main
        readonly FlowLayoutPanel panneauRedem = new FlowLayoutPanel { FlowDirection = FlowDirection.TopDown, AutoSize = true, Visible = false, Margin = new Padding(0, 8, 0, 0) };
        readonly Label explication = new Label { AutoSize = true, MaximumSize = new Size(500, 0) };
        readonly Label decompte = new Label { AutoSize = true, Font = new Font(SystemFonts.DefaultFont, FontStyle.Bold), Margin = new Padding(3, 6, 3, 6) };
        readonly Button maintenant = new Button { Text = "Redémarrer maintenant", AutoSize = true };
        readonly Button plusTard = new Button { Text = "Plus tard", AutoSize = true };
        readonly System.Windows.Forms.Timer minuteur = new System.Windows.Forms.Timer { Interval = 1000 };
        int restant;
        bool termine;

        public Fenetre(Options o)
        {
            Text = "Cyber Manager : installation du poste";
            Width = 560; Height = 560; StartPosition = FormStartPosition.CenterScreen; FormBorderStyle = FormBorderStyle.FixedDialog; MaximizeBox = false;
            serveur.Text = o.Serveur; code.Text = o.Code; nom.Text = o.Nom; compte.Checked = !o.SansCompte;

            var t = new TableLayoutPanel { Dock = DockStyle.Fill, ColumnCount = 1, Padding = new Padding(18), AutoSize = false };
            Action<string, Control> ajouter = (titre, c) =>
            {
                t.Controls.Add(new Label { Text = titre, AutoSize = true, Font = new Font(Font, FontStyle.Bold), Margin = new Padding(0, 8, 0, 2) });
                c.Dock = DockStyle.Top; t.Controls.Add(c);
            };
            ajouter("Adresse du serveur (trouvée automatiquement si elle est connue)", serveur);
            ajouter("Code d'installation (affiché dans l'interface web, page Enregistrement)", code);
            ajouter("Nom du poste (vide = attribué par le serveur : Poste 1, Poste 2…)", nom);
            t.Controls.Add(compte);
            installer.Dock = DockStyle.Top; installer.Margin = new Padding(0, 12, 0, 6);
            t.Controls.Add(installer); t.Controls.Add(barre); t.Controls.Add(etat);

            var boutons = new FlowLayoutPanel { AutoSize = true };
            boutons.Controls.Add(maintenant); boutons.Controls.Add(plusTard);
            panneauRedem.Controls.Add(explication); panneauRedem.Controls.Add(decompte); panneauRedem.Controls.Add(boutons);
            t.Controls.Add(panneauRedem);
            Controls.Add(t);

            minuteur.Tick += (s, e) =>
            {
                restant--;
                if (restant <= 0) { minuteur.Stop(); Redemarrer(); }
                else decompte.Text = "Redémarrage du poste dans " + restant + " s";
            };
            maintenant.Click += (s, e) => { minuteur.Stop(); Redemarrer(); };
            plusTard.Click += (s, e) =>
            {
                minuteur.Stop();
                decompte.Text = "Redémarrage reporté. Redémarrez le poste quand vous voulez : l'écran de verrouillage s'affichera alors.";
                plusTard.Enabled = false;
            };
            FormClosing += (s, e) => minuteur.Stop();

            installer.Click += async (s, e) =>
            {
                if (termine) { Close(); return; }

                installer.Enabled = false; serveur.Enabled = code.Enabled = nom.Enabled = compte.Enabled = false;
                var opt = new Options { Serveur = serveur.Text.Trim().TrimEnd('/'), Code = code.Text.Trim(), Nom = nom.Text.Trim(), SansCompte = !compte.Checked };
                try
                {
                    var res = await Task.Run(() => Installation.Executer(opt, (m, p) => { Journal.Ecrire(m); BeginInvoke((Action)(() => { etat.Text = m; barre.Value = Math.Max(0, Math.Min(100, p)); })); }));
                    etat.ForeColor = Color.SeaGreen;
                    termine = true;
                    installer.Text = "Fermer"; installer.Enabled = true;

                    if (!res.EcranAffiche) ProposerRedemarrage(res.Explication());
                }
                catch (Exception ex)
                {
                    Journal.Ecrire("ÉCHEC : " + ex.Message);
                    etat.ForeColor = Color.Firebrick; etat.Text = ex.Message;
                    installer.Enabled = true; serveur.Enabled = code.Enabled = nom.Enabled = compte.Enabled = true;
                }
            };
            AcceptButton = installer;
        }

        void ProposerRedemarrage(string texte)
        {
            restant = 60;
            explication.Text = texte;
            decompte.Text = "Redémarrage du poste dans " + restant + " s";
            plusTard.Enabled = true;
            panneauRedem.Visible = true;
            minuteur.Start();
        }

        static void Redemarrer()
        {
            try
            {
                // Jamais forcé (pas de /f) : Windows demande d'enregistrer les travaux ouverts
                Process.Start(new ProcessStartInfo("shutdown.exe", "/r /t 5 /c \"Cyber Manager : redémarrage pour terminer l'installation\"")
                { CreateNoWindow = true, UseShellExecute = false });
            }
            catch (Exception ex) { Journal.Ecrire("Redémarrage impossible : " + ex.Message); }
        }
    }

# Fabrique le paquet de l'Écran + le Gardien + l'installateur. Usage :
#   .\fabriquer.ps1                      -> paquet prêt, publié sur AUCUN poste
#   .\fabriquer.ps1 -Pilotes 3           -> pilote : poste d'identifiant 3 seulement
#   .\fabriquer.ps1 -General             -> tous les postes
param([int[]]$Pilotes = @(), [switch]$General)
$ErrorActionPreference = "Stop"

$commun  = $PSScriptRoot
$racine  = Split-Path -Parent $commun
$serveur = Join-Path $racine "cyber-manager"
$dist    = Join-Path $commun "dist"

$version = [regex]::Match((Get-Content "$commun\Protocole.cs" -Raw), 'VersionEcran\s*=\s*"([^"]+)"').Groups[1].Value
if (-not $version) { throw "Version introuvable dans Protocole.cs" }
Write-Host "== Fabrication de l'Écran $version ==" -ForegroundColor Cyan

if (Test-Path $dist) { Remove-Item $dist -Recurse -Force }
New-Item -ItemType Directory "$dist\ecran", "$dist\gardien", "$dist\installateur" | Out-Null

# Écran (autonome : pas de .NET à installer sur les postes)
dotnet publish "$racine\cyber-manager-agent" -c Release -r win-x64 --self-contained true -o "$dist\ecran"
if ($LASTEXITCODE -ne 0) { throw "Échec de la publication de l'Écran" }

# Sécurité : aucun fichier de configuration (appsettings*.json) dans le paquet
Get-ChildItem "$dist\ecran" -Filter "appsettings*.json" -ErrorAction SilentlyContinue | Remove-Item -Force

# Paquet + empreinte
$zip = Join-Path $dist "ecran-$version.zip"
Compress-Archive -Path "$dist\ecran\*" -DestinationPath $zip -CompressionLevel Optimal
$sha = (Get-FileHash $zip -Algorithm SHA256).Hash.ToLower()
Write-Host "Paquet : $zip" -ForegroundColor Green
Write-Host "SHA-256 : $sha"

# Dépôt PRIVÉ du serveur + manifeste
$depot = Join-Path $serveur "storage\app\private\agent-packages"
New-Item -ItemType Directory $depot -Force | Out-Null
Copy-Item $zip $depot -Force
@{ version = $version; fichier = "ecran-$version.zip"; sha256 = $sha; general = [bool]$General; pilotes = @($Pilotes) } |
    ConvertTo-Json | Set-Content (Join-Path $depot "manifest.json") -Encoding UTF8
Write-Host "Manifeste écrit (général = $([bool]$General), pilotes = $($Pilotes -join ','))" -ForegroundColor Green

# Gardien (un seul fichier)
Write-Host "== Fabrication du Gardien ==" -ForegroundColor Cyan
dotnet publish "$racine\cyber-manager-gardien" -c Release -r win-x64 --self-contained true -p:PublishSingleFile=true -o "$dist\gardien"
if ($LASTEXITCODE -ne 0) { throw "Échec de la publication du Gardien" }

# Installateur (petit exe .NET Framework 4.8, déjà présent sur Windows 10/11)
Write-Host "== Fabrication de l'installateur ==" -ForegroundColor Cyan
dotnet build "$racine\cyber-manager-installateur" -c Release -o "$dist\installateur"
if ($LASTEXITCODE -ne 0) { throw "Échec de la fabrication de l'installateur" }

# Dépôt PRIVÉ de l'installation : gardien.zip + installateur (webview2.exe est à déposer une seule fois à la main)
$depotInstall = Join-Path $serveur "storage\app\private\installateur"
New-Item -ItemType Directory $depotInstall -Force | Out-Null
$zipGardien = Join-Path $dist "gardien.zip"
Compress-Archive -Path "$dist\gardien\*" -DestinationPath $zipGardien -CompressionLevel Optimal
Copy-Item $zipGardien $depotInstall -Force
Copy-Item "$dist\installateur\CyberManagerInstall.exe" $depotInstall -Force
Write-Host "Installateur et Gardien déposés dans $depotInstall" -ForegroundColor Green

Write-Host "Terminé. Écran : $dist\ecran   Gardien : $dist\gardien   Installateur : $dist\installateur" -ForegroundColor Green

using System.Text.Json;
using System.Text.Json.Serialization;

public static class Constantes
{
    /// <summary>Version de l'Écran : DOIT correspondre au nom du dossier/paquet (lue par fabriquer.ps1).</summary>
    public const string VersionEcran = "2.2.0";
    public const string VersionGardien = "1.1.0";
    public const string NomPipe = "CyberManagerGardien";
}

/// <summary>
/// Message unique (champs optionnels) du canal local.
/// Gardien → Écran : etat, branding, cmd.     Écran → Gardien : pret, req, extinction, reveil, repcmd.
/// Gardien → Écran (réponse à req) : rep.
/// </summary>
public class MessageCanal
{
    [JsonPropertyName("t")] public string Type { get; set; } = "";
    [JsonPropertyName("id")] public int Id { get; set; }

    // Écran → Gardien : requête (activer, suspendre, reprendre, fermer, message)
    [JsonPropertyName("action")] public string? Action { get; set; }
    [JsonPropertyName("texte")] public string? Texte { get; set; }

    // Gardien → Écran : état du poste
    [JsonPropertyName("poste")] public string? Poste { get; set; }
    [JsonPropertyName("ip")] public string? Ip { get; set; }
    [JsonPropertyName("etat")] public string? Etat { get; set; }
    [JsonPropertyName("verrou")] public bool? Verrou { get; set; }
    [JsonPropertyName("session")] public SessionInfo? Session { get; set; }

    // Gardien → Écran : identité du cyber (logo en base64)
    [JsonPropertyName("nom")] public string? Nom { get; set; }
    [JsonPropertyName("slogan")] public string? Slogan { get; set; }
    [JsonPropertyName("couleur")] public string? Couleur { get; set; }
    [JsonPropertyName("logo")] public string? Logo { get; set; }

    // Gardien → Écran : commande du serveur
    [JsonPropertyName("commande")] public string? Commande { get; set; }
    [JsonPropertyName("payload")] public string? Payload { get; set; }

    // Réponses (rep, repcmd)
    [JsonPropertyName("succes")] public bool? Succes { get; set; }
    [JsonPropertyName("message")] public string? Message { get; set; }
    [JsonPropertyName("montant")] public int? Montant { get; set; }
    [JsonPropertyName("avertissement")] public string? Avertissement { get; set; }
}

public static class ProtocoleCanal
{
    private static readonly JsonSerializerOptions Options = new() { DefaultIgnoreCondition = JsonIgnoreCondition.WhenWritingNull };

    public static string Encoder(MessageCanal m) => JsonSerializer.Serialize(m, Options);

    public static MessageCanal? Decoder(string ligne)
    {
        try { return JsonSerializer.Deserialize<MessageCanal>(ligne, Options); }
        catch { return null; }
    }
}

using System.Collections.Concurrent;
using System.IO.Pipes;
using System.Security.AccessControl;
using System.Security.Principal;
using System.Text;

public sealed class CanalEcran
{
    private readonly object _verrou = new();
    private StreamWriter? _ecrivain;
    private readonly ConcurrentDictionary<int, TaskCompletionSource<MessageCanal>> _commandesEnAttente = new();
    private readonly ConcurrentDictionary<string, TaskCompletionSource<bool>> _pretAttendus = new();

    /// <summary>Message de l'Écran à traiter (req, extinction, reveil).</summary>
    public event Action<MessageCanal>? MessageRecu;

    /// <summary>L'Écran vient de se connecter et se déclare prêt (paramètre : sa version).</summary>
    public event Action<string>? EcranPret;

    public bool Connecte { get { lock (_verrou) return _ecrivain != null; } }

    public async Task ExecuterAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            NamedPipeServerStream? pipe = null;
            try
            {
                pipe = NamedPipeServerStreamAcl.Create(Constantes.NomPipe, PipeDirection.InOut, 1,
                    PipeTransmissionMode.Byte, PipeOptions.Asynchronous, 0, 0, Securite());
                await pipe.WaitForConnectionAsync(ct);

                var utf8 = new UTF8Encoding(false);
                using var lecteur = new StreamReader(pipe, utf8);
                var ecrivain = new StreamWriter(pipe, utf8) { AutoFlush = true };
                lock (_verrou) _ecrivain = ecrivain;

                string? ligne;
                while ((ligne = await lecteur.ReadLineAsync(ct)) != null) Traiter(ligne);
            }
            catch (OperationCanceledException) { break; }
            catch (Exception ex) { Journal.Ecrire($"Canal Écran : {ex.Message}"); }
            finally
            {
                lock (_verrou) _ecrivain = null;
                foreach (var attente in _commandesEnAttente.Values)
                    attente.TrySetResult(new MessageCanal { Succes = false, Message = "Écran déconnecté." });
                pipe?.Dispose();
            }

            try { await Task.Delay(300, ct); } catch (OperationCanceledException) { break; }
        }
    }

    private static PipeSecurity Securite()
    {
        var ps = new PipeSecurity();
        ps.AddAccessRule(new PipeAccessRule(new SecurityIdentifier(WellKnownSidType.LocalSystemSid, null),
            PipeAccessRights.FullControl, AccessControlType.Allow));
        ps.AddAccessRule(new PipeAccessRule(new SecurityIdentifier(WellKnownSidType.BuiltinAdministratorsSid, null),
            PipeAccessRights.FullControl, AccessControlType.Allow));
        ps.AddAccessRule(new PipeAccessRule(new SecurityIdentifier(WellKnownSidType.AuthenticatedUserSid, null),
            PipeAccessRights.ReadWrite, AccessControlType.Allow));
        return ps;
    }

    private void Traiter(string ligne)
    {
        var m = ProtocoleCanal.Decoder(ligne);
        if (m == null) return;

        switch (m.Type)
        {
            case "pret":
                var version = m.Texte ?? "";
                if (_pretAttendus.TryRemove(version, out var tcs)) tcs.TrySetResult(true);
                try { EcranPret?.Invoke(version); } catch { }
                break;

            case "repcmd":
                if (_commandesEnAttente.TryRemove(m.Id, out var attente)) attente.TrySetResult(m);
                break;

            default:
                // Traité hors de la boucle de lecture : une requête lente ne bloque jamais le canal
                _ = Task.Run(() => { try { MessageRecu?.Invoke(m); } catch (Exception ex) { Journal.Ecrire($"Message Écran : {ex.Message}"); } });
                break;
        }
    }

    public bool Envoyer(MessageCanal message)
    {
        try
        {
            lock (_verrou)
            {
                if (_ecrivain == null) return false;
                _ecrivain.WriteLine(ProtocoleCanal.Encoder(message));
                return true;
            }
        }
        catch { return false; }
    }

    /// <summary>Envoie une commande du serveur à l'Écran et attend son résultat.</summary>
    public async Task<(bool ok, string detail)> EnvoyerCommandeAsync(int id, string type, string? payload, TimeSpan delai)
    {
        var tcs = new TaskCompletionSource<MessageCanal>(TaskCreationOptions.RunContinuationsAsynchronously);
        _commandesEnAttente[id] = tcs;

        if (!Envoyer(new MessageCanal { Type = "cmd", Id = id, Commande = type, Payload = payload }))
        {
            _commandesEnAttente.TryRemove(id, out _);
            return (false, "Écran non connecté");
        }

        var fini = await Task.WhenAny(tcs.Task, Task.Delay(delai));
        _commandesEnAttente.TryRemove(id, out _);
        if (fini != tcs.Task) return (false, "Écran ne répond pas");

        var r = tcs.Task.Result;
        return (r.Succes == true, r.Message ?? "");
    }

    /// <summary>À appeler AVANT de lancer l'Écran : vrai si cette version se déclare prête dans le délai.</summary>
    public Task<bool> AttendrePretAsync(string version, TimeSpan delai)
    {
        var tcs = new TaskCompletionSource<bool>(TaskCreationOptions.RunContinuationsAsynchronously);
        _pretAttendus[version] = tcs;
        _ = Task.Delay(delai).ContinueWith(_ => tcs.TrySetResult(false));
        return tcs.Task;
    }
}

using System.Runtime.InteropServices;
using System.Security.Principal;

public enum ResultatNom { Applique, DejaAJour, CompteAbsent, Erreur }

/// <summary>Outils sur les comptes Windows : détection d'un administrateur, nom affiché (FullName).</summary>
public static class CompteWindows
{
    private const int TokenElevationType = 18;   // 1 = standard / UAC coupé, 2 = élevé, 3 = administrateur filtré par l'UAC
    private const int NERR_UserNotFound = 2221;

    [StructLayout(LayoutKind.Sequential)]
    private struct USER_INFO_1011 { public IntPtr usri1011_full_name; }

    [DllImport("advapi32.dll", SetLastError = true)]
    private static extern bool GetTokenInformation(IntPtr jeton, int classe, IntPtr info, int taille, out int requis);

    [DllImport("netapi32.dll", CharSet = CharSet.Unicode)]
    private static extern int NetUserGetInfo(string? serveur, string utilisateur, int niveau, out IntPtr tampon);

    [DllImport("netapi32.dll", CharSet = CharSet.Unicode)]
    private static extern int NetUserSetInfo(string? serveur, string utilisateur, int niveau, ref USER_INFO_1011 tampon, out int erreurParam);

    [DllImport("netapi32.dll")]
    private static extern int NetApiBufferFree(IntPtr tampon);

    /// <summary>
    /// Le propriétaire de ce jeton (session ouverte) est-il administrateur ?
    /// Administrateur avec UAC (jeton filtré) ou sans UAC (groupe actif). En cas de doute : vrai (on ne verrouille pas).
    /// </summary>
    public static bool JetonEstAdministrateur(IntPtr jeton)
    {
        try
        {
            var tampon = Marshal.AllocHGlobal(4);
            try
            {
                if (GetTokenInformation(jeton, TokenElevationType, tampon, 4, out _))
                {
                    var type = Marshal.ReadInt32(tampon);
                    if (type is 2 or 3) return true;
                }
            }
            finally { Marshal.FreeHGlobal(tampon); }

            using var identite = new WindowsIdentity(jeton);
            return new WindowsPrincipal(identite).IsInRole(WindowsBuiltInRole.Administrator);
        }
        catch (Exception ex)
        {
            Journal.Ecrire($"Type de compte indéterminé ({ex.Message}) : traité comme administrateur (aucun verrouillage).");
            return true;
        }
    }

    /// <summary>Change le nom AFFICHÉ (FullName) d'un compte. Le nom de connexion (SAM) et le profil ne sont jamais touchés.</summary>
    public static (ResultatNom, string) DefinirNomAffiche(string compte, string nom)
    {
        nom = Nettoyer(nom);
        if (nom.Length == 0) return (ResultatNom.Erreur, "Nom vide.");

        var tampon = IntPtr.Zero;
        try
        {
            var rc = NetUserGetInfo(null, compte, 10, out tampon);
            if (rc == NERR_UserNotFound) return (ResultatNom.CompteAbsent, $"Le compte « {compte} » n'existe pas sur ce PC : nom affiché non appliqué.");
            if (rc != 0) return (ResultatNom.Erreur, $"Lecture du compte « {compte} » impossible (code {rc}).");

            // USER_INFO_10 : name, comment, usr_comment, full_name (4 pointeurs)
            var actuel = Marshal.PtrToStringUni(Marshal.ReadIntPtr(tampon, 3 * IntPtr.Size)) ?? "";
            if (string.Equals(actuel, nom, StringComparison.Ordinal)) return (ResultatNom.DejaAJour, "");
        }
        finally { if (tampon != IntPtr.Zero) NetApiBufferFree(tampon); }

        var texte = Marshal.StringToHGlobalUni(nom);
        try
        {
            var info = new USER_INFO_1011 { usri1011_full_name = texte };
            var rc = NetUserSetInfo(null, compte, 1011, ref info, out _);
            return rc == 0
                ? (ResultatNom.Applique, $"Nom affiché du compte « {compte} » : « {nom} » (visible à la prochaine ouverture ou verrouillage de session).")
                : (ResultatNom.Erreur, $"Changement du nom affiché impossible (code {rc}).");
        }
        finally { Marshal.FreeHGlobal(texte); }
    }

    private static string Nettoyer(string nom)
    {
        var propre = new string((nom ?? "").Where(c => !char.IsControl(c)).ToArray()).Trim();
        return propre.Length > 100 ? propre[..100] : propre;
    }
}

using Microsoft.Extensions.Hosting;

/// <summary>
/// Garde le nom AFFICHÉ (FullName) du compte « Client » identique au nom du poste de l'interface web.
/// Fonctionne pour un compte créé par l'installateur comme pour un compte déjà existant.
/// Si le gérant renomme le poste, le nouveau nom est appliqué en moins de 30 s.
/// </summary>
public sealed class CompteWorker : BackgroundService
{
    private readonly GardienConfig _config;

    public CompteWorker(GardienConfig config) => _config = config;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        try { await Boucle(ct); }
        catch (OperationCanceledException) { }
        catch (Exception ex) { Journal.Ecrire($">>> ERREUR FATALE (nom du compte) : {ex}"); }
    }

    private async Task Boucle(CancellationToken ct)
    {
        await Task.Delay(5000, ct);
        string? dernierLog = null;

        while (!ct.IsCancellationRequested)
        {
            try
            {
                // Seulement une fois le poste enregistré : avant, NomPoste n'est que le nom Windows par défaut
                if (!string.IsNullOrEmpty(_config.AgentToken) && !string.IsNullOrWhiteSpace(_config.NomPoste))
                {
                    var (res, detail) = CompteWindows.DefinirNomAffiche(RestrictionsWindows.CompteClient, _config.NomPoste);
                    var ligne = $"{res}|{_config.NomPoste}|{detail}";
                    if (res != ResultatNom.DejaAJour && ligne != dernierLog)
                    {
                        Journal.Ecrire(detail);
                        dernierLog = ligne;
                    }
                }
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Nom du compte : erreur inattendue : {ex.Message}"); }

            await Task.Delay(20_000, ct);
        }
    }
}

using System.Net;
using System.Net.NetworkInformation;
using System.Net.Sockets;
using System.Text;
using System.Text.Json;

public sealed record Candidat(string Url, string ServeurId, string Nom);

/// <summary>
/// Chaîne de recherche du serveur (l'étape « dernière adresse connue » = ServerUrl, testée par DecouverteWorker) :
///   1. appel général UDP (broadcast) sur chaque carte réseau ;
///   2. balayage des adresses voisines (/24 de chaque carte), en secours.
/// Chaque candidat est vérifié par la signature Cyber Manager + l'identifiant du serveur déjà connu.
/// Aucune adresse n'est écrite en dur : tout vient du réseau local.
/// </summary>
public static class Decouverte
{
    public const int PortUdp = 47474;
    private const string Requete = "CYBER-MANAGER?1";
    private static readonly HttpClient Http = new() { Timeout = Timeout.InfiniteTimeSpan };

    public static async Task<Candidat?> ChercherAsync(GardienConfig config, CancellationToken ct)
    {
        foreach (var url in await BroadcastAsync(ct))
        {
            var c = await VerifierAsync(url, config.ServeurId, ct, 2000);
            if (c != null) { Journal.Ecrire($"Découverte : serveur trouvé par appel général ({c.Url})."); return c; }
        }

        Journal.Ecrire("Découverte : pas de réponse à l'appel général, balayage du réseau…");
        var b = await BalayerAsync(config, ct);
        if (b != null) Journal.Ecrire($"Découverte : serveur trouvé par balayage ({b.Url}).");
        return b;
    }

    /// <summary>Signature Cyber Manager + identifiant du serveur. Refuse un serveur dont l'identifiant diffère de celui déjà connu.</summary>
    public static async Task<Candidat?> VerifierAsync(string url, string serveurIdConnu, CancellationToken ct, int delaiMs = 1500)
    {
        try
        {
            if (!Uri.TryCreate(url, UriKind.Absolute, out var u) || u.Scheme is not ("http" or "https")) return null;

            using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
            cts.CancelAfter(delaiMs);
            using var rep = await Http.GetAsync(url.TrimEnd('/') + "/api/agent/decouverte", cts.Token);
            if (!rep.IsSuccessStatusCode) return null;

            var corps = await rep.Content.ReadAsStringAsync(cts.Token);
            if (corps.Length > 4096) return null;

            using var doc = JsonDocument.Parse(corps);
            var r = doc.RootElement;
            if (!r.TryGetProperty("cyber_manager", out var cm) || cm.ValueKind != JsonValueKind.True) return null;

            var id = r.TryGetProperty("serveur_id", out var i) && i.ValueKind == JsonValueKind.String ? i.GetString() ?? "" : "";
            if (id == "") return null;
            if (serveurIdConnu != "" && !string.Equals(id, serveurIdConnu, StringComparison.Ordinal)) return null;

            var nom = r.TryGetProperty("nom_cyber", out var n) && n.ValueKind == JsonValueKind.String ? n.GetString() ?? "" : "";
            return new Candidat(url.TrimEnd('/'), id, nom);
        }
        catch { return null; }
    }

    // ------------------------------------------------------------------

    private static async Task<List<string>> BroadcastAsync(CancellationToken ct)
    {
        var urls = new List<string>();
        var verrou = new object();
        var requete = Encoding.ASCII.GetBytes(Requete);

        var taches = Interfaces().Select(async itf =>
        {
            try
            {
                using var udp = new UdpClient(new IPEndPoint(itf.Ip, 0)) { EnableBroadcast = true };
                var cibles = new[]
                {
                    new IPEndPoint(AdresseBroadcast(itf.Ip, itf.Masque), PortUdp),
                    new IPEndPoint(IPAddress.Broadcast, PortUdp)
                };

                for (var essai = 0; essai < 2; essai++)   // UDP peut perdre un paquet : 2 envois
                {
                    foreach (var c in cibles) await udp.SendAsync(requete, requete.Length, c);

                    using var cts = CancellationTokenSource.CreateLinkedTokenSource(ct);
                    cts.CancelAfter(1300);
                    var erreurs = 0;
                    try
                    {
                        while (true)
                        {
                            UdpReceiveResult r;
                            try { r = await udp.ReceiveAsync(cts.Token); }
                            catch (SocketException) { if (++erreurs > 50) break; continue; }   // « port injoignable » d'un autre PC : on ignore

                            var port = LirePort(r.Buffer);
                            if (port <= 0) continue;
                            var url = "http://" + r.RemoteEndPoint.Address + (port == 80 ? "" : ":" + port);
                            lock (verrou) { if (!urls.Contains(url)) urls.Add(url); }
                        }
                    }
                    catch (OperationCanceledException) { }
                }
            }
            catch { }
        }).ToList();

        await Task.WhenAll(taches);
        return urls;
    }

    private static int LirePort(byte[] tampon)
    {
        try
        {
            using var doc = JsonDocument.Parse(tampon);
            var r = doc.RootElement;
            if (r.TryGetProperty("cyber_manager", out var cm) && cm.ValueKind == JsonValueKind.True
                && r.TryGetProperty("port", out var p) && p.TryGetInt32(out var n) && n is > 0 and < 65536)
                return n;
        }
        catch { }
        return 0;
    }

    private static async Task<Candidat?> BalayerAsync(GardienConfig config, CancellationToken ct)
    {
        var ports = new List<int>();
        if (Uri.TryCreate(config.ServerUrl, UriKind.Absolute, out var actuelle)) ports.Add(actuelle.Port);
        foreach (var p in new[] { 80, 8000, 8080 }) if (!ports.Contains(p)) ports.Add(p);

        var propres = new HashSet<string>();
        var reseaux = new HashSet<string>();
        foreach (var itf in Interfaces())
        {
            var b = itf.Ip.GetAddressBytes();
            propres.Add(itf.Ip.ToString());
            reseaux.Add($"{b[0]}.{b[1]}.{b[2]}");
        }

        var urls = new List<string>();
        foreach (var res in reseaux)
            for (var h = 1; h <= 254; h++)
            {
                var adresse = $"{res}.{h}";
                if (propres.Contains(adresse)) continue;
                foreach (var p in ports) urls.Add("http://" + adresse + (p == 80 ? "" : ":" + p));
            }

        using var stop = CancellationTokenSource.CreateLinkedTokenSource(ct);
        using var limite = new SemaphoreSlim(48);   // 48 tests en parallèle
        Candidat? gagnant = null;

        var taches = urls.Select(url => Task.Run(async () =>
        {
            try
            {
                await limite.WaitAsync(stop.Token);
                try
                {
                    if (stop.IsCancellationRequested) return;
                    var c = await VerifierAsync(url, config.ServeurId, stop.Token, 700);
                    if (c != null && Interlocked.CompareExchange(ref gagnant, c, null) == null) stop.Cancel();
                }
                finally { limite.Release(); }
            }
            catch (OperationCanceledException) { }
        })).ToList();

        await Task.WhenAll(taches);
        return gagnant;
    }

    // ------------------------------------------------------------------

    private static List<(IPAddress Ip, IPAddress Masque)> Interfaces()
    {
        var liste = new List<(IPAddress Ip, IPAddress Masque)>();
        try
        {
            foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
            {
                if (nic.OperationalStatus != OperationalStatus.Up) continue;
                if (nic.NetworkInterfaceType is NetworkInterfaceType.Loopback or NetworkInterfaceType.Tunnel) continue;

                foreach (var a in nic.GetIPProperties().UnicastAddresses)
                {
                    if (a.Address.AddressFamily != AddressFamily.InterNetwork || a.IPv4Mask == null) continue;
                    var b = a.Address.GetAddressBytes();
                    if (b[0] == 169 && b[1] == 254) continue;   // adresse automatique : pas de vrai réseau
                    liste.Add((a.Address, a.IPv4Mask));
                }
            }
        }
        catch { }
        return liste;
    }

    private static IPAddress AdresseBroadcast(IPAddress ip, IPAddress masque)
    {
        var a = ip.GetAddressBytes();
        var m = masque.GetAddressBytes();
        var r = new byte[4];
        for (var i = 0; i < 4; i++) r[i] = (byte)(a[i] | ~m[i]);
        return new IPAddress(r);
    }

/// <summary>
/// Surveille le serveur (toutes les 10 s). Après 3 échecs d'affilée (~30 s) : recherche automatique sur le réseau.
/// Si rien n'est trouvé, nouvelle recherche avec délai croissant (30 s, 1 min, 2 min, jusqu'à 5 min).
/// La dernière adresse trouvée est mémorisée dans gardien.json (ServerUrl) et reprise au démarrage suivant.
/// Les autres boucles du Gardien lisent ServerUrl à chaque requête : le changement est immédiat.
/// </summary>
public sealed class DecouverteWorker : BackgroundService
{
    private readonly GardienConfig _config;

    public DecouverteWorker(GardienConfig config) => _config = config;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        try { await Boucle(ct); }
        catch (OperationCanceledException) { }
        catch (Exception ex) { Journal.Ecrire($">>> ERREUR FATALE (découverte) : {ex}"); }
    }

    private async Task Boucle(CancellationToken ct)
    {
        await Task.Delay(5000, ct);

        var echecs = 0;
        var attente = 30;
        var prochaineRecherche = DateTime.MinValue;

        while (!ct.IsCancellationRequested)
        {
            try
            {
                var actuel = await Decouverte.VerifierAsync(_config.ServerUrl, _config.ServeurId, ct, 3000);

                if (actuel != null)
                {
                    echecs = 0;
                    attente = 30;
                    prochaineRecherche = DateTime.MinValue;

                    if (string.IsNullOrEmpty(_config.ServeurId))
                    {
                        _config.ServeurId = actuel.ServeurId;   // premier contact : on retient l'identité de CE serveur
                        _config.Save();
                        Journal.Ecrire("Identité du serveur mémorisée.");
                    }
                }
                else
                {
                    echecs++;
                    if (echecs >= 3 && DateTime.UtcNow >= prochaineRecherche)
                    {
                        Journal.Ecrire($"Serveur injoignable à {_config.ServerUrl} : recherche automatique sur le réseau…");
                        var trouve = await Decouverte.ChercherAsync(_config, ct);

                        if (trouve != null)
                        {
                            if (!string.Equals(trouve.Url, _config.ServerUrl, StringComparison.OrdinalIgnoreCase))
                                Journal.Ecrire($"Nouvelle adresse du serveur : {trouve.Url} (ancienne : {_config.ServerUrl}).");

                            _config.ServerUrl = trouve.Url;
                            if (string.IsNullOrEmpty(_config.ServeurId)) _config.ServeurId = trouve.ServeurId;
                            _config.Save();
                            echecs = 0;
                            attente = 30;
                        }
                        else
                        {
                            Journal.Ecrire($"Aucun serveur trouvé. Nouvelle recherche dans {attente} s.");
                            prochaineRecherche = DateTime.UtcNow.AddSeconds(attente);
                            attente = Math.Min(attente * 2, 300);
                        }
                    }
                }
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Découverte : erreur inattendue : {ex.Message}"); }

            await Task.Delay(10_000, ct);
        }
    }
}

using System.Net.NetworkInformation;
using System.Runtime.InteropServices;
using Microsoft.Win32;

public sealed record EmpreinteInfo(string Uuid, List<string> Macs);

/// <summary>
/// Empreinte matérielle stable : UUID SMBIOS + MAC des cartes PHYSIQUES (même débranchées).
/// Sans MAC locale ou virtuelle (VPN, Hyper-V, Wi-Fi Direct, MAC aléatoire). Ni adresse IP, ni nom de PC.
/// Un disque cloné sur un autre ordinateur donne une autre empreinte.
/// </summary>
public static class EmpreinteMaterielle
{
    private const uint SignatureRSMB = 0x52534D42;   // 'RSMB'
    private static readonly object Verrou = new();
    private static EmpreinteInfo? _cache;
    private static DateTime _luLe = DateTime.MinValue;

    // UUID « d'usine » que des cartes mères bon marché partagent : inutilisables pour reconnaître un PC
    private static readonly HashSet<string> UuidInvalides = new(StringComparer.OrdinalIgnoreCase)
    {
        "00000000-0000-0000-0000-000000000000", "FFFFFFFF-FFFF-FFFF-FFFF-FFFFFFFFFFFF",
        "03000200-0400-0500-0006-000700080009", "00020003-0004-0005-0006-000700080009",
        "12345678-1234-5678-90AB-CDDEEFAABBCC", "11111111-2222-3333-4444-555555555555",
    };

    [DllImport("kernel32.dll", SetLastError = true)]
    private static extern uint GetSystemFirmwareTable(uint fournisseur, uint idTable, byte[]? tampon, uint taille);

    public static EmpreinteInfo Lire()
    {
        lock (Verrou)
        {
            if (_cache == null || DateTime.UtcNow - _luLe > TimeSpan.FromMinutes(5))
            {
                _cache = new EmpreinteInfo(LireUuid(), LireMacs());
                _luLe = DateTime.UtcNow;
            }
            return _cache;
        }
    }

    private static string LireUuid()
    {
        try
        {
            var taille = GetSystemFirmwareTable(SignatureRSMB, 0, null, 0);
            if (taille < 16) return "";
            var t = new byte[taille];
            if (GetSystemFirmwareTable(SignatureRSMB, 0, t, taille) == 0) return "";

            var fin = Math.Min(t.Length, 8 + BitConverter.ToInt32(t, 4));
            var pos = 8;
            while (pos + 4 <= fin)
            {
                var type = t[pos];
                var longueur = t[pos + 1];
                if (longueur < 4 || type == 127) break;

                if (type == 1 && longueur >= 0x19 && pos + 24 <= fin)   // « System Information » : UUID à l'offset 8
                {
                    var octets = new byte[16];
                    Array.Copy(t, pos + 8, octets, 0, 16);
                    var uuid = new Guid(octets).ToString("D").ToUpperInvariant();
                    return UuidInvalides.Contains(uuid) ? "" : uuid;
                }

                var p = pos + longueur;   // saute la zone formatée puis les chaînes (fin = deux octets nuls)
                while (p + 1 < fin && !(t[p] == 0 && t[p + 1] == 0)) p++;
                pos = p + 2;
            }
        }
        catch { }
        return "";
    }

    private static List<string> LireMacs()
    {
        var pci = new SortedSet<string>();
        var usb = new SortedSet<string>();
        try
        {
            foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
            {
                if (nic.NetworkInterfaceType is not (NetworkInterfaceType.Ethernet or NetworkInterfaceType.GigabitEthernet
                    or NetworkInterfaceType.FastEthernetT or NetworkInterfaceType.FastEthernetFx or NetworkInterfaceType.Wireless80211))
                    continue;

                var o = nic.GetPhysicalAddress().GetAddressBytes();
                if (o.Length != 6 || (o[0] & 0x02) != 0 || o.All(b => b == 0)) continue;   // MAC locale / aléatoire / vide

                var mac = string.Join(":", o.Select(b => b.ToString("X2")));
                var pnp = IdMateriel(nic.Id);
                if (pnp.StartsWith(@"PCI\", StringComparison.OrdinalIgnoreCase)) pci.Add(mac);
                else if (pnp.StartsWith(@"USB\", StringComparison.OrdinalIgnoreCase)) usb.Add(mac);   // adaptateurs USB : en dernier recours
            }
        }
        catch { }
        return (pci.Count > 0 ? pci : usb).Take(8).ToList();
    }

    /// <summary>Identifiant matériel de la carte : PCI\… (physique), USB\… ; ROOT\…, SWD\… = virtuelle.</summary>
    private static string IdMateriel(string idCarte)
    {
        try
        {
            using var k = Registry.LocalMachine.OpenSubKey(
                $@"SYSTEM\CurrentControlSet\Control\Network\{{4D36E972-E325-11CE-BFC1-08002BE10318}}\{idCarte}\Connection");
            return k?.GetValue("PnpInstanceID") as string ?? "";
        }
        catch { return ""; }
    }
}

using Microsoft.Extensions.Hosting;

public sealed class GardienWorker : BackgroundService
{
    private const int DelaiArretSecondes = 5;
    private const string ArgsExtinction = "/s /t 0";   // ajouter /f pour forcer la fermeture des applications bloquantes

    private readonly GardienConfig _config;
    private readonly ServeurClient _api;
    private readonly CanalEcran _canal = new();
    private readonly LanceurEcran _lanceur = new();
    private readonly MiseAJour _maj;
    private readonly bool _console = Environment.UserInteractive;

    private CancellationToken _ct;
    private volatile bool _serveurOk;
    private string _etat = "serveur_injoignable";
    private bool _verrou = true;
    private bool _donneesFraiches;
    private SessionInfo? _session;
    private int _reveil;
    private long _arretDemandeTicks;
    private bool _brandingOccupe;

    public GardienWorker(GardienConfig config)
    {
        _config = config;
        _api = new ServeurClient(config, () => _maj?.VersionCourante() ?? Constantes.VersionEcran);
        _maj = new MiseAJour(config, _api, _lanceur, _canal, _console);
    }

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        _ct = ct;
        Journal.Ecrire($">>> Gardien {Constantes.VersionGardien} démarré — Écran {_maj.VersionCourante() ?? "non installé"} — mode {(_console ? "CONSOLE" : "SERVICE")}");
        Journal.Ecrire($"Config : poste='{_config.NomPoste}', serveur='{_config.ServerUrl}', id='{_config.IdentifiantUnique[..8]}…'");

        _canal.MessageRecu += SurMessageEcran;
        _canal.EcranPret += version =>
        {
            Journal.Ecrire($"Écran connecté (version {version}).");
            _maj.MarquerValidee(version);
            PousserBranding();
            PousserEtat();
        };

        var taches = new[]
        {
            Proteger(_canal.ExecuterAsync, ct),
            Proteger(BoucleEcranAsync, ct),
            Proteger(BoucleHeartbeatAsync, ct),
            Proteger(BoucleCommandesAsync, ct),
            Proteger(BoucleMiseAJourAsync, ct)
        };
        await Task.WhenAll(taches);
        Journal.Ecrire(">>> Gardien arrêté.");
    }

    private static async Task Proteger(Func<CancellationToken, Task> travail, CancellationToken ct)
    {
        try { await travail(ct); }
        catch (OperationCanceledException) { }
        catch (Exception ex) { Journal.Ecrire($">>> ERREUR FATALE dans une boucle : {ex}"); }
    }

    // ------------------------------------------------------------------
    // Supervision de l'Écran : il revient en 1-2 s s'il est fermé
    // ------------------------------------------------------------------

    private async Task BoucleEcranAsync(CancellationToken ct)
    {
        await _maj.ValiderAuDemarrageAsync(ct);

        while (!ct.IsCancellationRequested)
        {
            try
            {
                if (!_maj.EnCours && !_lanceur.EnCours)
                {
                    var version = _maj.VersionCourante();
                    if (version != null) _lanceur.Lancer(Chemins.ExeEcran(version), _console);
                }
            }
            catch (Exception ex) { Journal.Ecrire($"Supervision de l'Écran : {ex.Message}"); }

            await Task.Delay(1000, ct);
        }
    }

    // ------------------------------------------------------------------
    // Heartbeat + handshake (5 s, backoff si panne)
    // ------------------------------------------------------------------

    private static int Backoff(int echecs) => Math.Min(30_000, 2_000 << Math.Min(echecs, 4));   // 4s, 8s, 16s, 30s…

    private void DefinirServeur(bool ok)
    {
        if (ok != _serveurOk)
        {
            _serveurOk = ok;
            Journal.Ecrire(ok ? "Serveur joignable." : "Serveur injoignable — nouvelles tentatives avec délai croissant.");
        }
    }

    private async Task BoucleHeartbeatAsync(CancellationToken ct)
    {
        await Task.Delay(1000, ct);
        var echecs = 0;

        while (!ct.IsCancellationRequested)
        {
            var delai = 5_000;
            try
            {
                if (string.IsNullOrEmpty(_config.AgentToken))
                {
                    var hs = await _api.HandshakeAsync();
                    if (hs.Statut == ApiStatut.Ok)
                    {
                        Journal.Ecrire($"Handshake OK (poste '{_config.NomPoste}').");
                        echecs = 0;
                        delai = 300;
                    }
                    else if (hs.Code == "identifiant_a_regenerer")
                    {
                        Journal.Ecrire("Cet ordinateur est un doublon d'un autre poste : nouvelle identité générée, réenregistrement.");
                        _config.RegenererIdentifiant();
                        DefinirServeur(true);
                        echecs = 0;
                        delai = 300;
                    }
                    else if (hs.Code == "code_installation_invalide")
                    {
                        DefinirServeur(true);
                        _etat = "enregistrement_refuse";
                        _donneesFraiches = false;
                        echecs++;
                        delai = 30_000;
                        Journal.Ecrire("Enregistrement refusé : code d'installation absent ou invalide. Relancez : cyber-manager-gardien.exe --configurer --code=XXXX-XXXX (code affiché dans la page « Enregistrement » du serveur), puis redémarrez le service.");
                    }
                    else
                    {
                        echecs++;
                        if (hs.Statut == ApiStatut.Injoignable) DefinirServeur(false);
                        delai = Backoff(echecs);
                        Journal.Ecrire($"Handshake échoué ({hs.Statut}{(string.IsNullOrWhiteSpace(hs.Message) ? "" : " : " + hs.Message)}). Nouvel essai dans {delai / 1000}s.");
                    }
                }
                else
                {
                    var r = await _api.SendHeartbeatAsync();
                    switch (r.Statut)
                    {
                        case ApiStatut.Ok:
                            echecs = 0;
                            DefinirServeur(true);
                            _etat = r.Donnees!.Etat ?? "disponible";
                            _verrou = r.Donnees.DoitEtreVerrouille;
                            _session = r.Donnees.Session;
                            _donneesFraiches = true;

                            if (!string.IsNullOrWhiteSpace(r.Donnees.NomPoste) && r.Donnees.NomPoste != _config.NomPoste)
                            {
                                _config.NomPoste = r.Donnees.NomPoste;
                                _config.Save();
                            }
                            await AppliquerBrandingAsync(r.Donnees.Branding);
                            break;

                        case ApiStatut.NonAutorise:
                            Journal.Ecrire("Token refusé par le serveur : nouveau handshake.");
                            _config.AgentToken = "";
                            _config.Save();
                            delai = 200;
                            break;

                        default:   // injoignable ou erreur serveur : on garde l'état courant (mode dégradé)
                            echecs++;
                            DefinirServeur(false);
                            _donneesFraiches = false;
                            delai = Backoff(echecs);
                            break;
                    }
                }
                PousserEtat();
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Heartbeat : erreur inattendue : {ex.Message}"); }

            // Un réveil (demande de l'Écran) déclenche un heartbeat immédiat
            for (var attente = 0; attente < delai && Volatile.Read(ref _reveil) == 0; attente += 250)
                await Task.Delay(250, ct);
            Interlocked.Exchange(ref _reveil, 0);
        }
    }

    // ------------------------------------------------------------------
    // Commandes du serveur (3 s, backoff si panne)
    // ------------------------------------------------------------------

    private async Task BoucleCommandesAsync(CancellationToken ct)
    {
        await Task.Delay(3_000, ct);
        var echecs = 0;

        while (!ct.IsCancellationRequested)
        {
            var delai = 3_000;
            try
            {
                if (!string.IsNullOrEmpty(_config.AgentToken))
                {
                    var r = await _api.GetCommandsAsync();
                    if (r.Statut == ApiStatut.Ok)
                    {
                        echecs = 0;
                        foreach (var cmd in r.Donnees!)
                        {
                            Journal.Ecrire($"Commande reçue : {cmd.Type} (id={cmd.Id})");
                            var (ok, detail) = await ExecuterCommandeAsync(cmd);
                            await _api.ConfirmerCommandeAsync(cmd.Id, ok, detail);
                        }
                    }
                    else if (r.Statut != ApiStatut.NonAutorise)
                    {
                        echecs++;
                        delai = Backoff(echecs);
                    }
                }
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Commandes : erreur inattendue : {ex.Message}"); }

            await Task.Delay(delai, ct);
        }
    }

    private async Task<(bool ok, string detail)> ExecuterCommandeAsync(AgentCommand cmd)
    {
        try
        {
            switch (cmd.Type)
            {
                case "retour_arriere":
                    return _maj.RetourArriere();

                case "arreter":
                    Interlocked.Exchange(ref _arretDemandeTicks, DateTime.UtcNow.Ticks);
                    if (!_canal.Connecte)
                    {
                        // Pas d'Écran pour afficher le décompte : le Gardien éteint lui-même
                        _ = Task.Run(async () => { await Task.Delay(DelaiArretSecondes * 1000); Eteindre(); });
                        return (true, $"Extinction dans {DelaiArretSecondes} s (Écran absent)");
                    }
                    return await _canal.EnvoyerCommandeAsync(cmd.Id, cmd.Type, cmd.Payload, TimeSpan.FromSeconds(10));

                default:
                    return await _canal.EnvoyerCommandeAsync(cmd.Id, cmd.Type, cmd.Payload, TimeSpan.FromSeconds(10));
            }
        }
        catch (Exception ex)
        {
            Journal.Ecrire($"Exécution de '{cmd.Type}' échouée : {ex.Message}");
            return (false, "Erreur d'exécution");
        }
    }

    private void Eteindre()
    {
        try
        {
            Journal.Ecrire($"Extinction du poste : shutdown {ArgsExtinction}");
            System.Diagnostics.Process.Start(new System.Diagnostics.ProcessStartInfo("shutdown", ArgsExtinction)
            {
                CreateNoWindow = true,
                UseShellExecute = false
            });
        }
        catch (Exception ex) { Journal.Ecrire($"Extinction impossible : {ex.Message}"); }
    }

    // ------------------------------------------------------------------
    // Messages de l'Écran : clics du client, fin du décompte, réveil
    // ------------------------------------------------------------------

    private void SurMessageEcran(MessageCanal m)
    {
        switch (m.Type)
        {
            case "reveil":
                Volatile.Write(ref _reveil, 1);
                break;

            case "extinction":
                // On n'éteint que si le serveur l'a demandé dans la dernière minute
                var age = TimeSpan.FromTicks(DateTime.UtcNow.Ticks - Interlocked.Read(ref _arretDemandeTicks));
                if (age < TimeSpan.FromMinutes(1)) Eteindre();
                else Journal.Ecrire("Demande d'extinction de l'Écran ignorée (aucun ordre du serveur).");
                break;

            case "req":
                _ = Task.Run(async () => _canal.Envoyer(await TraiterRequeteAsync(m)));
                break;
        }
    }

    private async Task<MessageCanal> TraiterRequeteAsync(MessageCanal m)
    {
        var rep = new MessageCanal { Type = "rep", Id = m.Id };
        try
        {
            switch (m.Action)
            {
                case "activer":
                {
                    Journal.Ecrire("Activation demandée par le client…");
                    var r = await _api.ActiverAsync();
                    Journal.Ecrire(r.Succes
                        ? $"Activation OK{(r.Avertissement != null ? " (avertissement : " + r.Avertissement + ")" : "")}"
                        : $"Activation refusée : {r.Message}");
                    rep.Succes = r.Succes; rep.Message = r.Message; rep.Avertissement = r.Avertissement;
                    break;
                }
                case "suspendre":
                {
                    Journal.Ecrire("Suspension demandée…");
                    Remplir(rep, await _api.SuspendreSessionAsync());
                    break;
                }
                case "reprendre":
                {
                    Journal.Ecrire("Reprise demandée…");
                    Remplir(rep, await _api.ReprendreSessionAsync());
                    break;
                }
                case "fermer":
                {
                    Journal.Ecrire("Fermeture de session demandée…");
                    Remplir(rep, await _api.FermerSessionAsync());
                    break;
                }
                case "message":
                {
                    var texte = (m.Texte ?? "").Trim();
                    var r = texte.Length == 0 ? new ResultatAction(false, "Message vide.") : await _api.EnvoyerMessageAsync(texte);
                    Journal.Ecrire(r.Succes ? $"Message envoyé à l'accueil ({texte.Length} caractères)." : $"Message non envoyé : {r.Message}");
                    Remplir(rep, r);
                    break;
                }
                default:
                    rep.Succes = false; rep.Message = "Action inconnue.";
                    break;
            }
        }
        catch (Exception ex)
        {
            Journal.Ecrire($"Requête '{m.Action}' échouée : {ex.Message}");
            rep.Succes = false; rep.Message = "Erreur inattendue.";
        }
        return rep;
    }

    private async Task BoucleMiseAJourAsync(CancellationToken ct)
    {
        await Task.Delay(TimeSpan.FromSeconds(_console ? 10 : 30), ct);

        while (!ct.IsCancellationRequested)
        {
            try
            {
                if (!string.IsNullOrEmpty(_config.AgentToken) && _serveurOk && !_maj.EnCours)
                {
                    var courante = _maj.VersionCourante() ?? "";
                    var p = await _api.VerifierMiseAJourAsync(courante);

                    if (p != null && p.Version != courante && !_config.VersionsEchouees.Contains(p.Version) && PosteLibreEtVerrouille())
                    {
                        // Délai aléatoire : 20 postes ne téléchargent pas en même temps (CYBER_MAJ_DELAI=0 pour les tests)
                        var max = int.TryParse(Environment.GetEnvironmentVariable("CYBER_MAJ_DELAI"), out var d) ? d : 120;
                        Journal.Ecrire($"Mise à jour disponible : {courante} → {p.Version}.");
                        if (max > 0) await Task.Delay(TimeSpan.FromSeconds(Random.Shared.Next(0, max + 1)), ct);

                        if (PosteLibreEtVerrouille())
                        {
                            var (ok, detail) = await _maj.AppliquerAsync(p, ct);
                            await _api.SignalerResultatAsync(p.Version, ok, detail);
                        }
                    }
                }
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Mise à jour : erreur inattendue : {ex.Message}"); }

            await Task.Delay(60_000, ct);
        }
    }

using System.Diagnostics;
using System.Runtime.InteropServices;

public sealed class LanceurEcran
{
    private Process? _proc;
    private bool _aucuneSessionJournalisee;
    private bool _sessionAdminJournalisee;

    /// <summary>Test à la main (console) : pas de service, même utilisateur, aucune règle de session.</summary>
    private static readonly bool Interactif = Environment.UserInteractive;

    private const uint CREATE_UNICODE_ENVIRONMENT = 0x00000400;
    private const uint CREATE_NO_WINDOW = 0x08000000;
    private const uint MAXIMUM_ALLOWED = 0x02000000;
    private const int SecurityImpersonation = 2;
    private const int TokenPrimary = 1;

    private const string NomProcessus = "cyber-manager-agent";

    /// <summary>
    /// Un Écran légitime tourne-t-il ? Un Écran resté dans une session ADMINISTRATEUR de la console
    /// n'est pas légitime (Lancer() le retire). Un Écran d'une autre session (client en arrière-plan) l'est.
    /// </summary>
    public bool EnCours
    {
        get
        {
            try
            {
                if (_proc != null && !_proc.HasExited && Legitime(_proc)) return true;
                _proc = Process.GetProcessesByName(NomProcessus)
                    .FirstOrDefault(p => { try { return !p.HasExited && Legitime(p); } catch { return false; } });
                return _proc != null;
            }
            catch { return false; }
        }
    }

    private static bool Legitime(Process p)
    {
        if (Interactif) return true;
        return !(SessionConsoleEstAdmin(out var session) && p.SessionId == (int)session);
    }

    private static bool SessionConsoleEstAdmin(out uint session)
    {
        session = WTSGetActiveConsoleSessionId();
        if (session == 0xFFFFFFFF || !WTSQueryUserToken(session, out var jeton)) return false;
        try { return CompteWindows.JetonEstAdministrateur(jeton); }
        finally { CloseHandle(jeton); }
    }

    /// <summary>Un compte STANDARD (pas administrateur) a-t-il une session ouverte sur la console ?</summary>
    public static bool SessionDisponible()
    {
        var session = WTSGetActiveConsoleSessionId();
        if (session == 0xFFFFFFFF) return false;
        if (!WTSQueryUserToken(session, out var jeton)) return false;
        try { return !CompteWindows.JetonEstAdministrateur(jeton); }
        finally { CloseHandle(jeton); }
    }

    /// <summary>Lance l'Écran. modeConsole = test à la main (même utilisateur, pas de service).</summary>
    public bool Lancer(string exe, bool modeConsole)
    {
        if (EnCours) return true;
        if (!File.Exists(exe)) { Journal.Ecrire($"Écran introuvable : {exe}"); return false; }

        try
        {
            if (modeConsole)
            {
                _proc = Process.Start(new ProcessStartInfo(exe) { WorkingDirectory = Path.GetDirectoryName(exe)!, UseShellExecute = false });
                Journal.Ecrire("Écran lancé (mode console).");
                return _proc != null;
            }

            var session = WTSGetActiveConsoleSessionId();
            if (session == 0xFFFFFFFF || !WTSQueryUserToken(session, out var jetonUtilisateur))
            {
                if (!_aucuneSessionJournalisee) { Journal.Ecrire("Aucune session utilisateur ouverte : l'Écran sera lancé à l'ouverture de session."); _aucuneSessionJournalisee = true; }
                return false;
            }
            _aucuneSessionJournalisee = false;

            // L'Écran ne verrouille QUE les comptes standard : le gérant garde l'accès à sa session administrateur
            if (CompteWindows.JetonEstAdministrateur(jetonUtilisateur))
            {
                CloseHandle(jetonUtilisateur);
                ArreterDansSession(session);
                if (!_sessionAdminJournalisee)
                {
                    Journal.Ecrire("Session administrateur sur la console : l'Écran de verrouillage n'est pas lancé (seuls les comptes standard sont verrouillés).");
                    _sessionAdminJournalisee = true;
                }
                return false;
            }
            _sessionAdminJournalisee = false;

            IntPtr jetonPrimaire = IntPtr.Zero, environnement = IntPtr.Zero;
            try
            {
                if (!DuplicateTokenEx(jetonUtilisateur, MAXIMUM_ALLOWED, IntPtr.Zero, SecurityImpersonation, TokenPrimary, out jetonPrimaire))
                    throw new InvalidOperationException($"DuplicateTokenEx : erreur {Marshal.GetLastWin32Error()}");

                if (!CreateEnvironmentBlock(out environnement, jetonPrimaire, false))
                    environnement = IntPtr.Zero;

                var si = new STARTUPINFO { cb = Marshal.SizeOf<STARTUPINFO>(), lpDesktop = @"winsta0\default" };
                var ok = CreateProcessAsUser(jetonPrimaire, null, $"\"{exe}\"", IntPtr.Zero, IntPtr.Zero, false,
                    CREATE_UNICODE_ENVIRONMENT | CREATE_NO_WINDOW, environnement, Path.GetDirectoryName(exe), ref si, out var pi);
                if (!ok) throw new InvalidOperationException($"CreateProcessAsUser : erreur {Marshal.GetLastWin32Error()}");

                CloseHandle(pi.hThread);
                CloseHandle(pi.hProcess);
                _proc = Process.GetProcessById(pi.dwProcessId);
                Journal.Ecrire($"Écran lancé dans la session {session} (pid {pi.dwProcessId}).");
                return true;
            }
            finally
            {
                if (environnement != IntPtr.Zero) DestroyEnvironmentBlock(environnement);
                if (jetonPrimaire != IntPtr.Zero) CloseHandle(jetonPrimaire);
                CloseHandle(jetonUtilisateur);
            }
        }
        catch (Exception ex)
        {
            Journal.Ecrire($"Lancement de l'Écran impossible : {ex.Message}");
            return false;
        }
    }

    /// <summary>Retire l'Écran resté dans une session (devenue session administrateur).</summary>
    private void ArreterDansSession(uint session)
    {
        try
        {
            foreach (var p in Process.GetProcessesByName(NomProcessus))
            {
                try
                {
                    if (p.SessionId != (int)session) continue;
                    p.Kill(true);
                    p.WaitForExit(3000);
                    Journal.Ecrire($"Écran retiré de la session administrateur {session}.");
                }
                catch { }
            }
        }
        catch { }
        _proc = null;
    }

    /// <summary>Arrête l'Écran (et le WebView2 qui en dépend), quelle que soit sa version.</summary>
    public void Arreter()
    {
        try { _proc?.Kill(true); } catch { }
        try
        {
            foreach (var p in Process.GetProcessesByName(NomProcessus))
            {
                try { p.Kill(true); p.WaitForExit(3000); } catch { }
            }
        }
        catch { }
        _proc = null;
    }

    [StructLayout(LayoutKind.Sequential, CharSet = CharSet.Unicode)]
    private struct STARTUPINFO
    {
        public int cb;
        public string? lpReserved;
        public string? lpDesktop;
        public string? lpTitle;
        public int dwX, dwY, dwXSize, dwYSize, dwXCountChars, dwYCountChars, dwFillAttribute, dwFlags;
        public short wShowWindow, cbReserved2;
        public IntPtr lpReserved2, hStdInput, hStdOutput, hStdError;
    }

    [StructLayout(LayoutKind.Sequential)]
    private struct PROCESS_INFORMATION
    {
        public IntPtr hProcess, hThread;
        public int dwProcessId, dwThreadId;
    }

    [DllImport("kernel32.dll")] private static extern uint WTSGetActiveConsoleSessionId();
    [DllImport("wtsapi32.dll", SetLastError = true)] private static extern bool WTSQueryUserToken(uint sessionId, out IntPtr token);
    [DllImport("advapi32.dll", SetLastError = true)] private static extern bool DuplicateTokenEx(IntPtr existing, uint access, IntPtr attributs, int niveau, int type, out IntPtr nouveau);
    [DllImport("userenv.dll", SetLastError = true)] private static extern bool CreateEnvironmentBlock(out IntPtr env, IntPtr token, bool heriter);
    [DllImport("userenv.dll", SetLastError = true)] private static extern bool DestroyEnvironmentBlock(IntPtr env);

    private static extern bool CreateProcessAsUser(IntPtr token, string? application, string? ligneCommande, IntPtr attributsProcessus,
        IntPtr attributsThread, bool heriter, uint flags, IntPtr env, string? dossier, ref STARTUPINFO si, out PROCESS_INFORMATION pi);
    [DllImport("kernel32.dll", SetLastError = true)] private static extern bool CloseHandle(IntPtr h);
}

using System.IO.Compression;
using System.Security.Cryptography;

/// <summary>
/// Dossiers : Ecran\versions\&lt;version&gt;\ ; fichiers courante.txt / precedente.txt / validee.txt.
/// « validée » = cette version s'est déclarée prête après son démarrage.
/// </summary>
public sealed class MiseAJour
{
    private readonly GardienConfig _config;
    private readonly ServeurClient _api;
    private readonly LanceurEcran _lanceur;
    private readonly CanalEcran _canal;
    private readonly bool _console;

    /// <summary>Vrai pendant une mise à jour : le superviseur ne relance pas l'Écran à ce moment-là.</summary>
    public volatile bool EnCours;

    public MiseAJour(GardienConfig config, ServeurClient api, LanceurEcran lanceur, CanalEcran canal, bool console)
    {
        _config = config; _api = api; _lanceur = lanceur; _canal = canal; _console = console;
    }

    private static string? Lire(string chemin)
    {
        try
        {
            if (!File.Exists(chemin)) return null;
            var t = File.ReadAllText(chemin).Trim();
            return t.Length == 0 ? null : t;
        }
        catch { return null; }
    }

    private static void Ecrire(string chemin, string valeur)
    {
        Directory.CreateDirectory(Path.GetDirectoryName(chemin)!);
        File.WriteAllText(chemin, valeur);
    }

    public string? VersionCourante() => Lire(Chemins.Courante);
    private static string? Precedente() => Lire(Chemins.Precedente);
    private static string? Validee() => Lire(Chemins.Validee);

    /// <summary>L'Écran de cette version s'est connecté : elle est bonne.</summary>
    public void MarquerValidee(string version)
    {
        try { if (VersionCourante() == version && Validee() != version) Ecrire(Chemins.Validee, version); } catch { }
    }

    /// <summary>
    /// Au démarrage du service : si la dernière mise à jour n'avait pas été validée (coupure pendant l'opération),
    /// on la vérifie maintenant et on revient en arrière si elle ne démarre pas.
    /// </summary>
    public async Task ValiderAuDemarrageAsync(CancellationToken ct)
    {
        var courante = VersionCourante();
        if (courante == null) { Journal.Ecrire($"Aucune version de l'Écran installée dans {Chemins.RacineEcran}."); return; }
        if (courante == Validee()) return;

        Journal.Ecrire($"Version {courante} non validée : vérification au démarrage.");
        EnCours = true;
        try
        {
            var ok = await DemarrerEtVerifierAsync(courante, TimeSpan.FromMinutes(3), ct);
            if (!ok) await RevenirAsync(courante, "la version n'a pas démarré correctement au démarrage du poste");
        }
        finally { EnCours = false; }
    }

    public async Task<(bool ok, string detail)> AppliquerAsync(PropositionMaj p, CancellationToken ct)
    {
        EnCours = true;
        try
        {
            // 1. Téléchargement, vérification de l'empreinte, installation dans un nouveau dossier
            var dossier = Path.Combine(Chemins.Versions, p.Version);
            if (!File.Exists(Chemins.ExeEcran(p.Version)))
            {
                Directory.CreateDirectory(Chemins.Temp);
                var zip = Path.Combine(Chemins.Temp, $"ecran-{p.Version}.zip");
                Journal.Ecrire($"Mise à jour : téléchargement de la version {p.Version}…");

                if (!await _api.TelechargerPaquetAsync(p.Version, zip, ct))
                {
                    TrySupprimer(zip);
                    return (false, "Téléchargement impossible");
                }

                var empreinte = Sha256(zip);
                if (!string.Equals(empreinte, p.Sha256, StringComparison.OrdinalIgnoreCase))
                {
                    TrySupprimer(zip);
                    Journal.Ecrire("Mise à jour refusée : empreinte du paquet incorrecte.");
                    return (false, "Empreinte du paquet incorrecte");
                }

                var temporaire = dossier + ".tmp";
                if (Directory.Exists(temporaire)) Directory.Delete(temporaire, true);
                ZipFile.ExtractToDirectory(zip, temporaire);

                if (!File.Exists(Path.Combine(temporaire, "cyber-manager-agent.exe")))
                {
                    Directory.Delete(temporaire, true);
                    TrySupprimer(zip);
                    return (false, "Paquet incomplet (cyber-manager-agent.exe absent)");
                }

                if (Directory.Exists(dossier)) Directory.Delete(dossier, true);
                Directory.Move(temporaire, dossier);
                TrySupprimer(zip);
            }

            // 2. Bascule, démarrage contrôlé, retour arrière si échec
            var avant = VersionCourante();
            if (avant != null && avant != p.Version) Ecrire(Chemins.Precedente, avant);
            Ecrire(Chemins.Courante, p.Version);
            Journal.Ecrire($"Mise à jour : bascule {avant ?? "?"} → {p.Version}.");

            if (await DemarrerEtVerifierAsync(p.Version, TimeSpan.FromSeconds(45), ct))
            {
                Ecrire(Chemins.Validee, p.Version);
                Purger();
                Journal.Ecrire($"Mise à jour réussie : version {p.Version}.");
                return (true, "Version installée");
            }

            await RevenirAsync(p.Version, "la nouvelle version n'a pas démarré correctement");
            return (false, $"Retour arrière automatique vers {VersionCourante() ?? "?"} : la version {p.Version} n'a pas démarré");
        }
        catch (OperationCanceledException) { throw; }
        catch (Exception ex)
        {
            Journal.Ecrire($"Mise à jour échouée : {ex.Message}");
            try { if (VersionCourante() == p.Version && Validee() != p.Version) await RevenirAsync(p.Version, ex.Message); } catch { }
            return (false, ex.Message);
        }
        finally { EnCours = false; }
    }

    /// <summary>Retour arrière manuel (commande « retour_arriere » du serveur).</summary>
    public (bool ok, string detail) RetourArriere()
    {
        var precedente = Precedente();
        var courante = VersionCourante();
        if (precedente == null || !File.Exists(Chemins.ExeEcran(precedente)))
            return (false, "Aucune version précédente disponible");

        EnCours = true;
        try
        {
            if (courante != null)
            {
                Ecrire(Chemins.Precedente, courante);
                if (!_config.VersionsEchouees.Contains(courante)) { _config.VersionsEchouees.Add(courante); _config.Save(); }
            }
            Ecrire(Chemins.Courante, precedente);
            Ecrire(Chemins.Validee, precedente);
            _lanceur.Arreter();   // le superviseur relance la version précédente
            Journal.Ecrire($"Retour arrière manuel : {courante ?? "?"} → {precedente}.");
            return (true, $"Retour à la version {precedente}");
        }
        finally { EnCours = false; }
    }

    private async Task<bool> DemarrerEtVerifierAsync(string version, TimeSpan delai, CancellationToken ct)
    {
        var exe = Chemins.ExeEcran(version);
        _lanceur.Arreter();

        var attente = _canal.AttendrePretAsync(version, delai);   // enregistré AVANT le lancement
        var fin = DateTime.UtcNow + delai;

        while (!attente.IsCompleted && DateTime.UtcNow < fin)
        {
            if (!_lanceur.EnCours) _lanceur.Lancer(exe, _console);
            await Task.WhenAny(attente, Task.Delay(1000, ct));
        }

        if (attente.IsCompletedSuccessfully && attente.Result) return true;

        // Personne n'est connecté : impossible de conclure, on ne revient pas en arrière pour rien
        if (!_console && !LanceurEcran.SessionDisponible())
        {
            Journal.Ecrire("Aucune session ouverte : validation reportée à la prochaine ouverture de session.");
            return true;
        }
        return false;
    }

    private async Task RevenirAsync(string echouee, string raison)
    {
        if (!_config.VersionsEchouees.Contains(echouee)) { _config.VersionsEchouees.Add(echouee); _config.Save(); }

        var precedente = Precedente();
        Journal.Ecrire($"RETOUR ARRIÈRE : version {echouee} abandonnée ({raison}).");

        _lanceur.Arreter();
        if (precedente != null && File.Exists(Chemins.ExeEcran(precedente)))
        {
            Ecrire(Chemins.Courante, precedente);
            Ecrire(Chemins.Validee, precedente);
            Journal.Ecrire($"Version {precedente} remise en service.");
        }
        else Journal.Ecrire("Aucune version précédente : la version actuelle est conservée.");
        await Task.CompletedTask;
    }

    private void Purger()
    {
        try
        {
            var garder = new HashSet<string?>(StringComparer.OrdinalIgnoreCase) { VersionCourante(), Precedente() };
            if (!Directory.Exists(Chemins.Versions)) return;
            foreach (var d in Directory.GetDirectories(Chemins.Versions))
                if (!garder.Contains(Path.GetFileName(d))) { try { Directory.Delete(d, true); } catch { } }
        }
        catch { }
    }

    private static string Sha256(string fichier)
    {
        using var s = File.OpenRead(fichier);
        return Convert.ToHexString(SHA256.HashData(s)).ToLowerInvariant();
    }

    private static void TrySupprimer(string fichier) { try { File.Delete(fichier); } catch { } }
}

//    cyber-manager-gardien.exe --configurer --server=http://IP:PORT --code=XXXX-XXXX [--nom="POSTE 3"]

//  Lancement :  service « CyberManagerGardien »  — ou directement en console pour tester.
//  --server= est une adresse de DÉPART : si elle change, le Gardien retrouve seul le serveur (découverte).
// ============================================================================

var config = GardienConfig.LoadOrCreate();
var configurer = false;

foreach (var arg in args)
{
    if (arg.Equals("--configurer", StringComparison.OrdinalIgnoreCase)) configurer = true;
    else if (arg.StartsWith("--server=", StringComparison.OrdinalIgnoreCase))
    {
        var url = arg[9..].Trim().TrimEnd('/');
        if (Uri.TryCreate(url, UriKind.Absolute, out var u) && (u.Scheme is "http" or "https"))
        {
            config.ServerUrl = url;
            config.ServeurId = "";   // reconfiguration explicite : l'identité du serveur sera réapprise
        }
        else Console.WriteLine($"Option --server ignorée (URL invalide) : {url}");
    }
    else if (arg.StartsWith("--nom=", StringComparison.OrdinalIgnoreCase))
    {
        var nom = arg[6..].Trim();
        if (nom.Length is > 0 and <= 100) config.NomPoste = nom;
    }
    else if (arg.StartsWith("--code=", StringComparison.OrdinalIgnoreCase))
    {
        var code = arg[7..].Trim();
        if (code.Length is > 0 and <= 32) config.CodeInstallation = code;
        else Console.WriteLine("Option --code ignorée (longueur invalide).");
    }
    else if (arg.StartsWith("--importer=", StringComparison.OrdinalIgnoreCase))
    {
        try
        {
            using var doc = JsonDocument.Parse(File.ReadAllText(arg[11..].Trim('"')));
            var r = doc.RootElement;
            string? Lire(string c) => r.TryGetProperty(c, out var v) && v.ValueKind == JsonValueKind.String ? v.GetString() : null;

            if (!string.IsNullOrWhiteSpace(Lire("identifiant_unique"))) config.IdentifiantUnique = Lire("identifiant_unique")!;
            if (!string.IsNullOrWhiteSpace(Lire("agent_token"))) config.AgentToken = Lire("agent_token")!;
            if (!string.IsNullOrWhiteSpace(Lire("nom_poste"))) config.NomPoste = Lire("nom_poste")!;
            if (!string.IsNullOrWhiteSpace(Lire("server_url"))) config.ServerUrl = Lire("server_url")!;
            if (!string.IsNullOrWhiteSpace(Lire("code_installation"))) config.CodeInstallation = Lire("code_installation")!;
            Console.WriteLine("Identité de l'ancien agent importée.");
        }
        catch (Exception ex) { Console.WriteLine($"Import impossible : {ex.Message}"); }
    }
}

if (configurer)
{
    config.Save();
    Console.WriteLine($"Configuration enregistrée : poste='{config.NomPoste}', serveur='{config.ServerUrl}'.");
    Console.WriteLine($"Fichier : {Chemins.Config} (lisible uniquement par SYSTEM et les administrateurs).");
    return;
}

var builder = Host.CreateApplicationBuilder(args);
builder.Services.AddWindowsService(o => o.ServiceName = "CyberManagerGardien");
builder.Services.AddSingleton(config);
builder.Services.AddHostedService<GardienWorker>();
builder.Services.AddHostedService<RestrictionsWorker>();   // restrictions Windows du compte Client (étape 5)
builder.Services.AddHostedService<DecouverteWorker>();     // retrouve seul le serveur si son adresse change (étape 6)
builder.Services.AddHostedService<CompteWorker>();         // nom affiché du compte Client = nom du poste
builder.Logging.ClearProviders();   // notre journal (agent ProgramData\CyberManager\logs) suffit

await builder.Build().RunAsync();

using System.Diagnostics;
using System.Runtime.InteropServices;
using System.Security.Principal;
using System.Text.Json.Serialization;

/// <summary>Profil reçu du serveur. Par défaut (jamais reçu) : tout est bloqué, maintenance éteinte.</summary>
public sealed class RestrictionsInfo
{
    [JsonPropertyName("maintenance")] public bool Maintenance { get; set; }
    [JsonPropertyName("gestionnaire_taches")] public bool GestionnaireTaches { get; set; } = true;
    [JsonPropertyName("verrouiller")] public bool Verrouiller { get; set; } = true;
    [JsonPropertyName("deconnexion")] public bool Deconnexion { get; set; } = true;
    [JsonPropertyName("cmd")] public bool Cmd { get; set; } = true;
    [JsonPropertyName("powershell")] public bool PowerShell { get; set; } = true;
    [JsonPropertyName("regedit")] public bool Regedit { get; set; } = true;
    [JsonPropertyName("executer")] public bool Executer { get; set; } = true;
    [JsonPropertyName("panneau")] public bool Panneau { get; set; } = true;

    public string Empreinte() =>
        string.Concat(new[] { Maintenance, GestionnaireTaches, Verrouiller, Deconnexion, Cmd, PowerShell, Regedit, Executer, Panneau }
            .Select(b => b ? '1' : '0'));
}

public enum ResultatRestrictions { Applique, CompteAbsent, ProfilAbsent, Refuse, Erreur }

/// <summary>
/// Applique les restrictions dans le registre du compte « Client » UNIQUEMENT (ruche HKU\SID de ce compte).
/// Aucune écriture dans HKLM, aucune écriture dans la ruche d'un administrateur.
/// </summary>
public static class RestrictionsWindows
{
    public const string CompteClient = "Client";

    private const string PolSysteme = @"Software\Microsoft\Windows\CurrentVersion\Policies\System";
    private const string PolExplorateur = @"Software\Microsoft\Windows\CurrentVersion\Policies\Explorer";
    private const string PolCmd = @"Software\Policies\Microsoft\Windows\System";
    private static readonly string[] ProgrammesPowerShell = { "powershell.exe", "powershell_ise.exe", "pwsh.exe" };

    [DllImport("netapi32.dll", CharSet = CharSet.Unicode)]
    private static extern int NetLocalGroupGetMembers(string? serveur, string groupe, int niveau, out IntPtr tampon, int max, out int lus, out int total, IntPtr reprise);

    [DllImport("netapi32.dll")]
    private static extern int NetApiBufferFree(IntPtr tampon);

    public static (ResultatRestrictions, string) Appliquer(RestrictionsInfo r)
    {
        SecurityIdentifier sid;
        try { sid = (SecurityIdentifier)new NTAccount(Environment.MachineName, CompteClient).Translate(typeof(SecurityIdentifier)); }
        catch (IdentityNotMappedException) { return (ResultatRestrictions.CompteAbsent, $"Le compte « {CompteClient} » n'existe pas sur ce PC : aucune restriction appliquée."); }
        catch (Exception ex) { return (ResultatRestrictions.Erreur, $"Compte « {CompteClient} » introuvable : {ex.Message}"); }

        // GARDE-FOU : jamais un administrateur (en cas de doute, on refuse)
        if (EstAdministrateur(sid))
            return (ResultatRestrictions.Refuse, $"Le compte « {CompteClient} » est administrateur : restrictions REFUSÉES (le compte administrateur n'est jamais touché).");

        var profil = CheminProfil(sid);
        var ntuser = profil == null ? null : Path.Combine(profil, "NTUSER.DAT");
        if (ntuser == null || !File.Exists(ntuser))
            return (ResultatRestrictions.ProfilAbsent, $"Le compte « {CompteClient} » ne s'est encore jamais connecté (pas de profil) : nouvel essai plus tard.");

        var nomRuche = sid.Value;
        var chargeParNous = false;
        try
        {
            using var hu = RegistryKey.OpenBaseKey(RegistryHive.Users, RegistryView.Registry64);
            var racine = hu.OpenSubKey(nomRuche, true);   // présente si une session du Client est ouverte

            if (racine == null)
            {
                nomRuche = "CM_Client_Restrictions";
                if (Reg("load", @"HKU\" + nomRuche, ntuser) != 0)
                    return (ResultatRestrictions.Erreur, "Chargement du profil du Client impossible.");
                chargeParNous = true;
                racine = hu.OpenSubKey(nomRuche, true);
                if (racine == null) return (ResultatRestrictions.Erreur, "Ruche du Client illisible après chargement.");
            }

            try { Ecrire(racine, r); }
            finally { racine.Dispose(); }
        }
        catch (Exception ex) { return (ResultatRestrictions.Erreur, $"Écriture des restrictions impossible : {ex.Message}"); }
        finally
        {
            if (chargeParNous)
            {
                GC.Collect(); GC.WaitForPendingFinalizers();
                for (var i = 0; i < 4 && Reg("unload", @"HKU\" + nomRuche, null) != 0; i++) Thread.Sleep(500);
            }
        }

        return (ResultatRestrictions.Applique,
            r.Maintenance ? "MAINTENANCE : restrictions levées sur le compte Client." : "Restrictions appliquées sur le compte Client.");
    }

    // ------------------------------------------------------------------

    private static void Ecrire(RegistryKey racine, RestrictionsInfo r)
    {
        var bloquer = !r.Maintenance;   // maintenance = tout levé

        Valeur(racine, PolSysteme, "DisableTaskMgr", bloquer && r.GestionnaireTaches);
        Valeur(racine, PolSysteme, "DisableLockWorkstation", bloquer && r.Verrouiller);
        Valeur(racine, PolSysteme, "DisableRegistryTools", bloquer && r.Regedit);

        Valeur(racine, PolExplorateur, "NoLogoff", bloquer && r.Deconnexion);
        Valeur(racine, PolExplorateur, "NoRun", bloquer && r.Executer);
        Valeur(racine, PolExplorateur, "NoControlPanel", bloquer && r.Panneau);

        Valeur(racine, PolCmd, "DisableCMD", bloquer && r.Cmd, 2);   // 2 = invite bloquée, scripts .bat autorisés

        // PowerShell : liste « programmes interdits » de l'Explorateur
        var interdire = bloquer && r.PowerShell;
        Valeur(racine, PolExplorateur, "DisallowRun", interdire);
        if (interdire)
        {
            using var k = racine.CreateSubKey(PolExplorateur + @"\DisallowRun", true);
            for (var i = 0; i < ProgrammesPowerShell.Length; i++)
                k.SetValue((i + 1).ToString(), ProgrammesPowerShell[i], RegistryValueKind.String);
        }
        else
        {
            try { racine.DeleteSubKeyTree(PolExplorateur + @"\DisallowRun", false); } catch { }
        }
    }

    private static void Valeur(RegistryKey racine, string chemin, string nom, bool actif, int valeur = 1)
    {
        if (actif)
        {
            using var k = racine.CreateSubKey(chemin, true);
            k.SetValue(nom, valeur, RegistryValueKind.DWord);
        }
        else
        {
            using var k = racine.OpenSubKey(chemin, true);
            k?.DeleteValue(nom, false);
        }
    }

    private static string? CheminProfil(SecurityIdentifier sid)
    {
        try
        {
            using var k = Registry.LocalMachine.OpenSubKey(@"SOFTWARE\Microsoft\Windows NT\CurrentVersion\ProfileList\" + sid.Value);
            var p = k?.GetValue("ProfileImagePath") as string;
            return string.IsNullOrWhiteSpace(p) ? null : Environment.ExpandEnvironmentVariables(p);
        }
        catch { return null; }
    }

    /// <summary>Vrai si le compte est « Administrateur » (RID 500) ou membre du groupe Administrateurs. En cas de doute : vrai.</summary>
    private static bool EstAdministrateur(SecurityIdentifier sid)
    {
        if (sid.Value.EndsWith("-500", StringComparison.Ordinal)) return true;
        try
        {
            var admins = new SecurityIdentifier(WellKnownSidType.BuiltinAdministratorsSid, null);
            var nom = ((NTAccount)admins.Translate(typeof(NTAccount))).Value;   // nom localisé : « Administrateurs »
            nom = nom[(nom.IndexOf('\\') + 1)..];

            if (NetLocalGroupGetMembers(null, nom, 0, out var tampon, -1, out var lus, out _, IntPtr.Zero) != 0) return true;
            try
            {
                for (var i = 0; i < lus; i++)
                {
                    var psid = Marshal.ReadIntPtr(tampon, i * IntPtr.Size);   // LOCALGROUP_MEMBERS_INFO_0 = un pointeur vers un SID
                    if (new SecurityIdentifier(psid).Equals(sid)) return true;
                }
            }
            finally { NetApiBufferFree(tampon); }
            return false;
        }
        catch { return true; }
    }

    private static int Reg(string verbe, string cle, string? fichier)
    {
        var psi = new ProcessStartInfo("reg.exe") { UseShellExecute = false, CreateNoWindow = true, RedirectStandardOutput = true, RedirectStandardError = true };
        psi.ArgumentList.Add(verbe);
        psi.ArgumentList.Add(cle);
        if (fichier != null) psi.ArgumentList.Add(fichier);
        using var p = Process.Start(psi)!;
        p.StandardOutput.ReadToEnd(); p.StandardError.ReadToEnd();
        if (!p.WaitForExit(15000)) { try { p.Kill(); } catch { } return -1; }
        return p.ExitCode;
    }
}

using System.Net.Http;
using System.Text.Json;
using Microsoft.Extensions.Hosting;

/// <summary>
/// Lit le profil de restrictions + le drapeau maintenance du poste (toutes les 10 s), en garde une copie locale
/// (restrictions.json : le poste reste protégé serveur éteint) et l'applique sur le compte Client.
/// Application LOCALE par le Gardien (SYSTEM) : jamais de registre à distance.
/// </summary>
public sealed class RestrictionsWorker : BackgroundService
{
    private static string FichierLocal => Path.Combine(Chemins.Donnees, "restrictions.json");

    private readonly GardienConfig _config;

    public RestrictionsWorker(GardienConfig config) => _config = config;

    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        try { await Boucle(ct); }
        catch (OperationCanceledException) { }
        catch (Exception ex) { Journal.Ecrire($">>> ERREUR FATALE (restrictions) : {ex}"); }
    }

    private async Task Boucle(CancellationToken ct)
    {
        await Task.Delay(2000, ct);
        using var http = new HttpClient { Timeout = TimeSpan.FromSeconds(10) };

        var voulu = ChargerLocal() ?? new RestrictionsInfo();   // jamais reçu = tout bloqué
        string? empreinteAppliquee = null;
        var dernierEssai = DateTime.MinValue;
        var dernierResultat = ResultatRestrictions.Erreur;
        string? dernierLog = null;

        while (!ct.IsCancellationRequested)
        {
            try
            {
                var recu = await LireServeurAsync(http, ct);
                if (recu != null && recu.Empreinte() != voulu.Empreinte())
                {
                    voulu = recu;
                    SauverLocal(voulu);
                    Journal.Ecrire($"Restrictions : nouveau profil reçu (maintenance = {(voulu.Maintenance ? "OUI" : "non")}).");
                }

                var emp = voulu.Empreinte();
                var delaiRepeat = dernierResultat == ResultatRestrictions.Applique ? TimeSpan.FromMinutes(5) : TimeSpan.FromSeconds(30);
                if (emp != empreinteAppliquee || DateTime.UtcNow - dernierEssai > delaiRepeat)
                {
                    var (res, detail) = RestrictionsWindows.Appliquer(voulu);
                    dernierEssai = DateTime.UtcNow;
                    dernierResultat = res;
                    empreinteAppliquee = emp;

                    var ligne = $"{res}|{emp}|{detail}";
                    if (ligne != dernierLog) { Journal.Ecrire("Restrictions : " + detail); dernierLog = ligne; }
                }
            }
            catch (Exception ex) when (ex is not OperationCanceledException) { Journal.Ecrire($"Restrictions : erreur inattendue : {ex.Message}"); }

            await Task.Delay(10_000, ct);
        }
    }

    private async Task<RestrictionsInfo?> LireServeurAsync(HttpClient http, CancellationToken ct)
    {
        if (string.IsNullOrEmpty(_config.AgentToken)) return null;
        try
        {
            using var req = new HttpRequestMessage(HttpMethod.Get, _config.ServerUrl.TrimEnd('/') + "/api/agent/restrictions");
            req.Headers.Add("X-Agent-Token", _config.AgentToken);
            req.Headers.Add("Accept", "application/json");
            using var rep = await http.SendAsync(req, ct);
            if (!rep.IsSuccessStatusCode) return null;
            return JsonSerializer.Deserialize<RestrictionsInfo>(await rep.Content.ReadAsStringAsync(ct));
        }
        catch (Exception ex) when (ex is not OperationCanceledException) { return null; }   // serveur injoignable : on garde la copie locale
    }

    private static RestrictionsInfo? ChargerLocal()
    {
        try { return File.Exists(FichierLocal) ? JsonSerializer.Deserialize<RestrictionsInfo>(File.ReadAllText(FichierLocal)) : null; }
        catch { return null; }
    }

    private static void SauverLocal(RestrictionsInfo r)
    {
        try
        {
            Directory.CreateDirectory(Chemins.Donnees);
            File.WriteAllText(FichierLocal, JsonSerializer.Serialize(r));
        }
        catch (Exception ex) { Journal.Ecrire($"Copie locale des restrictions impossible : {ex.Message}"); }
    }
}

    {
        var e = EmpreinteMaterielle.Lire();

    public async Task<HandshakeResultat> HandshakeAsync()
    {
        var net = Reseau;
        var payload = new
        {
            identifiant_unique = _config.IdentifiantUnique,
            nom_poste = _config.NomPoste,
            nom_windows = Environment.MachineName,
            adresse_ip = net.Ip,
            adresse_mac = net.Mac,
            agent_version = _versionAgent(),
            code_installation = _config.CodeInstallation,
            empreinte = Empreinte()
        };

        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, "/api/agent/handshake", payload, avecToken: false);
        if (statut != ApiStatut.Ok)
            return new HandshakeResultat(statut, LireChamp(corps, "code"), LireChamp(corps, "message"));

        try
        {
            using var doc = JsonDocument.Parse(corps!);
            var root = doc.RootElement;
            var token = root.TryGetProperty("agent_token", out var t) ? t.GetString() : null;
            var nom = root.TryGetProperty("nom_poste", out var n) ? n.GetString() : null;

            if (string.IsNullOrEmpty(token)) return new HandshakeResultat(ApiStatut.Erreur);

            var modifie = token != _config.AgentToken;
            _config.AgentToken = token;
            if (!string.IsNullOrWhiteSpace(nom) && nom != _config.NomPoste)
            {
                _config.NomPoste = nom;
                modifie = true;
            }
            if (modifie) _config.Save();
            return new HandshakeResultat(ApiStatut.Ok);
        }
        catch { return new HandshakeResultat(ApiStatut.Erreur); }
    }

    public async Task<ApiReponse<HeartbeatResponse>> SendHeartbeatAsync()
    {
        var (statut, corps) = await EnvoyerAsync(HttpMethod.Post, "/api/agent/heartbeat",

        // Clone détecté (autre empreinte alors que l'original répond) : nouvelle identité, nouveau handshake

            _config.RegenererIdentifiant();
            return new(ApiStatut.NonAutorise);
        }

        if (statut != ApiStatut.Ok) return new(statut);

        try
        {
            var hb = JsonSerializer.Deserialize<HeartbeatResponse>(corps!);
            return hb is { Ok: true } ? new(ApiStatut.Ok, hb) : new(ApiStatut.Erreur);
        }
        catch { return new(ApiStatut.Erreur); }
    }

    private static InfosReseau LireReseau(string serverUrl)
    {
        string ip = "", mac = "";
        try
        {
            if (Uri.TryCreate(serverUrl, UriKind.Absolute, out var uri))
            {
                using var s = new Socket(AddressFamily.InterNetwork, SocketType.Dgram, 0);
                s.Connect(uri.Host, uri.Port);
                var local = (s.LocalEndPoint as IPEndPoint)?.Address;
                if (local != null)
                {
                    ip = local.ToString();
                    foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
                    {
                        if (nic.OperationalStatus != OperationalStatus.Up) continue;
                        if (!nic.GetIPProperties().UnicastAddresses.Any(a => a.Address.Equals(local))) continue;
                        mac = FormaterMac(nic.GetPhysicalAddress().ToString());
                        if (mac != "") break;
                    }
                }
            }
        }
        catch { }

        if (mac == "")
        {
            try
            {
                foreach (var nic in NetworkInterface.GetAllNetworkInterfaces())
                {
                    if (nic.OperationalStatus != OperationalStatus.Up) continue;
                    if (nic.NetworkInterfaceType == NetworkInterfaceType.Loopback) continue;
                    mac = FormaterMac(nic.GetPhysicalAddress().ToString());
                    if (mac != "") break;
                }
            }
            catch { }
        }

        return new InfosReseau(ip, mac);
    }

    private static string FormaterMac(string brut) =>
        brut.Length == 12
            ? string.Join(":", Enumerable.Range(0, 6).Select(i => brut.Substring(i * 2, 2))).ToUpperInvariant()
            : "";
}

/// L'Écran ne contient AUCUN secret : le jeton et l'adresse du serveur sont chez le Gardien (ProgramData).
/// Cette classe ne donne plus que le dossier de travail de l'utilisateur (journal, données WebView2).

    {
        get
        {
            var dir = Path.Combine(
                Environment.GetFolderPath(Environment.SpecialFolder.LocalApplicationData),
                "CyberManagerAgent");
            Directory.CreateDirectory(dir);
            return Path.Combine(dir, "ecran.dat");
        }
    }

    public static BrandingData BrandingParDefaut => new("Cyber Manager", "Gestion de cybercafé", "#7c3aed", null);

    /// <summary>Convertit le message « branding » du Gardien (logo en base64) en données d'affichage.</summary>
    public static BrandingData DepuisMessage(MessageCanal m)
    {
        byte[]? logo = null;
        try { if (!string.IsNullOrEmpty(m.Logo)) logo = Convert.FromBase64String(m.Logo); } catch { }

        return new BrandingData(
            string.IsNullOrWhiteSpace(m.Nom) ? "Cyber Manager" : m.Nom,
            m.Slogan ?? "",
            string.IsNullOrWhiteSpace(m.Couleur) ? "#7c3aed" : m.Couleur,
            logo);
    }
}

using System.Collections.Concurrent;
using System.IO.Pipes;
using System.Text;

/// <summary>
/// Client du Gardien (canal local). Remplace l'ancien client HTTP : le serveur et le jeton
/// ne sont connus que du Gardien. Se reconnecte tout seul si le Gardien redémarre.
/// </summary>
public class ApiClient
{
    public const string Version = Constantes.VersionEcran;

    private readonly Action<string> _journal;
    private readonly object _verrou = new();
    private StreamWriter? _ecrivain;
    private readonly ConcurrentDictionary<int, TaskCompletionSource<MessageCanal>> _attentes = new();
    private int _prochainId;

    /// <summary>Message du Gardien : etat, branding ou cmd.</summary>
    public event Action<MessageCanal>? MessageRecu;
    public event Action? Connecte;
    public event Action? Deconnecte;

    public ApiClient(Action<string> journal) { _journal = journal; }

    public async Task BoucleAsync()
    {
        while (true)
        {
            var etaitConnecte = false;
            try
            {
                using var pipe = new NamedPipeClientStream(".", Constantes.NomPipe, PipeDirection.InOut, PipeOptions.Asynchronous);
                await pipe.ConnectAsync(3000);

                var utf8 = new UTF8Encoding(false);
                using var lecteur = new StreamReader(pipe, utf8);
                var ecrivain = new StreamWriter(pipe, utf8) { AutoFlush = true };
                lock (_verrou) _ecrivain = ecrivain;
                etaitConnecte = true;

                _journal("Connecté au Gardien.");
                Envoyer(new MessageCanal { Type = "pret", Texte = Version });
                try { Connecte?.Invoke(); } catch { }

                string? ligne;
                while ((ligne = await lecteur.ReadLineAsync()) != null) Traiter(ligne);
            }
            catch { /* Gardien absent ou redémarrage : on réessaie */ }
            finally
            {
                lock (_verrou) _ecrivain = null;
                foreach (var a in _attentes.Values)
                    a.TrySetResult(new MessageCanal { Succes = false, Message = "Liaison avec le Gardien perdue." });
                if (etaitConnecte)
                {
                    _journal("Liaison avec le Gardien perdue.");
                    try { Deconnecte?.Invoke(); } catch { }
                }
            }

            await Task.Delay(1000);
        }
    }

    private void Traiter(string ligne)
    {
        var m = ProtocoleCanal.Decoder(ligne);
        if (m == null) return;

        if (m.Type == "rep")
        {
            if (_attentes.TryRemove(m.Id, out var a)) a.TrySetResult(m);
            return;
        }
        try { MessageRecu?.Invoke(m); } catch (Exception ex) { _journal($"Message du Gardien : {ex.Message}"); }
    }

    public bool Envoyer(MessageCanal message)
    {
        try
        {
            lock (_verrou)
            {
                if (_ecrivain == null) return false;
                _ecrivain.WriteLine(ProtocoleCanal.Encoder(message));
                return true;
            }
        }
        catch { return false; }
    }

    private async Task<MessageCanal> RequeteAsync(string action, string? texte = null)
    {
        var id = Interlocked.Increment(ref _prochainId);
        var tcs = new TaskCompletionSource<MessageCanal>(TaskCreationOptions.RunContinuationsAsynchronously);
        _attentes[id] = tcs;

        if (!Envoyer(new MessageCanal { Type = "req", Id = id, Action = action, Texte = texte }))
        {
            _attentes.TryRemove(id, out _);
            return new MessageCanal { Succes = false, Message = "Le Gardien n'est pas joignable. Réessayez dans quelques instants." };
        }

        var fini = await Task.WhenAny(tcs.Task, Task.Delay(TimeSpan.FromSeconds(75)));
        _attentes.TryRemove(id, out _);
        return fini == tcs.Task ? tcs.Task.Result : new MessageCanal { Succes = false, Message = "Le Gardien ne répond pas." };
    }

    public async Task<ActivationResult> ActiverAsync()
    {
        var r = await RequeteAsync("activer");
        return new ActivationResult(r.Succes == true, r.Message ?? (r.Succes == true ? "Poste activé." : "Activation refusée."), r.Avertissement);
    }

    public Task<ResultatAction> SuspendreSessionAsync() => ActionAsync("suspendre", "Session suspendue.");
    public Task<ResultatAction> FermerSessionAsync() => ActionAsync("fermer", "Session terminée.");
    public Task<ResultatAction> ReprendreSessionAsync() => ActionAsync("reprendre", "Session reprise.");
    public Task<ResultatAction> EnvoyerMessageAsync(string texte) => ActionAsync("message", "Message envoyé.", texte);

    private async Task<ResultatAction> ActionAsync(string action, string messageParDefaut, string? texte = null)
    {
        var r = await RequeteAsync(action, texte);
        return new ResultatAction(r.Succes == true, r.Message ?? (r.Succes == true ? messageParDefaut : "Action refusée."), r.Montant);
    }
}

    private async void OnWebMessage(object? sender, CoreWebView2WebMessageReceivedEventArgs e)
    {
        string message;
        try { message = e.TryGetWebMessageAsString(); } catch { return; }

        if (message == "activer") { await ActiverAsync(); return; }
        if (message == "annuler_arret") { AnnulerArret(); return; }
        if (message.StartsWith("msg:", StringComparison.Ordinal)) await EnvoyerDepuisVerrouAsync(message[4..]);
        
        if (message == "reprendre" && SuspendreHandler != null) { await SuspendreHandler(); return; }
        if (message == "quitter" && QuitterHandler != null)
        {
            var r = await QuitterHandler();
            if (r.Succes && r.Montant is > 0) MessageBox.Show($"Montant à payer : {SessionWidget.FormaterAr(r.Montant.Value)}", "Session terminée");
            return;

    }

    private IntPtr HookCallback(int nCode, IntPtr wParam, IntPtr lParam)
    {
        if (nCode >= 0 && _locked)
        {
            var msg = (int)wParam;
            if (msg is WM_KEYDOWN or WM_KEYUP or WM_SYSKEYDOWN or WM_SYSKEYUP)
            {
                var vk = Marshal.ReadInt32(lParam);
                var flags = Marshal.ReadInt32(lParam, 8);
                var alt = (flags & 0x20) != 0;
                var ctrl = (GetAsyncKeyState(0x11) & 0x8000) != 0;

                var bloquer = vk is 0x5B or 0x5C or 0x5D || (alt && vk is (0x09 or 0x1B or 0x73 or 0x20)) || (ctrl && vk == 0x1B);
                if (bloquer) return (IntPtr)1;
            }
        }
        return CallNextHookEx(IntPtr.Zero, nCode, wParam, lParam);
    }

    gardien.Connecte += () => { /* le Gardien répond en poussant branding + état */ };

    gardien.Deconnecte += () =>
    {
        etatServeur = liaison.Etat == EtatLiaison.CableDebranche ? "cable_debranche" : "serveur_injoignable";
        PousserInfos();
    };

    gardien.MessageRecu += m =>
    {
        switch (m.Type)
        {
            case "etat":
                poste = m.Poste ?? poste;
                ip = m.Ip ?? ip;
                etatServeur = m.Etat ?? "serveur_injoignable";
                if (etatServeur == "serveur_injoignable" && liaison.Etat == EtatLiaison.CableDebranche)
                    etatServeur = "cable_debranche";   // on distingue « câble débranché » de « serveur injoignable »
                PousserInfos();
                // Verrou absent = serveur injoignable : on garde l'état courant (mode dégradé)
                if (m.Verrou is bool doitEtreVerrouille)
                {
                    liaison.SignalerSucces();   // donnée fraîche du serveur : la liaison est bonne
                    fenetreUi.MettreAJourSession(m.Session);
                    fenetreUi.SyncFromServer(doitEtreVerrouille);
                }
                break;

            case "branding":
                fenetreUi.MettreAJourBranding(AgentConfig.DepuisMessage(m));
                break;

            case "cmd":
            {
                var (ok, detail) = Executer(m.Commande ?? "", m.Payload, fenetreUi);
                gardien.Envoyer(new MessageCanal { Type = "repcmd", Id = m.Id, Succes = ok, Message = detail });
                break;
            }
        }
    };

    _ = Task.Run(gardien.BoucleAsync);

    // Exécution d'une commande reçue du serveur (via le Gardien)
    (bool ok, string detail) Executer(string type, string? payload, ClientWindow fenetre)
    {
        Log($"Commande reçue : {type}");
        try
        {
            switch (type)
            {
                case "demarrer":
                case "debloquer":
                case "reprendre":
                    fenetre.UnlockScreen();
                    return (true, "Poste déverrouillé");

                case "verrouiller":
                case "suspendre":
                case "fermer":
                    fenetre.LockScreen();
                    return (true, "Poste verrouillé");

                case "desactiver":
                    // L'écran affiche tout de suite « Poste désactivé » (ACTIVER devient INDISPONIBLE)
                    etatServeur = "desactive";
                    PousserInfos();
                    fenetre.LockScreen();   // ferme aussi le widget s'il est ouvert
                    Reveiller();
                    return (true, "Poste désactivé et verrouillé");

                case "arreter":
                    fenetre.DemanderArret(DelaiArretSecondes);
                    return (true, $"Extinction dans {DelaiArretSecondes} s");

                case "recharger":
                {
                    var minutes = LireEntier(payload, "minutes_ajoutees");
                    fenetre.NotifierRecharge(minutes);
                    Reveiller();   // le widget se recale tout de suite sur le temps du serveur
                    return (true, "Temps ajouté");
                }

                case "message":
                {
                    var texte = LireTexte(payload, "message");
                    if (string.IsNullOrWhiteSpace(texte)) return (false, "Message vide");
                    fenetre.RecevoirMessage(texte.Trim());
                    return (true, "Message affiché");
                }

                default:
                    return (false, $"Commande inconnue : {type}");
            }
        }
        catch (Exception ex)
        {
            Log($"Exécution de '{type}' échouée : {ex.Message}");
            return (false, "Erreur d'exécution");
        }
    }

    uiThread.Join();
    Log(">>> Application terminée.");
}
catch (Exception ex)
{
    Log($">>> ERREUR FATALE : {ex}");
}

using System.Collections.Concurrent;
using System.IO.Pipes;
using System.Text;

/// <summary>
/// Client du Gardien (canal local). Remplace l'ancien client HTTP : le serveur et le jeton
/// ne sont connus que du Gardien. Se reconnecte tout seul si le Gardien redémarre.
/// </summary>
public class ApiClient
{
    public const string Version = Constantes.VersionEcran;

    private readonly Action<string> _journal;
    private readonly object _verrou = new();
    private StreamWriter? _ecrivain;
    private readonly ConcurrentDictionary<int, TaskCompletionSource<MessageCanal>> _attentes = new();
    private int _prochainId;

    /// <summary>Message du Gardien : etat, branding ou cmd.</summary>
    public event Action<MessageCanal>? MessageRecu;
    public event Action? Connecte;
    public event Action? Deconnecte;

    public ApiClient(Action<string> journal) { _journal = journal; }

    public async Task BoucleAsync()
    {
        while (true)
        {
            var etaitConnecte = false;
            try
            {
                using var pipe = new NamedPipeClientStream(".", Constantes.NomPipe, PipeDirection.InOut, PipeOptions.Asynchronous);
                await pipe.ConnectAsync(3000);

                var utf8 = new UTF8Encoding(false);
                using var lecteur = new StreamReader(pipe, utf8);
                var ecrivain = new StreamWriter(pipe, utf8) { AutoFlush = true };
                lock (_verrou) _ecrivain = ecrivain;
                etaitConnecte = true;

                _journal("Connecté au Gardien.");
                Envoyer(new MessageCanal { Type = "pret", Texte = Version });
                try { Connecte?.Invoke(); } catch { }

                string? ligne;
                while ((ligne = await lecteur.ReadLineAsync()) != null) Traiter(ligne);
            }
            catch { /* Gardien absent ou redémarrage : on réessaie */ }
            finally
            {
                lock (_verrou) _ecrivain = null;
                foreach (var a in _attentes.Values)
                    a.TrySetResult(new MessageCanal { Succes = false, Message = "Liaison avec le Gardien perdue." });
                if (etaitConnecte)
                {
                    _journal("Liaison avec le Gardien perdue.");
                    try { Deconnecte?.Invoke(); } catch { }
                }
            }

            await Task.Delay(1000);
        }
    }

    private void Traiter(string ligne)
    {
        var m = ProtocoleCanal.Decoder(ligne);
        if (m == null) return;

        if (m.Type == "rep")
        {
            if (_attentes.TryRemove(m.Id, out var a)) a.TrySetResult(m);
            return;
        }
        try { MessageRecu?.Invoke(m); } catch (Exception ex) { _journal($"Message du Gardien : {ex.Message}"); }
    }

    public bool Envoyer(MessageCanal message)
    {
        try
        {
            lock (_verrou)
            {
                if (_ecrivain == null) return false;
                _ecrivain.WriteLine(ProtocoleCanal.Encoder(message));
                return true;
            }
        }
        catch { return false; }
    }

    private async Task<MessageCanal> RequeteAsync(string action, string? texte = null)
    {
        var id = Interlocked.Increment(ref _prochainId);
        var tcs = new TaskCompletionSource<MessageCanal>(TaskCreationOptions.RunContinuationsAsynchronously);
        _attentes[id] = tcs;

        if (!Envoyer(new MessageCanal { Type = "req", Id = id, Action = action, Texte = texte }))
        {
            _attentes.TryRemove(id, out _);
            return new MessageCanal { Succes = false, Message = "Le Gardien n'est pas joignable. Réessayez dans quelques instants." };
        }

        var fini = await Task.WhenAny(tcs.Task, Task.Delay(TimeSpan.FromSeconds(75)));
        _attentes.TryRemove(id, out _);
        return fini == tcs.Task ? tcs.Task.Result : new MessageCanal { Succes = false, Message = "Le Gardien ne répond pas." };
    }

    public async Task<ActivationResult> ActiverAsync()
    {
        var r = await RequeteAsync("activer");
        return new ActivationResult(r.Succes == true, r.Message ?? (r.Succes == true ? "Poste activé." : "Activation refusée."), r.Avertissement);
    }

    public Task<ResultatAction> SuspendreSessionAsync() => ActionAsync("suspendre", "Session suspendue.");
    public Task<ResultatAction> FermerSessionAsync() => ActionAsync("fermer", "Session terminée.");
    public Task<ResultatAction> ReprendreSessionAsync() => ActionAsync("reprendre", "Session reprise.");
    public Task<ResultatAction> EnvoyerMessageAsync(string texte) => ActionAsync("message", "Message envoyé.", texte);

    private async Task<ResultatAction> ActionAsync(string action, string messageParDefaut, string? texte = null)
    {
        var r = await RequeteAsync(action, texte);
        return new ResultatAction(r.Succes == true, r.Message ?? (r.Succes == true ? messageParDefaut : "Action refusée."), r.Montant);
    }
}

    var ligne = $"[{DateTime.Now:yyyy-MM-dd HH:mm:ss}] {msg}";
    Console.WriteLine(ligne);
    lock (logVerrou)
    {
        try { File.AppendAllText(logPath, ligne + Environment.NewLine); } catch { }
    }
}

// Lecture d'un champ du payload JSON d'une commande (jamais journalisé)
static string? LireTexte(string? json, string champ)
{
    if (string.IsNullOrWhiteSpace(json)) return null;
    try
    {
        using var doc = System.Text.Json.JsonDocument.Parse(json);
        return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == System.Text.Json.JsonValueKind.String
            ? v.GetString() : null;
    }
    catch { return null; }
}

static int? LireEntier(string? json, string champ)
{
    if (string.IsNullOrWhiteSpace(json)) return null;
    try
    {
        using var doc = System.Text.Json.JsonDocument.Parse(json);
        return doc.RootElement.TryGetProperty(champ, out var v) && v.ValueKind == System.Text.Json.JsonValueKind.Number
            ? v.GetInt32() : null;
    }
    catch { return null; }
}

    var ip = "";
    var etatServeur = "serveur_injoignable";

    void PousserInfos() => fenetreUi.MettreAJourInfos(poste, ip, etatServeur);

    // --- Surveillance de la liaison avec le serveur ----------------------------
    // Grâce de 15 s (< tolérance du serveur : Poste::DELAI_HORS_LIGNE = 30 s). Au-delà : poste verrouillé,
    // usage gratuit impossible. Reprise automatique : le prochain état frais du Gardien déverrouille.
    using var liaison = new SurveillanceLiaison();
    liaison.EtatChange += (etat, message) =>
    {
        Log($"Liaison avec le serveur : {etat} — {message}");
        switch (etat)
        {
            case EtatLiaison.CableDebranche:
                etatServeur = "cable_debranche";
                PousserInfos();
                fenetreUi.LockScreen();
                break;
            case EtatLiaison.ServeurInjoignable:
                etatServeur = "serveur_injoignable";
                PousserInfos();
                fenetreUi.LockScreen();
                break;
        }
    };
    liaison.Demarrer();

    // Heartbeat immédiat demandé (recharge, désactivation, fin de forfait…)
    void Reveiller() => gardien.Envoyer(new MessageCanal { Type = "reveil" });
    fenetreUi.ReveillerHeartbeat = Reveiller;

    // Clic sur ACTIVER → Gardien → serveur
    fenetreUi.ActivationHandler = () => gardien.ActiverAsync();

    // Widget : Suspendre / Reprendre / Quitter → Gardien → serveur
    fenetreUi.SuspendreHandler = async () =>

        // Le client n'a que le bouton Reprendre quand le poste est suspendu
        if (etatServeur == "suspendu") return await gardien.ReprendreSessionAsync();
        return await gardien.SuspendreSessionAsync();
    };

    fenetreUi.QuitterHandler = () => gardien.FermerSessionAsync();

    // Chat client → accueil
    fenetreUi.MessageHandler = texte => gardien.EnvoyerMessageAsync(texte);

    // Fin du décompte : c'est le Gardien qui éteint réellement Windows (et seulement si le serveur l'a demandé)
    fenetreUi.ExtinctionHandler = () =>

        Log("Décompte terminé : demande d'extinction envoyée au Gardien.");
        return gardien.Envoyer(new MessageCanal { Type = "extinction" });

    fenetreUi.MettreAJourBranding(AgentConfig.BrandingParDefaut);
    PousserInfos();

    // --- Messages du Gardien ---------------------------------------------------
    gardien.Connecte += () => { /* le Gardien répond en poussant branding + état */ };

    gardien.Deconnecte += () =>
    {
        etatServeur = liaison.Etat == EtatLiaison.CableDebranche ? "cable_debranche" : "serveur_injoignable";
        PousserInfos();
    };

    gardien.MessageRecu += m =>
    {
        switch (m.Type)
        {
            case "etat":
                poste = m.Poste ?? poste;
                ip = m.Ip ?? ip;
                etatServeur = m.Etat ?? "serveur_injoignable";
                if (etatServeur == "serveur_injoignable" && liaison.Etat == EtatLiaison.CableDebranche)
                    etatServeur = "cable_debranche";   // on distingue « câble débranché » de « serveur injoignable »
                PousserInfos();
                // Verrou absent = serveur injoignable : on garde l'état courant (mode dégradé)
                if (m.Verrou is bool doitEtreVerrouille)
                {
                    liaison.SignalerSucces();   // donnée fraîche du serveur : la liaison est bonne
                    fenetreUi.MettreAJourSession(m.Session);
                    fenetreUi.SyncFromServer(doitEtreVerrouille);
                }
                break;

            case "branding":
                fenetreUi.MettreAJourBranding(AgentConfig.DepuisMessage(m));
                break;

            case "cmd":
            {
                var (ok, detail) = Executer(m.Commande ?? "", m.Payload, fenetreUi);
                gardien.Envoyer(new MessageCanal { Type = "repcmd", Id = m.Id, Succes = ok, Message = detail });
                break;
            }
        }
    };

    _ = Task.Run(gardien.BoucleAsync);

    // Exécution d'une commande reçue du serveur (via le Gardien)
    (bool ok, string detail) Executer(string type, string? payload, ClientWindow fenetre)
    {
        Log($"Commande reçue : {type}");
        try
        {
            switch (type)
            {
                case "demarrer":
                case "debloquer":
                case "reprendre":
                    fenetre.UnlockScreen();
                    return (true, "Poste déverrouillé");

                case "verrouiller":
                case "suspendre":
                case "fermer":
                    fenetre.LockScreen();
                    return (true, "Poste verrouillé");

                case "desactiver":
                    // L'écran affiche tout de suite « Poste désactivé » (ACTIVER devient INDISPONIBLE)
                    etatServeur = "desactive";
                    PousserInfos();
                    fenetre.LockScreen();   // ferme aussi le widget s'il est ouvert
                    Reveiller();
                    return (true, "Poste désactivé et verrouillé");

                case "arreter":
                    fenetre.DemanderArret(DelaiArretSecondes);
                    return (true, $"Extinction dans {DelaiArretSecondes} s");

                case "recharger":
                {
                    var minutes = LireEntier(payload, "minutes_ajoutees");
                    fenetre.NotifierRecharge(minutes);
                    Reveiller();   // le widget se recale tout de suite sur le temps du serveur
                    return (true, "Temps ajouté");
                }

                case "message":
                {
                    var texte = LireTexte(payload, "message");
                    if (string.IsNullOrWhiteSpace(texte)) return (false, "Message vide");
                    fenetre.RecevoirMessage(texte.Trim());
                    return (true, "Message affiché");
                }

                default:
                    return (false, $"Commande inconnue : {type}");
            }
        }
        catch (Exception ex)
        {
            Log($"Exécution de '{type}' échouée : {ex.Message}");
            return (false, "Erreur d'exécution");
        }
    }

    uiThread.Join();
    Log(">>> Application terminée.");
}
catch (Exception ex)
{
    Log($">>> ERREUR FATALE : {ex}");
}

    private static Color Assombrir(Color c, float f) =>
        Color.FromArgb(c.A, (int)(c.R * f), (int)(c.G * f), (int)(c.B * f));

    /// <summary>
    /// Icône de l'application : le logo du cyber (sur fond blanc arrondi) s'il existe,
    /// sinon un écran blanc sur fond dégradé à la couleur du cyber. Mise en cache.
    /// </summary>
    public static Icon IconeApp()
    {
        if (_iconeApp != null) return _iconeApp;
        try
        {
            using var bmp = new Bitmap(64, 64);
            using (var g = Graphics.FromImage(bmp))
            {
                g.SmoothingMode = SmoothingMode.AntiAlias;
                g.InterpolationMode = InterpolationMode.HighQualityBicubic;
                g.Clear(Color.Transparent);

                var rect = new Rectangle(2, 2, 60, 60);
                const int r = 14;
                using var path = new GraphicsPath();
                path.AddArc(rect.X, rect.Y, r * 2, r * 2, 180, 90);
                path.AddArc(rect.Right - r * 2, rect.Y, r * 2, r * 2, 270, 90);
                path.AddArc(rect.Right - r * 2, rect.Bottom - r * 2, r * 2, r * 2, 0, 90);
                path.AddArc(rect.X, rect.Bottom - r * 2, r * 2, r * 2, 90, 90);
                path.CloseFigure();

                var logoDessine = false;
                if (_logoOctets is { Length: > 0 })
                {
                    try
                    {
                        using var ms = new MemoryStream(_logoOctets);
                        using var img = Image.FromStream(ms);
                        using (var blanc = new SolidBrush(Color.White)) g.FillPath(blanc, path);
                        var ratio = Math.Min(52f / img.Width, 52f / img.Height);
                        var w = img.Width * ratio;
                        var h = img.Height * ratio;
                        g.DrawImage(img, (64 - w) / 2f, (64 - h) / 2f, w, h);
                        logoDessine = true;
                    }
                    catch { logoDessine = false; }
                }

                if (!logoDessine)
                {
                    var accent = CouleurAccent();
                    using (var fond = new LinearGradientBrush(rect, accent, Assombrir(accent, 0.65f), 45f))
                        g.FillPath(fond, path);

                    using var stylo = new Pen(Color.White, 4f) { StartCap = LineCap.Round, EndCap = LineCap.Round };
                    g.DrawRectangle(stylo, 14, 15, 36, 24);   // écran
                    g.DrawLine(stylo, 32, 40, 32, 47);        // pied
                    g.DrawLine(stylo, 22, 48, 42, 48);        // socle
                }
            }

            var hIcone = bmp.GetHicon();
            try
            {
                using var temporaire = Icon.FromHandle(hIcone);
                _iconeApp = (Icon)temporaire.Clone();
            }
            finally { DestroyIcon(hIcone); }
        }
        catch
        {
            _iconeApp = SystemIcons.Application;
        }
        return _iconeApp;
    }

    /// <summary>
    /// Montant temps réel : STRICTEMENT max(minimum, minutes × prix/minute).
    /// Le paramètre « arrondi » des tarifs n'est volontairement PAS appliqué ici
    /// (même formule que le panneau admin et que PosteSessionService::montant() côté Laravel).
    /// </summary>
    private int MontantPourMinutes(int minutes)
    {
        var t = _tarif;
        long brut = (long)Math.Max(0, minutes) * t.ParMinute;
        return (int)Math.Max((long)t.Minimum, brut);
    }

    private void Rafraichir()
    {
        var delta = (int)Math.Max(0, (Environment.TickCount64 - _recuTick) / 1000);
        var forfait = _mode == "forfait" && _restantBase.HasValue;

        _lblCaption.Text = forfait ? "TEMPS RESTANT" : "TEMPS ÉCOULÉ";
        var secondes = forfait ? Math.Max(0, _restantBase!.Value - delta) : _ecouleBase + delta;
        _lblTemps.Text = Duree(secondes);
        _lblTemps.ForeColor = forfait && secondes <= 300 ? Color.FromArgb(248, 113, 113) : Ambre;

        if (_mode == "libre") _lblMontant.Text = "Sans facturation";
        else if (_mode == "forfait") _lblMontant.Text = $"Forfait : {FormaterAr(_forfaitTotal)}";
        else {
            var min = secondes / 60;
            _lblMontant.Text = "À payer : " + FormaterAr(MontantPourMinutes(min));
        }

        _lblDebut.Text = "Début : " + _debut;

        if (!forfait) return;

        if (secondes > 300) _alerte5 = false;
        else if (secondes > 0 && !_alerte5) { _alerte5 = true; CinqMinutes?.Invoke(); }

        if (secondes > 0) _finSignalee = false;
        else if (!_finSignalee) { _finSignalee = true; ForfaitTermine?.Invoke(); }
    }

    private static string Duree(int secondes) { var t = TimeSpan.FromSeconds(Math.Max(0, secondes)); return $"{(int)t.TotalHours:00}:{t.Minutes:00}:{t.Seconds:00}"; }
    private static string FormaterHeure(string? iso) => DateTimeOffset.TryParse(iso, CultureInfo.InvariantCulture, DateTimeStyles.None, out var d) ? d.ToLocalTime().ToString("HH:mm", CultureInfo.InvariantCulture) : "—";

    private async Task EnvoyerChatAsync()
    {
        var texte = _chatSaisie.Text.Trim();
        if (texte.Length == 0 || MessageHandler == null || _occupe) return;
        _chatSaisie.Clear();
        await MessageHandler(texte);
    }

    private async Task ExecuterAsync(Func<Task<ResultatAction>>? handler, string question, string titre)
    {
        if (_occupe || handler == null) return;
        var choix = MessageBox.Show(this, question, titre, MessageBoxButtons.YesNo, MessageBoxIcon.Question, MessageBoxDefaultButton.Button2);
        if (choix != DialogResult.Yes || IsDisposed) return;

        _occupe = true;
        try {
            var r = await handler();
            if (!IsDisposed && !r.Succes) MessageBox.Show(this, r.Message, titre, MessageBoxButtons.OK, MessageBoxIcon.Warning);
        } catch (Exception ex) { ClientWindow.Journal($"[Widget] Erreur : {ex.Message}"); }
        finally { if (!IsDisposed) _occupe = false; }
    }

using System.Net.NetworkInformation;
using System.Net.Sockets;

namespace CyberManager.Agent;

public enum EtatLiaison { Ok, CableDebranche, ServeurInjoignable }

/// <summary>
/// Surveille la liaison avec le serveur. Le délai de grâce (15 s par défaut, fourni par le serveur
/// dans « delai_grace_agent_s ») est inférieur à la tolérance du serveur (30 s).
/// L'événement EtatChange est levé sur un thread du pool : le consommateur doit repasser sur le thread UI.
/// </summary>
public sealed class SurveillanceLiaison : IDisposable
{
    private readonly object _verrou = new();
    private DateTime _dernierSucces = DateTime.UtcNow;
    private EtatLiaison _etat = EtatLiaison.Ok;
    private System.Threading.Timer? _minuteur;

    public TimeSpan DelaiGrace { get; set; } = TimeSpan.FromSeconds(15);

    public event Action<EtatLiaison, string>? EtatChange;

    public EtatLiaison Etat { get { lock (_verrou) return _etat; } }

    /// <summary>Faux dès que la grâce est dépassée : chrono et montant figés, poste verrouillé.</summary>
    public bool UsageAutorise => Etat == EtatLiaison.Ok;

    public void Demarrer() => _minuteur ??= new System.Threading.Timer(_ => Evaluer(), null, 1000, 1000);

    /// <summary>À appeler après chaque heartbeat réussi (en passant delai_grace_agent_s de la réponse).</summary>
    public void SignalerSucces(int? delaiGraceServeurSecondes = null)
    {
        lock (_verrou) _dernierSucces = DateTime.UtcNow;
        if (delaiGraceServeurSecondes is int s && s > 0) DelaiGrace = TimeSpan.FromSeconds(s);
        Evaluer();
    }

    private void Evaluer()
    {
        TimeSpan silence;
        lock (_verrou) silence = DateTime.UtcNow - _dernierSucces;

        var nouveau = silence < DelaiGrace
            ? EtatLiaison.Ok
            : (ReseauLocalPresent() ? EtatLiaison.ServeurInjoignable : EtatLiaison.CableDebranche);

        lock (_verrou)
        {
            if (nouveau == _etat) return;
            _etat = nouveau;
        }

        var message = nouveau switch
        {
            EtatLiaison.CableDebranche     => "Câble réseau débranché ou Wi-Fi coupé. Le poste est verrouillé, votre temps est suspendu. Reprise automatique au retour du réseau.",
            EtatLiaison.ServeurInjoignable => "Le serveur de la caisse ne répond pas. Le poste est verrouillé, votre temps est suspendu. Reprise automatique au retour du serveur.",
            _                              => "Connexion rétablie.",
        };
        EtatChange?.Invoke(nouveau, message);
    }

    private static bool ReseauLocalPresent()
    {
        try
        {
            foreach (var ni in NetworkInterface.GetAllNetworkInterfaces())
            {
                if (ni.OperationalStatus != OperationalStatus.Up) continue;
                if (ni.NetworkInterfaceType is NetworkInterfaceType.Loopback or NetworkInterfaceType.Tunnel) continue;

                foreach (var a in ni.GetIPProperties().UnicastAddresses)
                {
                    if (a.Address.AddressFamily == AddressFamily.InterNetwork
                        && !a.Address.ToString().StartsWith("169.254.", StringComparison.Ordinal))
                        return true;
                }
            }
        }
        catch { /* en cas de doute : on considère le réseau présent */ return true; }
        return false;
    }

    public void Dispose() => _minuteur?.Dispose();
}

using System.Runtime.InteropServices;

public enum NiveauToast { Info, Message, Alerte, Danger }

/// <summary>
/// Notification flottante non bloquante (bas-droite, au-dessus du widget), avec animation
/// d'entrée / sortie. N'a jamais le focus. Thread UI uniquement.
/// </summary>
public class ToastForm : Form
{
    private static readonly List<ToastForm> Actifs = new();

    private static readonly Color Fond = Color.FromArgb(19, 28, 58);
    private static readonly Color Texte = Color.FromArgb(226, 232, 240);

    private readonly System.Windows.Forms.Timer _anim = new() { Interval = 15 };
    private readonly System.Windows.Forms.Timer _vie;
    private readonly string _titre, _texte;
    private readonly Color _accent;
    private readonly float _k;
    private int _etape, _cibleX;
    private bool _sortie;

    /// <summary>Appelé si l'utilisateur clique sur la notification.</summary>
    public Action? Clic { get; set; }

    private ToastForm(string titre, string texte, NiveauToast niveau, int dureeMs)
    {
        _k = Math.Max(1f, DeviceDpi / 96f);
        _titre = titre;
        _texte = texte;
        _accent = niveau switch
        {
            NiveauToast.Danger => Color.FromArgb(220, 38, 38),
            NiveauToast.Alerte => Color.FromArgb(245, 158, 11),
            NiveauToast.Message => Color.FromArgb(139, 92, 246),
            _ => Color.FromArgb(56, 189, 248)
        };

        FormBorderStyle = FormBorderStyle.None;
        StartPosition = FormStartPosition.Manual;
        ShowInTaskbar = false;
        TopMost = true;
        BackColor = Fond;
        DoubleBuffered = true;
        Cursor = Cursors.Hand;
        ClientSize = new Size(S(310), S(88));
        Opacity = 0;

        _anim.Tick += (_, _) => Animer();
        _vie = new System.Windows.Forms.Timer { Interval = Math.Max(1500, dureeMs) };
        _vie.Tick += (_, _) => { _vie.Stop(); Fermer(false); };
    }

    /// <summary>Affiche une notification (3 maximum à l'écran).</summary>
    public static ToastForm Afficher(string titre, string texte, NiveauToast niveau = NiveauToast.Info,
        int dureeMs = 7000, Action? clic = null)
    {
        if (Actifs.Count >= 3) Actifs[0].Fermer(true);

        var t = new ToastForm(titre, texte, niveau, dureeMs) { Clic = clic };
        Actifs.Add(t);
        Repositionner();
        t.Show();
        t._anim.Start();
        t._vie.Start();

        try
        {
            if (niveau is NiveauToast.Alerte or NiveauToast.Danger) System.Media.SystemSounds.Exclamation.Play();
            else if (niveau == NiveauToast.Message) System.Media.SystemSounds.Asterisk.Play();
        }
        catch { }
        return t;
    }

    public static void FermerTous()
    {
        foreach (var t in Actifs.ToArray()) t.Fermer(true);
    }

    public void Fermer(bool immediat)
    {
        if (IsDisposed) return;
        if (immediat) { Close(); return; }
        _sortie = true;
        _anim.Start();
    }

    // ------------------------------------------------------------------

    private static void Repositionner()
    {
        var zone = Screen.PrimaryScreen?.WorkingArea ?? SystemInformation.WorkingArea;
        var y = 0;
        // Les plus récentes en bas, au-dessus de la zone du widget (180 px + marges)
        for (var i = Actifs.Count - 1; i >= 0; i--)
        {
            var t = Actifs[i];
            var bas = zone.Bottom - t.S(12 + 180 + 8) - y;
            t._cibleX = zone.Right - t.Width - t.S(12);
            t.Location = new Point(t._cibleX + (t._etape < 14 ? t.S(40) : 0), bas - t.Height);
            y += t.Height + t.S(8);
        }
    }

    private void Animer()
    {
        if (IsDisposed) return;
        _etape += _sortie ? -1 : 1;
        var t = Math.Clamp(_etape / 14.0, 0, 1);
        var e = 1 - Math.Pow(1 - t, 3);
        Opacity = e;
        Left = _cibleX + (int)Math.Round(S(40) * (1 - e));

        if (_sortie && _etape <= 0) { Close(); return; }
        if (!_sortie && _etape >= 14) _anim.Stop();
    }

    protected override bool ShowWithoutActivation => true;

    protected override CreateParams CreateParams
    {
        get
        {
            var cp = base.CreateParams;
            cp.ExStyle |= 0x80 | 0x08000000;   // WS_EX_TOOLWINDOW | WS_EX_NOACTIVATE
            return cp;
        }
    }

    protected override void OnMouseClick(MouseEventArgs e)
    {
        base.OnMouseClick(e);
        try { Clic?.Invoke(); } catch { }
        Fermer(false);
    }

    protected override void OnPaint(PaintEventArgs e)
    {
        base.OnPaint(e);
        using var accent = new SolidBrush(_accent);
        e.Graphics.FillRectangle(accent, 0, 0, S(5), Height);
        using var pen = new Pen(Color.FromArgb(90, 139, 92, 246));
        e.Graphics.DrawRectangle(pen, 0, 0, Width - 1, Height - 1);

        using var fTitre = new Font("Segoe UI", 9.5f, FontStyle.Bold);
        using var fTexte = new Font("Segoe UI", 9f);
        TextRenderer.DrawText(e.Graphics, _titre, fTitre,
            new Rectangle(S(16), S(8), Width - S(26), S(20)), Color.White,
            TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding);
        TextRenderer.DrawText(e.Graphics, _texte, fTexte,
            new Rectangle(S(16), S(30), Width - S(26), Height - S(38)), Texte,
            TextFormatFlags.WordBreak | TextFormatFlags.EndEllipsis | TextFormatFlags.NoPadding);
    }

    protected override void OnFormClosed(FormClosedEventArgs e)
    {
        Actifs.Remove(this);
        _anim.Dispose();
        _vie.Dispose();
        Repositionner();
        base.OnFormClosed(e);
    }

    private int S(int v) => (int)Math.Round(v * _k);
}

using System.Drawing.Drawing2D;
using System.Drawing.Text;

/// <summary>
/// Voyant « Serveur : connecté ✓ / injoignable ». Alimenté UNIQUEMENT par l'état que le Gardien tire du heartbeat :
/// il ne dépend jamais de la sonde NCSI de Windows.
/// Anti-clignotement : vert immédiat ; rouge seulement après 3 s d'injoignabilité ; redessin uniquement si l'état change.
/// </summary>
public sealed class VoyantServeur : Control
{
    private enum Etat { Inconnu, Connecte, Injoignable }

    private static readonly Color Fond = Color.FromArgb(19, 28, 58);
    private static readonly Color Vert = Color.FromArgb(34, 197, 94);
    private static readonly Color Rouge = Color.FromArgb(239, 68, 68);
    private static readonly Color Gris = Color.FromArgb(148, 163, 184);

    private readonly Font _police;
    private readonly System.Windows.Forms.Timer _delai = new() { Interval = 3000 };
    private Etat _etat = Etat.Inconnu;
    private bool _dernierConnecte;

    public VoyantServeur(float taillePolice)
    {
        SetStyle(ControlStyles.UserPaint | ControlStyles.AllPaintingInWmPaint | ControlStyles.OptimizedDoubleBuffer | ControlStyles.ResizeRedraw, true);
        _police = new Font("Segoe UI", taillePolice, FontStyle.Bold);
        TabStop = false;
        AccessibleName = "État de la liaison avec le serveur";
        _delai.Tick += (_, _) =>
        {
            _delai.Stop();
            if (!_dernierConnecte) Appliquer(Etat.Injoignable);
        };
    }

    /// <summary>À appeler sur le thread UI. « immediat » : pas de délai avant le rouge (création du widget).</summary>
    public void Definir(bool connecte, bool immediat = false)
    {
        _dernierConnecte = connecte;
        if (connecte) { _delai.Stop(); Appliquer(Etat.Connecte); return; }
        if (immediat) { _delai.Stop(); Appliquer(Etat.Injoignable); return; }
        if (_etat != Etat.Injoignable && !_delai.Enabled) _delai.Start();
    }

    private void Appliquer(Etat e)
    {
        if (_etat == e) return;
        _etat = e;
        Invalidate();
    }

    private static GraphicsPath Pilule(Rectangle r)
    {
        var p = new GraphicsPath();
        var d = Math.Max(2, r.Height);
        p.AddArc(r.X, r.Y, d, d, 90, 180);
        p.AddArc(r.Right - d, r.Y, d, d, 270, 180);
        p.CloseFigure();
        return p;
    }

    protected override void OnResize(EventArgs e)
    {
        base.OnResize(e);
        if (Width <= 0 || Height <= 0) return;
        using var chemin = Pilule(new Rectangle(0, 0, Width, Height));
        var ancienne = Region;
        Region = new Region(chemin);
        ancienne?.Dispose();
    }

    protected override void OnPaint(PaintEventArgs e)
    {
        var g = e.Graphics;
        g.SmoothingMode = SmoothingMode.AntiAlias;
        g.TextRenderingHint = TextRenderingHint.ClearTypeGridFit;
        g.Clear(Fond);

        var couleur = _etat switch { Etat.Connecte => Vert, Etat.Injoignable => Rouge, _ => Gris };
        var texte = _etat switch
        {
            Etat.Connecte => "Serveur : connecté",
            Etat.Injoignable => "Serveur : injoignable",
            _ => "Serveur : vérification…",
        };

        using (var bordure = new Pen(couleur, 2f))
        using (var chemin = Pilule(new Rectangle(1, 1, Width - 3, Height - 3)))
            g.DrawPath(bordure, chemin);

        var d = Math.Max(8, Height - 10);
        var cercle = new Rectangle(8, (Height - d) / 2, d, d);

        if (_etat == Etat.Inconnu)
        {
            using var p = new Pen(couleur, 2f);
            g.DrawEllipse(p, cercle);
        }
        else
        {
            using (var b = new SolidBrush(couleur)) g.FillEllipse(b, cercle);
            using var trait = new Pen(Color.White, Math.Max(1.5f, d / 9f)) { StartCap = LineCap.Round, EndCap = LineCap.Round, LineJoin = LineJoin.Round };
            if (_etat == Etat.Connecte)
            {
                g.DrawLines(trait, new[]
                {
                    new PointF(cercle.X + d * 0.26f, cercle.Y + d * 0.52f),
                    new PointF(cercle.X + d * 0.43f, cercle.Y + d * 0.70f),
                    new PointF(cercle.X + d * 0.74f, cercle.Y + d * 0.32f),
                });
            }
            else
            {
                g.DrawLine(trait, cercle.X + d * 0.3f, cercle.Y + d * 0.3f, cercle.X + d * 0.7f, cercle.Y + d * 0.7f);
                g.DrawLine(trait, cercle.X + d * 0.7f, cercle.Y + d * 0.3f, cercle.X + d * 0.3f, cercle.Y + d * 0.7f);
            }
        }

        var zone = new Rectangle(cercle.Right + 8, 0, Math.Max(10, Width - cercle.Right - 14), Height);
        using var format = new StringFormat { LineAlignment = StringAlignment.Center, Alignment = StringAlignment.Near, Trimming = StringTrimming.EllipsisCharacter, FormatFlags = StringFormatFlags.NoWrap };
        using var pinceau = new SolidBrush(_etat == Etat.Inconnu ? Gris : Color.White);
        g.DrawString(texte, _police, pinceau, zone, format);
    }

    protected override void Dispose(bool disposing)
    {
        if (disposing) { _delai.Dispose(); _police.Dispose(); }
        base.Dispose(disposing);
    }
}
